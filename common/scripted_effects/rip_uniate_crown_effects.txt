# The missing country-scope bridge called by both signing options in
# uniate_crown.2. A signature changes the state confession and one capital;
# the parishes still require the ordinary conversion system.
rip_ucr_take_the_church_effect = {
	if = {
		limit = { rip_ucr_can_sign_the_union = yes }
		set_country_flag = rip_ucr_church_taken
		set_country_flag = pursuing_uniate_union
		set_country_flag = union_of_brest_happened
		change_religion = greek_catholic
		add_country_modifier = { name = rip_ucr_church_of_the_palace duration = -1 }
		activate_greek_catholic_reformation = yes
		capital_scope = {
			if = {
				limit = { religion = orthodox }
				change_religion = greek_catholic
			}
		}
	}
}
