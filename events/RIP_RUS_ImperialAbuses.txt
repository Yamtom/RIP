		name = rip_rus_reckoning.2.a

				add_unrest = 6
				add_province_modifier = {
					name = rip_rus_silenced_presses
					duration = 7300
				}
			}
		}
		else = {
			if = {
				limit = { owns = 289 }
				289 = {
					add_unrest = 6
					add_province_modifier = {
						name = rip_rus_silenced_presses
						duration = 7300
					}
				}
			}
			else = {
				if = {
					limit = { any_owned_province = { rip_diaspora_eligible_culture_province = yes } }
					random_owned_province = {
						limit = { rip_diaspora_eligible_culture_province = yes }
						add_unrest = 6
						add_province_modifier = {
							name = rip_rus_silenced_presses
							duration = 7300
						}
					}
				}
			}
		}
		add_country_modifier = {
			name = rip_rus_imperial_uniformity
			duration = 5475
		}
	}
}

# --- Disaster: rip_rus_abolition_of_hetmanate (.3 start, .4 pulse, .5 end)

country_event = {
	id = rip_rus_reckoning.3
	title = rip_rus_reckoning.3.t
	desc = rip_rus_reckoning.3.d
	picture = COUNTRY_COLLAPSE_eventPicture
	is_triggered_only = yes

	option = {
		name = rip_rus_reckoning.3.a
		ai_chance = { factor = 60 }
		add_absolutism = 10
		add_stability = -1
		if = {
			limit = {
				any_owned_province = {
					OR = {
						area = east_dniepr_area
						area = chernigov_area
						area = zaporizhia_area
						area = west_dniepr_area
					}
				}
			}
			random_owned_province = {
				limit = {
					OR = {
						area = east_dniepr_area
						area = chernigov_area
						area = zaporizhia_area
						area = west_dniepr_area
					}
				}
				add_devastation = 30
				spawn_rebels = {
					type = particularist_rebels
					size = 1
				}
			}
		}
	}
	option = {
		name = rip_rus_reckoning.3.b
		ai_chance = { factor = 40 }
		add_absolutism = -10
		add_adm_power = -50
		if = {
			limit = { any_owned_province = { region = ruthenia_region } }
			random_owned_province = {
				limit = { region = ruthenia_region }
				add_local_autonomy = 20
			}
		}
	}
}
