chr_left_bank_monastery_investment = {
	every_owned_province = {
		limit = {
			OR = {
				area = east_dniepr_area
				area = sloboda_ukraine_area
			}
			has_building = temple
		}
		if = {
			limit = { has_province_modifier = chr_fortified_monastery_lvl1 }
			hidden_effect = { remove_province_modifier = chr_fortified_monastery_lvl1 }
			add_province_modifier = {
				name = chr_fortified_monastery_lvl2
				duration = -1
			}
		}
		else_if = {
			limit = {
				NOT = { has_province_modifier = chr_fortified_monastery_lvl2 }
			}
			add_province_modifier = {
				name = chr_fortified_monastery_lvl1
				duration = -1
			}
		}
	}
}

# Reconcile the permanent grain bonus after loading an older save. Ownership
# and trade-good changes are handled immediately by the vanilla on-action
# override; this effect is only the save-compatible backstop.
chr_refresh_grain_directorate_effect = {
	if = {
		limit = {
			has_reform = chr_grain_directorate_reform
		}
		every_owned_province = {
			if = {
				limit = {
					trade_goods = grain
					NOT = { has_province_modifier = chr_grain_directorate_province }
				}
				add_province_modifier = {
					name = chr_grain_directorate_province
					duration = -1
					hidden = yes
				}
			}
			else_if = {
				limit = {
					NOT = { trade_goods = grain }
					has_province_modifier = chr_grain_directorate_province
				}
				remove_province_modifier = chr_grain_directorate_province
			}
		}
	}
	else = {
		every_owned_province = {
			limit = {
				has_province_modifier = chr_grain_directorate_province
			}
			remove_province_modifier = chr_grain_directorate_province
		}
	}
}

chr_border_compact_forge_effect = {
	if = {
		limit = {
			chr_can_forge_border_compact = yes
		}
		random_country = {
			limit = {
				chr_border_compact_partner_candidate = yes
			}
			save_event_target_as = chr_border_compact_partner
			add_opinion = {
				who = ROOT
				modifier = chr_border_compact_support
			}
			if = {
				limit = {
					has_opinion = { who = ROOT value = 80 }
					trust = { who = ROOT value = 50 }
				}
				custom_tooltip = chr_border_compacts_confidence_tt
				add_country_modifier = {
					name = chr_border_compact_confidence_partner
					duration = 1825
				}
				ROOT = {
					add_country_modifier = {
						name = chr_border_compact_confidence
						duration = 1825
					}
				}
			}
			else = {
				custom_tooltip = chr_border_compacts_tension_tt
				add_country_modifier = {
					name = chr_border_compact_watchposts
					duration = 1095
				}
				ROOT = {
					add_country_modifier = {
						name = chr_border_compact_tension
						duration = 1095
					}
				}
			}
		}
		if = {
			limit = {
				exists = event_target:chr_border_compact_partner
			}
			add_dip_power = -40
			add_country_modifier = {
				name = chr_border_compact_cooldown
				duration = 1825
			}
			hidden_effect = {
				set_country_flag = chr_border_compacts_pending_review
				country_event = {
					id = chr_flavor.36
					days = 540
				}
			}
		}
	}
}
