# Country scope. Only an explicit alternative-history schism authorizes separation.
rip_church_ro_history_monthly_effect = {
 rip_ro_reform_desire_migrate_effect = yes
 # One-time migration to the institutional confession; parish majorities stay.
 if = { limit = { religion = orthodox has_country_flag = rip_ro_stage_autocephaly NOT = { has_country_flag = rip_church_confession_v4 } }
  rip_church_v3_initialize_effect = yes
  rip_faith_adopt_muscovite_church_effect = yes
 }
 if = { limit = { rip_church_ro_jurisdiction = yes }
  set_country_flag = rip_church_confession_v4
 }
 if = { limit = { rip_church_ro_jurisdiction = yes has_country_flag = rip_church_ro_provisional is_year = 1593 }
  rip_church_ro_recognize_effect = yes
 }
 rip_church_ro_refresh_communion_effect = yes
 # Institutional policy cannot survive a conversion out of this church family.
 if = { limit = { NOT = { rip_church_ro_jurisdiction = yes } }
  remove_country_modifier = rip_ro_synodal_government
  remove_country_modifier = rip_ro_preserved_patriarchate
  remove_country_modifier = rip_ro_patriarchal_vacancy
  remove_country_modifier = rip_ro_ritual_compromise
 }
}

# Target only Orthodox parishes in a Muscovite-confession country. Native EU4
# still calls a different religion 'heretic'; these supported local modifiers
# neutralize its practical penalties without tolerating Catholicism by accident.
rip_church_ro_refresh_communion_effect = {
 remove_country_modifier = rip_ro_unbroken_communion
 every_owned_province = {
  if = { limit = { religion = orthodox owner = { religion = russian_orthodox NOT = { has_country_flag = rip_church_ro_schismatic } } }
   if = { limit = { owner = { OR = { has_country_modifier = orthodox_inquisition has_country_modifier = persecution_of_heretics has_country_modifier = forced_russification } } }
    remove_province_modifier = rip_ro_shared_communion
    if = { limit = { NOT = { has_province_modifier = rip_ro_restricted_communion } }
     add_province_modifier = { name = rip_ro_restricted_communion duration = -1 }
    }
   }
   else = {
    remove_province_modifier = rip_ro_restricted_communion
    if = { limit = { NOT = { has_province_modifier = rip_ro_shared_communion } }
     add_province_modifier = { name = rip_ro_shared_communion duration = -1 }
    }
   }
  }
  else = {
   remove_province_modifier = rip_ro_shared_communion
   remove_province_modifier = rip_ro_restricted_communion
  }
 }
}

rip_church_ro_formal_raskol_effect = {
 if = { limit = { rip_church_ro_jurisdiction = yes is_year = 1666 has_country_flag = rip_ro_books_reformed }
  set_global_flag = raskol_schism
  set_country_flag = rip_ro_stage_raskol
  rip_ob_begin_schism_effect = yes
  # A dispute over rites within Orthodoxy, not a new state confession.
  rip_church_ro_refresh_communion_effect = yes
 }
}

rip_church_ro_ritual_compromise_effect = {
 # The pre-schism compromise has its own modifier, not an Old Believer policy.
 clr_country_flag = rip_ob_toleration
 clr_country_flag = rip_ob_repression
 clr_country_flag = rip_ob_policy_changed
 remove_country_modifier = rip_ob_tolerant_policy
 remove_country_modifier = rip_ob_repressive_policy
 every_owned_province = {
  remove_province_modifier = rip_ob_local_toleration
  remove_province_modifier = rip_ob_local_repression
 }
 clr_country_flag = rip_ro_stage_raskol
 set_country_flag = rip_ro_ritual_compromise
 set_country_flag = rip_ro_raskol_settled
 remove_country_modifier = suppression_old_believers
 remove_country_modifier = old_believer_underground
 add_country_modifier = { name = rip_ro_ritual_compromise duration = -1 }
 add_patriarch_authority = -0.1
 add_stability = 1
 rip_church_ro_refresh_communion_effect = yes
}

# Existing saves acquire the milestones already recorded in their church flags.
rip_ro_reform_desire_migrate_effect = {
 if = {
  limit = {
   rip_church_ro_jurisdiction = yes
   NOT = { has_country_flag = rip_ro_reform_desire_migration_done }
  }
  set_country_flag = rip_ro_reform_desire_migration_done
  if = { limit = { has_country_flag = rip_ro_stage_autocephaly } rip_ro_reform_desire_1448_effect = yes }
  if = {
   limit = {
    OR = {
     has_country_flag = rip_ro_stage_patriarchate
     has_country_flag = rip_church_ro_provisional
     has_country_flag = rip_church_ro_recognized
     has_country_flag = rip_church_ro_schismatic
    }
   }
   rip_ro_reform_desire_1589_effect = yes
  }
  if = { limit = { has_country_flag = rip_church_ro_recognized } rip_ro_reform_desire_1593_effect = yes }
  if = {
   limit = { OR = { has_country_flag = rip_ro_books_reformed has_country_flag = rip_ro_ritual_compromise } }
   rip_ro_reform_desire_1653_effect = yes
  }
  if = {
   limit = {
    OR = {
     has_country_flag = rip_ro_stage_raskol
     AND = {
      has_country_flag = rip_ro_books_reformed
      has_country_flag = rip_ro_ritual_compromise
     }
    }
   }
   rip_ro_reform_desire_1666_effect = yes
  }
  if = {
   limit = {
    OR = {
     has_country_flag = rip_ro_patriarchal_interregnum
     has_country_flag = rip_ro_synodal_settlement
     has_country_modifier = rip_ro_patriarchal_vacancy
     has_country_modifier = rip_ro_preserved_patriarchate
     has_country_modifier = rip_ro_synodal_government
    }
   }
   rip_ro_reform_desire_1700_effect = yes
  }
  if = {
   limit = {
    OR = {
     has_country_flag = rip_ro_synodal_government
     AND = {
      is_year = 1721
      has_country_flag = rip_ro_synodal_settlement
      NOT = { has_country_flag = rip_ro_patriarchal_interregnum }
     }
    }
   }
   rip_ro_reform_desire_1721_effect = yes
  }
 }
}
