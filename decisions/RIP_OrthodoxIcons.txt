	# Russian Orthodoxy keeps has_patriarchs, so patriarch authority still
	# accrues and Consecrate Patriarch still sits in the province view. What it
	# cannot keep is the icon switcher: that lives in orthodox_specific_window,
	# and this faith spends its one panel on holy sites and blessings instead -
	# see the note in common/religions/russian_orthodox.txt.
	#
	# So the icons are bought here on vanilla's own terms: 0.1 patriarch
	# authority apiece (ORTHODOX_ICON_AUTHORITY_COST), twenty years apiece
	# (ORTHODOX_ICON_DURATION_MONTHS = 240, hence 7300 days), one at a time,
	# with vanilla's payloads unchanged.

	rip_ro_icon_michael = {
		potential = {
			religion = russian_orthodox

		}
		allow = {
			patriarch_authority = 0.1
		}
		effect = {
			add_patriarch_authority = -0.1
			rip_ro_clear_the_icons_effect = yes
			add_country_modifier = {
				name = rip_ro_icon_michael

			}
		}

			factor = 1
			modifier = {
				factor = 3
