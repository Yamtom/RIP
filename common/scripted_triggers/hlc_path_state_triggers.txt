# RIP - hlc path state triggers.

hlc_has_any_path = {
	OR = {
		has_country_flag = rip_galicia_path_austria
		has_country_flag = rip_galicia_path_poland
		has_country_flag = rip_galicia_path_boyar
		has_country_flag = rip_galicia_path_metropolitanate
	}
}

hlc_has_poland_path = {
	has_country_flag = rip_galicia_path_poland
}

hlc_has_austria_path = {
	has_country_flag = rip_galicia_path_austria
}

# decisions/HLCRepublicPaths.txt - the boyars depose the dynasty outright
# instead of settling into either patron's institutions.
hlc_has_boyar_path = {
	has_country_flag = rip_galicia_path_boyar
}

# decisions/HLCRepublicPaths.txt - the Metropolitan of Halych becomes head of
# state. Requires Orthodox or Greek Catholic on its own decision's potential;
# folded into hlc_has_any_path above so the founding event does not re-offer
# the Poland/Austria/Boyar forks afterward.
hlc_has_metropolitanate_path = {
	has_country_flag = rip_galicia_path_metropolitanate
}

hlc_boyar_succession_crisis = {
	OR = {
		AND = {
			NOT = { has_heir = yes }
			legitimacy < 60
		}
		legitimacy < 50
	}
}

hlc_metropolitan_authority_ready = {
	OR = {
		AND = {
			religion = orthodox
			patriarch_authority = 0.5
		}
		AND = {
			religion = greek_catholic
			rip_gc_hierarchy_standing_mid = yes
		}
	}
}

hlc_has_confirmed_metropolitanate_path = {

# The rip_vol_origin_* names are retained for existing saves. They now mark
# lineage through both supported Ruthenian formables: VOL and UKR.
west_ukraine_hlc_origin_country = {
	OR = {
		tag = HLC
		AND = {
			OR = {
				tag = VOL
				tag = UKR
			}
			has_country_flag = rip_vol_origin_hlc
		}
	}
}

west_ukraine_hlc_legacy_missions = {
	OR = {
		west_ukraine_hlc_origin_country = yes
		AND = {
			tag = VOL
			OR = {
				hlc_has_any_path = yes
				hlc_has_confessional_settlement = yes
				hlc_has_constitution_settlement = yes
				mission_completed = hlc_halych_sejm_mission
				mission_completed = hlc_carpathian_bulwark
				mission_completed = hlc_confessional_compact
				mission_completed = hlc_constitutional_settlement
			}
		}
	}
}

west_ukraine_hlc_vol_starter = {
	OR = {
		tag = HLC
		AND = {
			tag = VOL
			NOT = { has_country_flag = rip_vol_origin_vln }
		}
	}
}
