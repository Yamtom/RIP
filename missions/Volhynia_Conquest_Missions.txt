	VOL_demolish_poland = {
		icon = mission_cannons_firing
		required_missions = { VOL_integrate_north }
		position = 8

		trigger = {
			custom_trigger_tooltip = {
				tooltip = vol_poland_hegemony_broken_tt
				OR = {
					NOT = { exists = POL }
					POL = { is_subject_of = ROOT }
					POL = {
						NOT = { any_owned_province = { region = ruthenia_region } }
						NOT = { any_owned_province = { area = red_ruthenia_area } }
						NOT = { any_owned_province = { area = malopolska_area } }
						NOT = { any_owned_province = { area = sandomierz_area } }
						NOT = { is_great_power = yes }
						NOT = {
							army_strength = {
								who = ROOT
								value = 0.5
							}
						}
					}
				}
			}
			OR = {
				AND = {
					alliance_with = HLC
					HLC = {
						num_of_owned_provinces_with = {
							value = 5
						}
						custom_trigger_tooltip = {
							tooltip = vol_demolish_poland_hlc_path_req_tt
							hlc_has_any_path = yes
						}
					}
				}
				custom_trigger_tooltip = {
					tooltip = vol_demolish_poland_military_route_req_tt
					AND = {
						total_development = 250
						army_size = 45
						army_tradition = 40
					}
				}
			}
		}
		effect = {
			add_prestige = 25
			add_mil_power = 100
			if = {
				limit = { alliance_with = HLC }
				add_dip_power = 50
			}
			else = {
				add_army_tradition = 5
			}
			add_prestige = 10
		}
	}
}
