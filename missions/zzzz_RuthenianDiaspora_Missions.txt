# Load after national series sharing columns 2-4, including the zzz_ transitions.
# The tree exists twice, because columns stack one series under another and the
# national trees differ in height (up to row 22 for CHR):
#   RIP_diaspora_missions_*        rows 24-32, for every country except those below;
#   RIP_diaspora_kyiv_missions_*   rows 15-23, same missions with the suffix _kyiv, for
#                                  rip_diaspora_short_tree (KIE, KRU, UKR), whose trees
#                                  end at row 12 in these columns.
# Both copies share the flags set by their effects, so the decisions in
# decisions/RIP_RuthenianDiaspora.txt serve either. Titles and descriptions of the
# copy are references to the originals (localisation/RIP_shared_ruthenian_expeditions).
# tests/check_diaspora_layout.py keeps the two copies identical apart from ids and rows.
RIP_diaspora_missions_1 = {
	slot = 3
	generic = no
	ai = yes
	potential = {
		NOT = { map_setup = map_setup_random }
		rip_diaspora_eligible_culture = yes
		government_rank = 2
		NOT = { rip_diaspora_short_tree = yes }
	}
	has_country_shield = yes

	rip_diaspora_root = {
		icon = mission_exploration
		position = 24
		trigger = {
			# The tree opens with the age of revolutions, in 1750 at the latest, or as
			# soon as the realm's diplomats reach technology 23: whichever comes first.
			# A hard 1800 left nearly every campaign without the branch.
			OR = {
				is_year = 1750
				current_age = age_of_revolutions
				dip_tech = 23
			}
			is_subject = no
			any_owned_province = {
				has_port = yes
			}
		}
		effect = {
			add_prestige = 5
			set_country_flag = rip_diaspora_root_flag
		}
	}

	rip_lysianskyi_circumnavigation = {
		icon = mission_explore_the_world
		required_missions = { rip_diaspora_root }
		position = 26
		trigger = {
			is_year = 1806
			has_country_flag = circumnavigation_done
		}
		effect = {
			add_prestige = 15
			add_country_modifier = {
				name = rip_diaspora_of_rus_legacy
				duration = 3650
			}
			set_country_flag = rip_lysianskyi_circumnavigation_flag
		}
	}

	rip_diaspora_of_rus = {
		icon = mission_new_world_power
		required_missions = { rip_diaspora_root }
		position = 32
		trigger = {
			is_year = 1806
			OR = {
				has_country_flag = rip_diaspora_usa_flag
				has_country_flag = rip_diaspora_usa_settlers_flag
			}
			OR = {
				has_country_flag = rip_diaspora_south_america_flag
				has_country_flag = rip_diaspora_south_america_settlers_flag
			}
			OR = {
				has_country_flag = rip_diaspora_australia_flag
				has_country_flag = rip_diaspora_australia_settlers_flag
			}
			OR = {
				has_country_flag = rip_green_wedge_flag
				has_country_flag = rip_green_wedge_settlers_flag
				2447 = {
					rip_diaspora_established_community = yes
				}
			}
			OR = {
				has_country_flag = rip_grey_wedge_flag
				has_country_flag = rip_grey_wedge_settlers_flag
				# South-western Siberia and the northern Kazakh steppe: the Irtysh,
				# Ishim and Akmola areas and the Kazakh steppe itself.
				yrtesh_area = {
					rip_diaspora_established_community = yes
				}
				ishim_area = {
					rip_diaspora_established_community = yes
				}
				aqmola_area = {
					rip_diaspora_established_community = yes
				}
				kazakhstan_area = {
					rip_diaspora_established_community = yes
				}
			}
		}
		effect = {
			add_prestige = 25
			# A voyage of our own round the world, if we made one, adds renown;
			# it is no longer a gate for the southern branches.
			if = {
				limit = { has_country_flag = rip_lysianskyi_circumnavigation_flag }
				add_prestige = 10
			}
			add_country_modifier = {
				name = rip_diaspora_of_rus_legacy
				duration = -1
			}
			set_country_flag = rip_diaspora_of_rus_completed_flag
		}
	}
}

		}
	}

	rip_diaspora_usa = {
		icon = mission_rb_the_thirteen_colonies
		required_missions = { rip_diaspora_root }

		trigger = {
			OR = {
				has_country_flag = rip_colonize_canada_flag

	rip_diaspora_australia = {

			}
		}
		effect = {
			add_prestige = 10
			add_navy_tradition = 10
			set_country_flag = rip_diaspora_australia_flag
		}
	}

	rip_green_wedge = {
		icon = mission_afghan_frontier
		required_missions = { rip_diaspora_root }
		position = 30
		# The Green Wedge was a scatter of settler communities in a mixed land,
		# so owning one more province with a local population must not undo the
		# mission: three provinces, at least two of them with Ruthenian or Rusyn
		# communities.
		trigger = {
			is_subject = no
			num_of_owned_provinces_with = {
				region = manchuria_region
				value = 3
			}
			num_of_owned_provinces_with = {
				region = manchuria_region
				rip_diaspora_eligible_culture_province = yes
				value = 2
			}
		}
		effect = {
			add_prestige = 10
			set_country_flag = rip_green_wedge_flag
			country_event = { id = rip_diaspora.20 days = 5 }
		}
	}

	rip_grey_wedge = {
		icon = mission_rajputana_frontier
