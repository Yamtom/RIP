# ========================================================================
# The Kyivan succession question, 1444-1569, in the order it was argued.
# Two claims on the old table at Kyiv compete for the whole chain: the
# rotating seniorate (the ladder of thrones the Rurikids once climbed) and
# the cesarstvo (one seat, held outright). Every option nudges one of two
# hidden leanings; whichever reaches the threshold first opens the matching
# decision in decisions/KyivTriggers.txt (kyiv_shogunate_path for the
# seniorate, kyiv_cesarstvo_path for the cesarstvo) and closes off the other
# for good. See common/scripted_effects/rip_kyiv_succession_effects.txt for
# the threshold check and common/scripted_triggers/rip_kyiv_succession_triggers.txt
# for who is eligible.
#
# Authenticity: B - the frame is documented (the Olelkovichi princedom, its
# 1471 abolition into a voivodeship, the 1481 conspiracy, the Lithuanian
# Statute, Moscow's 1547 coronation, the approach of Lublin), the choices
# inside it are the mod's own.
# Sources:
#   Casimir IV's restoration of the Kyiv princedom for Olelko Volodymyrovych
#   after 1440; Semen Olelkovych's tenure from 1454; his death in 1470 and
#   Casimir's refusal to seat another Olelkovich, replaced by a crown
#   voivode from 1471; the 1481 conspiracy of Mykhailo Olelkovych, Ivan
#   Holshansky and Fedir Belsky; the first Lithuanian Statute, finished in
#   1529; Ivan IV's coronation as tsar in Moscow in 1547; the Union of
#   Lublin negotiations that closed the window in 1569 (history/provinces/280
#   already hands Kyiv to PLC on that date).
# ========================================================================

namespace = kyiv_succession_history

# --- 1. The princedom restored. Casimir has given Kyiv back a prince of its
#        own line, and nobody has written down whether the seat rotates or
#        descends.
country_event = {
	id = kyiv_succession_history.1
	title = kyiv_succession_history.1.t
	desc = kyiv_succession_history.1.d
	picture = easterngfx_COURT_eventPicture

	fire_only_once = yes

	trigger = {
		kyiv_succession_eligible_country = yes
		NOT = { has_country_flag = rus_flag }
		NOT = { has_country_flag = kyiv_path_chosen }
		NOT = { has_country_flag = kyiv_seniorate_road_open }
		NOT = { has_country_flag = kyiv_cesarstvo_road_open }
		owns_or_non_sovereign_subject_of = 280 # Kyiv
		NOT = { is_year = 1455 }
	}

	mean_time_to_happen = {
		months = 36
	}

	immediate = {
		set_variable = { which = rip_kyiv_seniorate_lean value = 0 }
		set_variable = { which = rip_kyiv_cesarstvo_lean value = 0 }
	}

	option = {
		name = kyiv_succession_history.1.a
		ai_chance = { factor = 40 }
		change_variable = { which = rip_kyiv_seniorate_lean value = 25 }
		rip_kyiv_succession_check_threshold_effect = yes
	}

	option = {
		name = kyiv_succession_history.1.b
		ai_chance = { factor = 40 }
		change_variable = { which = rip_kyiv_cesarstvo_lean value = 25 }
		rip_kyiv_succession_check_threshold_effect = yes
	}

	option = {
		name = kyiv_succession_history.1.c
		ai_chance = { factor = 20 }
		change_variable = { which = rip_kyiv_seniorate_lean value = 10 }
		change_variable = { which = rip_kyiv_cesarstvo_lean value = 5 }
		rip_kyiv_succession_check_threshold_effect = yes
	}
}
