uzh_palanok_captaincy_reform = {
	icon = "trading_city"
	potential = {
		OR = {
			has_reform = uzh_palanok_captaincy_reform
			tag = UZH
		}
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	lock_level_when_selected = yes
	queen = yes
	heir = no
	duration = 20
	has_term_election = yes
	rulers_can_be_generals = yes
	royal_marriage = yes
	allow_convert = no
	trigger = {
		OR = {
			has_reform = uzh_palanok_captaincy_reform
			tag = UZH
		}
	}
	modifiers = {
		defensiveness = 0.1
		fort_maintenance_modifier = -0.1
		# The captaincy's structural drag, constant whichever party is in
		# power; the dynamic trade/military upside sits on the faction
		# modifiers below instead. Kept literal (check_government_reforms.py)
		# so the burden never becomes a no-op under one party.
		land_morale = -0.05
		global_unrest = 1
	}
	custom_attributes = {
		locked_government_type = yes
		enables_plutocratic_idea_group = yes
		enables_aristocratic_idea_group = no
	}
	# The Sejm party and the Captains' party - a real vanilla FACTION contest
	# (base-game, like the KRU rada), not the Res Publica states_general scale
	# it used to fake. common/factions/RIP_uzh_palanok_factions.txt.
	factions = {
		uzh_partia_soymu
		uzh_partia_kapitaniv
	}
}
