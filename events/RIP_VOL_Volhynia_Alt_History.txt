country_event = {
	id = rip_vol_alt_history.1
	title = rip_vol_alt_history.1.t
	desc = rip_vol_alt_history.1.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	option = {
		name = rip_vol_alt_history.1.a
		trigger = { exists = HLC alliance_with = HLC HLC = { is_subject = no } }
		# Propose a binding bilateral Carpathian compact to Halych
		HLC = { country_event = { id = rip_vol_alt_history.9 } }
	}
	option = {
		name = rip_vol_alt_history.1.b
		# Reclaim full fiscal and executive sovereignty instead of shared upkeep
		set_country_flag = rip_vol_sovereign_crown
		remove_country_modifier = rip_vol_carpathian_compact
		remove_country_modifier = rip_vol_federal_compact
		add_country_modifier = { name = rip_vol_sovereign_crown_modifier duration = -1 }
		add_absolutism = 10
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = { estate = estate_nobles loyalty = -5 }
		}
	}
}

country_event = {
	id = rip_vol_alt_history.3
	title = rip_vol_alt_history.3.t
	desc = rip_vol_alt_history.3.d
	picture = RELIGION_eventPicture
	is_triggered_only = yes
	# Alternative history: the synod is a game model, not a recorded council.
	# Every option leaves rip_vol_synod_held behind and enacts a permanent
	# confessional rule with distinct estate and tolerance trade-offs.
	option = {
		name = rip_vol_alt_history.3.a
		# Orthodox Foundation
		set_country_flag = rip_vol_synod_held
		set_country_flag = rip_vol_synod_orthodox
		rip_faith_mission_resource_effect = {
			PA = 0.1
			GC = 0.03
			PI = 15
			ADM = 25
		}
		add_country_modifier = { name = rip_vol_synod_orthodox_settlement duration = -1 }
		if = {
			limit = { has_estate = estate_church }
			add_estate_loyalty = { estate = estate_church loyalty = 15 }
		}
		if = {
			limit = { has_estate = estate_burghers }
			add_estate_loyalty = { estate = estate_burghers loyalty = -5 }
		}
	}
	option = {
		name = rip_vol_alt_history.3.b
		# Uniate Concord
		set_country_flag = rip_vol_synod_held
		set_country_flag = rip_vol_synod_uniate
		rip_faith_mission_resource_effect = {
			PA = 0.1
			GC = 0.03
			PI = 25
			ADM = 25
		}
		add_country_modifier = { name = rip_vol_synod_uniate_settlement duration = -1 }
		if = {
			limit = { has_estate = estate_church }
			add_estate_loyalty = { estate = estate_church loyalty = 5 }
		}
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = { estate = estate_nobles loyalty = 5 }
		}
	}
	option = {
		name = rip_vol_alt_history.3.c
		# Charter of Tolerance
		set_country_flag = rip_vol_synod_held
		set_country_flag = rip_vol_synod_pluralist
		add_country_modifier = { name = rip_vol_synod_pluralist_settlement duration = -1 }
		if = {
			limit = { has_estate = estate_burghers }
			add_estate_loyalty = { estate = estate_burghers loyalty = 15 }
		}
		if = {
			limit = { has_estate = estate_church }
			add_estate_loyalty = { estate = estate_church loyalty = -10 }
		}
	}
}
