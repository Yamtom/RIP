# The Synod button. Either enactment counts as a synod having sat: it sets
# rip_uc_synod_held, the flag decisions/GreekCatholicDecisions.txt reads for
# preserve_eastern_rite_decision. Since the Curia was removed on 27 September
# only the crown path and these two effects set it. The flag is set inside the
# guarded if, so a refused button sets nothing.
rip_church_gc_infrastructure_effect = {
 if = { limit = { rip_church_gc_can_infrastructure = yes }
  add_patriarch_authority = -0.2 add_treasury = -75
  add_country_modifier = { name = rip_church_gc_infrastructure duration = 3650 }
  set_country_flag = rip_uc_synod_held
  rip_church_refresh_pa_effect = yes
 }
}
rip_church_gc_coexistence_effect = {
 if = { limit = { rip_church_gc_can_coexistence = yes }
  add_patriarch_authority = -0.2 add_treasury = -100
  add_country_modifier = { name = rip_church_gc_coexistence duration = 3650 }
  set_country_flag = rip_uc_synod_held
 }
}
rip_church_recognize_rite_effect = {
 if = { limit = { rip_church_can_recognize_rite = yes }
  export_to_variable = { which = rip_church_rite_price value = development }
  while = { limit = { check_variable = { which = rip_church_rite_price value = 1 } }
   owner = { add_adm_power = -1 } subtract_variable = { which = rip_church_rite_price value = 1 }
  }
  set_province_flag = rip_church_rite_recognized
  add_province_modifier = { name = rip_church_rite duration = -1 }
  owner = { rip_church_gc_refresh_parishes_effect = yes rip_church_gui_refresh_parishes_effect = yes }
 }
}
rip_church_clear_rite_effect = {
 clr_province_flag = rip_church_rite_recognized
 remove_province_modifier = rip_church_rite
 remove_province_modifier = rip_church_rite_favoured
 remove_province_modifier = rip_church_rite_ecumenical
}
rip_church_revoke_rite_effect = {
 if = { limit = { rip_church_can_revoke_rite = yes }
  rip_church_clear_rite_effect = yes
  add_province_modifier = { name = rip_church_rite_revoked duration = 3650 }
  owner = {
   set_country_flag = rip_church_forced_integration
   rip_church_gc_refresh_parishes_effect = yes
   rip_church_gui_refresh_parishes_effect = yes
  }
 }
}
rip_church_gc_refresh_parishes_effect = {
 set_variable = { which = rip_church_recognized_parishes value = 0 }
 every_owned_province = {
  remove_province_modifier = rip_church_rite_favoured
  remove_province_modifier = rip_church_rite_ecumenical
  remove_province_modifier = rip_church_gc_roman_resistance
  if = { limit = { has_province_flag = rip_church_rite_recognized }
   if = { limit = { NOT = { owner = { religion = greek_catholic } } } rip_church_clear_rite_effect = yes }
   else_if = { limit = { NOT = { OR = { religion = orthodox religion = russian_orthodox religion = catholic } } } rip_church_clear_rite_effect = yes }
   else = {
    owner = { change_variable = { which = rip_church_recognized_parishes value = 1 } }
    if = { limit = { owner = { has_country_flag = rip_church_ecumenical } } add_province_modifier = { name = rip_church_rite_ecumenical duration = -1 } }
    else = { add_province_modifier = { name = rip_church_rite_favoured duration = -1 } }
   }
  }
 }
}

rip_church_complete_florence_effect = {
 if = { limit = {
  has_country_flag = rip_church_florence_pending had_country_flag = { flag = rip_church_florence_pending days = 1825 }
  OR = { religion = catholic religion = orthodox } exists = PAP NOT = { war_with = PAP }
  is_at_war = no stability = 2 PAP = { has_opinion = { who = ROOT value = 100 } }
  any_owned_province = { rip_church_florence_parish = yes }
 }
  clr_country_flag = rip_church_florence_pending
  set_country_flag = rip_church_florentine_union
  enable_religion = greek_catholic
  if = { limit = { religion = catholic }
   set_country_flag = rip_church_supports_union
   clr_country_flag = rip_church_opposes_union
   random_owned_province = { limit = { rip_church_florence_parish = yes } change_religion = greek_catholic }
  }
  else = { rip_faith_adopt_union_effect = yes }
  set_country_flag = rip_church_union_founded
  country_event = { id = rip_church.5 }
 }
}
