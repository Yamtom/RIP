				}

			}
		}

		global_prosperity_growth = 0.25
		autonomy_change_time = -0.25
		min_autonomy_in_territories = -0.05
	}
	custom_attributes = {
		regimental_provincial_system = yes
		improved_military_buildings = yes
	}
	effect = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					is_city = yes

					}
				}
				add_province_modifier = {
					name = the_provincial_system

			}
		}
	}
	removed_effect = {
		hidden_effect = {

				limit = {
