# The county names of the Hungarian crown for the Carpathians.
#
# 25 names: 22 attested, 2 built from a real institution, 1 speculative.
# Speculative entries are marked inline. ASCII only - vanilla stores these
# files as Windows-1252 and a diacritic would render as mojibake.

### Poland - malopolska_area
262  = "Krakko"                        # standing Hungarian form for the Polish coronation city, shared crown of the Anjou and Jagiellon dynasties

### Poland - red_ruthenia_area
261  = "Halics"                        # the Halics of rex Galiciae, claimed in the royal style since Andrew II and briefly held for the crown in the 1370s
2961 = "Ilyvo"                         # the received Hungarian name of the Red Ruthenian staple city, unchanged since the Angevin governorship

### Poland - sandomierz_area
259  = "Szandomir"                     # Hungarian rendering of the Vistula palatinate the Anjou kings passed through between Buda and Krakow

### Poland - silesia_area
264  = "Boroszlo"                      # the Silesian staple town under Matthias Corvinus as king of Bohemia after 1469, so named in Hungarian usage
4723 = "Opuly"                         # duchy of Ladislaus Opuliai, palatine of Hungary and the crown's governor of Ruthenia 1372-1378

### Carpathia - moldavia_area
268  = "Kunorszag"                     # SPECULATIVE: the crown's dormant rex Cumaniae title revived as a chancery designation for the lands past the eastern passes
1756 = "Nesztorfehervar"               # Hungarian chancery form for the white fortress on the Dniester liman, the Genoese and Moldavian salt and grain port
4529 = "Jaszvasar"                     # the Hungarian name of the Moldavian princely town, market of the Jasz settled on the Bahlui

### Carpathia - slovakia_area
154  = "Nyitra"                        # county and bishopric of the upper Vag-Nyitra basin, older than the Ersekujvar fortress raised in it in 1573
162  = "Ungvar"                        # seat of Ung comitatus at the Uzh gate of the Carpathian passes, held by the Druget family under the crown
1318 = "Szepes"                        # the comitatus of the Saxon Twenty-Four Towns, thirteen of which stood pledged to the Polish king from 1412
1772 = "Pozsony"                       # county seat and diet town on the Danube gate, where the estates were summoned and kings crowned after 1526
4236 = "Trencsen"                      # comitatus and castle of the upper Vag, the crown's hold on the Moravian approaches

### Carpathia - southern_transylvania_area
156  = "Temes"                         # the comitatus and captaincy of the lower Danube frontier, held by Hunyadi as ispan from 1441
1951 = "Vajdahunyad"                   # the Hunyadi castle above the Cserna, the family seat behind the southern Transylvanian passes
4128 = "Nagyszeben"                    # seat of the Saxon Universitas on the Kiralyfold, the king's land held by charter since 1224

### Carpathia - transdanubia_area
4240 = "Szekesfehervar"                # the coronation and burial church of the kings and seat of Fejer comitatus

### Carpathia - transylvania_area
157  = "Bihar"                         # comitatus of the Partium and see of Nagyvarad, the crown's eastern chapter and place of authentication
158  = "Erdely"                        # the voivodate beyond the forest, governed from Gyulafehervar; the crown's own word for it
1952 = "Huszt"                         # crown fortress commanding the Tisza gorge and the salt road out of Maramaros
4127 = "Torda"                         # the salt town where the Transylvanian diet habitually met, including the 1568 assembly on the received confessions

### Carpathia - wallachia_area
160  = "Kis-Olahorszag"                # the Hungarian chancery term for the lesser voivodate west of the Olt, current in the Habsburg administration of 1718-1739
161  = "Havasalfold"                   # the land below the snowy mountains, the standing Hungarian name for the voivodate held in vassalage from the crown
4532 = "Szoreny"                       # the banate erected on the Danube bend in 1233, one of the crown's two standing marcher offices toward Wallachia
