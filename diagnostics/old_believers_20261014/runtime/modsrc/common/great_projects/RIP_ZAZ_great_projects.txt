# =============================================================================
# RIP_ZAZ_great_projects
# ZAZ / HET / RUS - the Zaporozhian Host. One monument: zaporozhian_sich.
# Ordered as vanilla orders a monument: start, date, time, build_cost,
# can_be_moved, move_days_per_unit_distance, starting_tier, type, build_trigger,
# on_built, on_destroyed, can_use_modifiers_trigger, can_upgrade_trigger,
# keep_trigger, tier_0 .. tier_3.
# =============================================================================

# The Zaporozhian Sich, the fortified seat of the Host below the Dnieper
# rapids. Founded in the middle of the sixteenth century, it was fortress,
# rada and school at once, and the Host moved it whenever the river, the
# Tatars or an overlord made the old site untenable - which is why this is one
# of the few monuments the mod leaves movable.
#
# Distinct from the objects that share the name: zaporozhian_sich_capital is a
# province modifier, zaporozhian_sich_founded a country modifier, and
# zaporozhian_sich_crisis the ZAZ disaster in common/disasters/zaz_disasters.txt.
#
# The 1552 founding is gated on the Host existing - a Cossack polity or the
# Cossack estate - rather than on is_year, so an early or late Sich follows the
# institution instead of the calendar. date stays 1552.01.01: it is the
# game-start placement gate only. No vanilla great project uses is_year at all.
#
# build_trigger carries area = zaporizhia_area: `start` is only the game-start
# seat and does not constrain where a project may be BUILT. Without that gate a
# 1.37.5 save has the Sich standing in province 4539 Kremenets, in Volhynia,
# under a German owner. The area (283 2411 2412) still leaves the Host room to
# move the Sich along the rapids, which is the point of can_be_moved = yes.
#
# Ladder and payloads stay inside the envelope measured across all 138 vanilla
# monuments: costs 1000 / 2500 / 5000, times 120 / 240 / 480 months, at most
# six country modifiers a tier, no combat pips, prestige never above 1.

zaporozhian_sich = {
	start = 283	# Zaporozhia
	date = 1552.01.01
	time = {
		months = 60
	}
	build_cost = 3000
	can_be_moved = yes
	move_days_per_unit_distance = 5
	starting_tier = 0
	type = monument

	build_trigger = {
		OR = {
			owner = {
				tag = ZAZ
			}
			owner = {
				tag = HET
			}
			owner = {
				tag = RUS
			}
			culture = ruthenian
		}
		area = zaporizhia_area
		owner = {
			custom_trigger_tooltip = {
				tooltip = zaporozhian_sich_host_tt
				OR = {
					is_cossack_polity = yes
					has_estate = estate_cossacks
				}
			}
		}
		has_owner_accepted_culture = yes
		OR = {
			province_has_center_of_trade_of_level = 1
			has_terrain = steppe
			has_terrain = grasslands
		}
	}

	on_built = {
		owner = {
			country_event = { id = zaz_monument.1 days = 3 }
		}
	}

	on_destroyed = {
	}

	can_use_modifiers_trigger = {
		custom_trigger_tooltip = {
			tooltip = zaporozhian_sich_culture_tt
			OR = {
				culture = ruthenian
				owner = {
					accepted_culture = ruthenian
				}
			}
		}
	}

	can_upgrade_trigger = {
		custom_trigger_tooltip = {
			tooltip = zaporozhian_sich_upgrade_tt
			OR = {
				culture = ruthenian
				owner = {
					accepted_culture = ruthenian
				}
			}
		}
	}

	keep_trigger = {
	}

	tier_0 = {
		upgrade_time = {
			months = 0
		}
		cost_to_upgrade = {
			factor = 0
		}
		province_modifiers = {
		}
		area_modifier = {
		}
		country_modifiers = {
		}
		on_upgraded = {
		}
	}

	tier_1 = {
		upgrade_time = {
			months = 120
		}
		cost_to_upgrade = {
			factor = 1000
		}
		province_modifiers = {
			local_defensiveness = 0.2
			local_hostile_movement_speed = -0.15
		}
		area_modifier = {
			local_manpower_modifier = 0.15
		}
		country_modifiers = {
			cavalry_power = 0.1
			cavalry_cost = -0.1
			cossacks_loyalty_modifier = 0.1
		}
		on_upgraded = {
		}
	}

	tier_2 = {
		upgrade_time = {
			months = 240
		}
		cost_to_upgrade = {
			factor = 2500
		}
		province_modifiers = {
			local_defensiveness = 0.3
			local_hostile_movement_speed = -0.25
		}
		area_modifier = {
			local_manpower_modifier = 0.2
		}
		country_modifiers = {
			cavalry_power = 0.15
			cavalry_cost = -0.15
			hostile_attrition = 1
			cossacks_loyalty_modifier = 0.15
		}
		on_upgraded = {
			owner = {
				country_event = { id = zaz_monument.2 days = 3 }
			}
		}
	}

	tier_3 = {
		upgrade_time = {
			months = 480
		}
		cost_to_upgrade = {
			factor = 5000
		}
		province_modifiers = {
			local_defensiveness = 0.4
			local_hostile_movement_speed = -0.35
		}
		area_modifier = {
			local_manpower_modifier = 0.25
		}
		country_modifiers = {
			cavalry_power = 0.2
			cavalry_cost = -0.2
			hostile_attrition = 2
			cossacks_loyalty_modifier = 0.2
		}
		on_upgraded = {
			custom_tooltip = zaporozhian_sich_tier_3_tt
			owner = {
				country_event = { id = zaz_monument.3 days = 3 }
			}
		}
	}
}
