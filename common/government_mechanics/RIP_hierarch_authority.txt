# ==========================================================================
# THE AUTHORITY OF THE HIERARCHS
#
# WHY THIS AND NOT A .gui EDIT
#
# The first two attempts at giving this faith a panel both failed, and they
# failed the same way: EU4's religion windows are engine-owned. There are
# exactly eleven `*_specific_window` names in the executable, every widget in
# them is bound by name inside its own window, and no vanilla .gui carries a
# single `trigger`, `visible` or `potential` block. Moving those widgets
# somewhere better is a gamble, and moving them into the religion view's lower
# band was worse than a gamble - `province_listbox` sits at y=402 and the
# panels landed on top of it.
#
# common/government_mechanics is the part of EU4 that IS meant to be extended
# this way. It gives a scripted power with a real progress bar, scripted
# buttons with their own icons that spend it, cooldowns, mutual exclusivity
# and AI weighting - and it draws through vanilla's own default GUI
# (government_shared_power / government_interaction_type), so there is no new
# window to bind and nothing to guess at. It is a unique object of ours built
# out of vanilla parts, which is exactly the brief.
#
# WHAT THE BAR IS
#
# Something between piety and patriarch authority, as asked:
#   - like patriarch authority, it fills by itself, scales with the size of
#     the church (development_scaled_monthly_growth) and pays out in
#     proportion to how full it is (scaled_modifier);
#   - like piety, the BOTTOM of the bar is a state with its own character
#     rather than merely an absence of bonuses (reverse_scaled_modifier), so
#     a church that has spent its authority is a church with a problem.
#
# THE FOUR ICONS
#
# Four interactions, one cooldown token between them, so choosing one puts
# the other three out of reach for ten years: that is the "four slots, pick
# one, changing it costs something" the brief asks for. The cost is the
# authority itself. Each grants the same country modifier the vanilla icon
# mechanic granted - the payloads in RIP_faith_modifiers.txt are untouched -
# for vanilla's own icon duration of twenty years.
#
# The fervour bar stays what it is: the Sobor, in common/fervor.
# ==========================================================================

rip_ro_hierarch_authority_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	powers = {
		rip_ro_hierarch_authority = {
			max = 100
			default = 40
			reset_on_new_ruler = no
			is_good = yes

			# The church gathers standing on its own, faster when there is
			# more of it to gather from, and faster under an able ruler who
			# can work with his metropolitan rather than against him.
			base_monthly_growth = 0.25
			development_scaled_monthly_growth = 0.15
			monarch_power = ADM

			# The patriarch-authority half: pays out in proportion.
			scaled_modifier = {
				modifier = {
					global_tax_modifier = 0.10
					global_missionary_strength = 0.01
					global_unrest = -1
				}
			}

			# The piety half: the empty end of the bar is its own condition,
			# not just the absence of the bonus above.
			reverse_scaled_modifier = {
				modifier = {
					stability_cost_modifier = 0.15
					church_loyalty_modifier = -0.1
				}
			}

			# And the two poles, applied whole rather than scaled, so the ends
			# of the bar read as states of the church rather than as numbers.
			range_modifier = {
				start_value = 90
				end_value = 100
				modifier = {
					prestige = 0.5
					advisor_cost = -0.1
				}
			}
			range_modifier = {
				start_value = 0
				end_value = 10
				modifier = {
					global_autonomy = 0.05
				}
			}
		}
	}

	interactions = {
		# --- 1. The Archangel: the icon carried in front of the regiments ---
		rip_ro_take_icon_michael = {
			icon = GFX_sanction_holy_war_button
			cost_type = rip_ro_hierarch_authority
			cost = 25
			cooldown_years = 10
			cooldown_token = rip_ro_icon_choice

			trigger = {
				religion = russian_orthodox
			}

			effect = {
				hidden_effect = {
					remove_country_modifier = rip_ro_icon_eleusa
					remove_country_modifier = rip_ro_icon_pancreator
					remove_country_modifier = rip_ro_icon_climacus
				}
				add_country_modifier = {
					name = rip_ro_icon_michael
					duration = 7300
				}
			}

			ai_chance = {
				factor = 1
				modifier = {
					factor = 4
					is_at_war = yes
				}
				modifier = {
					factor = 0.2
					is_at_war = no
				}
			}
		}

		# --- 2. The Eleusa: the icon carried through a frightened town ------
		rip_ro_take_icon_eleusa = {
			icon = GFX_hold_religious_festival_interaction
			cost_type = rip_ro_hierarch_authority
			cost = 25
			cooldown_years = 10
			cooldown_token = rip_ro_icon_choice

			trigger = {
				religion = russian_orthodox
			}

			effect = {
				hidden_effect = {
					remove_country_modifier = rip_ro_icon_michael
					remove_country_modifier = rip_ro_icon_pancreator
					remove_country_modifier = rip_ro_icon_climacus
				}
				add_country_modifier = {
					name = rip_ro_icon_eleusa
					duration = 7300
				}
			}

			ai_chance = {
				factor = 1
				modifier = {
					factor = 4
					NOT = { stability = 0 }
				}
				modifier = {
					factor = 3
					num_of_rebel_armies = 1
				}
			}
		}

		# --- 3. The Pantocrator: the icon over the builders ----------------
		rip_ro_take_icon_pancreator = {
			icon = GFX_praise_patron_deity_interaction
			cost_type = rip_ro_hierarch_authority
			cost = 25
			cooldown_years = 10
			cooldown_token = rip_ro_icon_choice

			trigger = {
				religion = russian_orthodox
			}

			effect = {
				hidden_effect = {
					remove_country_modifier = rip_ro_icon_michael
					remove_country_modifier = rip_ro_icon_eleusa
					remove_country_modifier = rip_ro_icon_climacus
				}
				add_country_modifier = {
					name = rip_ro_icon_pancreator
					duration = 7300
				}
			}

			ai_chance = {
				factor = 1
				modifier = {
					factor = 2
					is_at_war = no
				}
			}
		}

		# --- 4. The Ladder: the icon over the schools ----------------------
		rip_ro_take_icon_climacus = {
			icon = GFX_reform_religion_interaction
			cost_type = rip_ro_hierarch_authority
			cost = 25
			cooldown_years = 10
			cooldown_token = rip_ro_icon_choice

			trigger = {
				religion = russian_orthodox
			}

			effect = {
				hidden_effect = {
					remove_country_modifier = rip_ro_icon_michael
					remove_country_modifier = rip_ro_icon_eleusa
					remove_country_modifier = rip_ro_icon_pancreator
				}
				add_country_modifier = {
					name = rip_ro_icon_climacus
					duration = 7300
				}
			}

			ai_chance = {
				factor = 1
				modifier = {
					factor = 4
					NOT = { has_institution = printing_press }
				}
			}
		}
	}
}
