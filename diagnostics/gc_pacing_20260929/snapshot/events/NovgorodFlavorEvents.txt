# Fall of Muscovite Autocracy to the Veche
country_event = {
	id = nov_flavor.1
	title = nov_flavor.1.t
	desc = nov_flavor.1.d
	picture = DEBATE_REPUBLICAN_eventPicture

	trigger = {
		OR = {
			tag = NOV
			primary_culture = novgorodian
		}
		owns_core_province = 295 # Moskva
		is_at_war = no
		NOT = { has_country_flag = nov_fall_of_moscow_handled }
	}

	mean_time_to_happen = {
		months = 6
	}

	immediate = {
		set_country_flag = nov_fall_of_moscow_handled
	}

	option = {
		name = nov_flavor.1.a # Introduce veche governance across Zalesye
		ai_chance = { factor = 70 }
		add_republican_tradition = 20
		add_adm_power = 50
		295 = {
			add_province_modifier = {
				name = "nov_veche_rule_in_zalesye"
				duration = -1
			}
		}
		307 = {
			if = {
				limit = { owned_by = ROOT }
				add_province_modifier = {
					name = "nov_veche_rule_in_zalesye"
					duration = -1
				}
			}
		}
		1956 = {
			if = {
				limit = { owned_by = ROOT }
				add_province_modifier = {
					name = "nov_veche_rule_in_zalesye"
					duration = -1
				}
			}
		}
	}

	option = {
		name = nov_flavor.1.b # Establish commercial trading factories
		ai_chance = { factor = 20 }
		add_treasury = 150
		add_dip_power = 50
		295 = {
			add_base_tax = 1
			add_base_production = 2
		}
	}

	option = {
		name = nov_flavor.1.c # Direct centralized incorporation
		ai_chance = { factor = 10 }
		add_prestige = 25
		add_mil_power = 50
		add_stability_or_adm_power = yes
	}
}
