# ========================================================================
# KRU - the Kyivan Confederation.
#
# The rada is the vanilla FACTION mechanic. common/factions/RIP_kru_factions.txt
# defines the three blocs (kru_knyazi / kru_boyary / kru_starshyna); the three
# ruthenian_factional_empire_*_reform ids turn it on with
# `factions = { ... }`. Vanilla accrues the influence, names the bloc in power
# and applies its modifier, and draws the panel. This file only:
#   - drifts influence slightly toward the bloc the country's state favours,
#   - records the current and previous dominant factions and selects their
#     coalition reform,
#   - centers the native States General bar when no faction reaches 60,
#   - gives the previous dominant faction a junior voice if it retains 38.
# The former custom variables and pairwise balance powers are retired;
# kru_clear_faction_modifiers_effect migrates old KRU modifiers off saves.
# ========================================================================

# A small yearly nudge, on top of vanilla drift, events and the court buttons.
kru_faction_state_drift_effect = {
	if = { limit = { legitimacy_equivalent = 60 } add_faction_influence = { faction = kru_knyazi influence = 1 } }
	if = { limit = { num_of_subjects = 2 } add_faction_influence = { faction = kru_knyazi influence = 1 } }
	if = { limit = { stability = 2 } add_faction_influence = { faction = kru_boyary influence = 1 } }
	if = { limit = { treasury = 300 } add_faction_influence = { faction = kru_boyary influence = 1 } }
	if = { limit = { army_tradition = 40 } add_faction_influence = { faction = kru_starshyna influence = 1 } }
	if = { limit = { is_at_war = yes } add_faction_influence = { faction = kru_starshyna influence = 1 } }
}

kru_clear_faction_modifiers_effect = {
	remove_country_modifier = KRU_junior_knyazi
	remove_country_modifier = KRU_junior_boyary
	remove_country_modifier = KRU_junior_starshyna
	remove_country_modifier = KRU_factional_pair_shift_cooldown
	# migration - the retired pair modifiers
	remove_country_modifier = KRU_pair_knyazi_boyary
	remove_country_modifier = KRU_pair_knyazi_hetmany
	remove_country_modifier = KRU_pair_boyary_hetmany
	remove_country_modifier = KRU_courting_knyazi
	remove_country_modifier = KRU_courting_boyary
	remove_country_modifier = KRU_courting_starshyna
}

# The reform id names the coalition: the currently dominant faction and the
# faction that dominated immediately before it. The remaining faction is the
# opposition. At fewer than 60 influence there is no dominant bloc, so the
# reform returns to its neutral base and the native States General scale is
# centered.
kru_store_current_as_previous_dominant_effect = {
	clr_country_flag = kru_previous_dominant_knyazi
	clr_country_flag = kru_previous_dominant_boyary
	clr_country_flag = kru_previous_dominant_starshyna

	if = {
		limit = { has_country_flag = kru_current_dominant_knyazi }
		set_country_flag = kru_previous_dominant_knyazi
	}
	if = {
		limit = { has_country_flag = kru_current_dominant_boyary }
		set_country_flag = kru_previous_dominant_boyary
	}
	if = {
		limit = { has_country_flag = kru_current_dominant_starshyna }
		set_country_flag = kru_previous_dominant_starshyna
	}
}

kru_clear_current_dominant_effect = {
	clr_country_flag = kru_current_dominant_knyazi
	clr_country_flag = kru_current_dominant_boyary
	clr_country_flag = kru_current_dominant_starshyna
}

kru_set_fallback_previous_dominant_effect = {
	if = {
		limit = {
			faction_in_power = kru_knyazi
			NOT = {
				OR = {
					has_country_flag = kru_previous_dominant_boyary
					has_country_flag = kru_previous_dominant_starshyna
				}
			}
		}
		clr_country_flag = kru_previous_dominant_knyazi
		clr_country_flag = kru_previous_dominant_boyary
		clr_country_flag = kru_previous_dominant_starshyna
		set_country_flag = kru_previous_dominant_boyary
	}
	else_if = {
		limit = {
			faction_in_power = kru_boyary
			NOT = {
				OR = {
					has_country_flag = kru_previous_dominant_knyazi
					has_country_flag = kru_previous_dominant_starshyna
				}
			}
		}
		clr_country_flag = kru_previous_dominant_knyazi
		clr_country_flag = kru_previous_dominant_boyary
		clr_country_flag = kru_previous_dominant_starshyna
		set_country_flag = kru_previous_dominant_knyazi
	}
	else_if = {
		limit = {
			faction_in_power = kru_starshyna
			NOT = {
				OR = {
					has_country_flag = kru_previous_dominant_knyazi
					has_country_flag = kru_previous_dominant_boyary
				}
			}
		}
		clr_country_flag = kru_previous_dominant_knyazi
		clr_country_flag = kru_previous_dominant_boyary
		clr_country_flag = kru_previous_dominant_starshyna
		set_country_flag = kru_previous_dominant_knyazi
	}
}

kru_center_states_general_effect = {
	if = {
		limit = { has_states_general_mechanic = yes }
		change_statists_vs_orangists = -10
		change_statists_vs_orangists = 1
	}
}

kru_record_dominant_faction_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					faction_in_power = kru_knyazi
					faction_influence = { faction = kru_knyazi influence = 60 }
				}
				AND = {
					faction_in_power = kru_boyary
					faction_influence = { faction = kru_boyary influence = 60 }
				}
				AND = {
					faction_in_power = kru_starshyna
					faction_influence = { faction = kru_starshyna influence = 60 }
				}
			}
		}
		set_country_flag = kru_factional_dominance_active
		if = {
			limit = {
				OR = {
					AND = {
						faction_in_power = kru_knyazi
						NOT = { has_country_flag = kru_current_dominant_knyazi }
					}
					AND = {
						faction_in_power = kru_boyary
						NOT = { has_country_flag = kru_current_dominant_boyary }
					}
					AND = {
						faction_in_power = kru_starshyna
						NOT = { has_country_flag = kru_current_dominant_starshyna }
					}
				}
			}
			if = {
				limit = {
					OR = {
						has_country_flag = kru_current_dominant_knyazi
						has_country_flag = kru_current_dominant_boyary
						has_country_flag = kru_current_dominant_starshyna
					}
				}
				kru_store_current_as_previous_dominant_effect = yes
			}
			kru_set_fallback_previous_dominant_effect = yes
			kru_clear_current_dominant_effect = yes
			if = {
				limit = { faction_in_power = kru_knyazi }
				set_country_flag = kru_current_dominant_knyazi
			}
			else_if = {
				limit = { faction_in_power = kru_boyary }
				set_country_flag = kru_current_dominant_boyary
			}
			else_if = {
				limit = { faction_in_power = kru_starshyna }
				set_country_flag = kru_current_dominant_starshyna
			}
			kru_center_states_general_effect = yes
			if = {
				limit = { has_states_general_mechanic = yes }
				change_statists_vs_orangists = -0.15
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = kru_factional_dominance_active }
			kru_store_current_as_previous_dominant_effect = yes
			kru_clear_current_dominant_effect = yes
			clr_country_flag = kru_factional_dominance_active
		}
		kru_center_states_general_effect = yes
	}
}

kru_sync_reform_to_faction_effect = {
	if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			OR = {
				AND = {
					faction_in_power = kru_knyazi
					faction_influence = { faction = kru_knyazi influence = 60 }
					has_country_flag = kru_previous_dominant_boyary
				}
				AND = {
					faction_in_power = kru_boyary
					faction_influence = { faction = kru_boyary influence = 60 }
					has_country_flag = kru_previous_dominant_knyazi
				}
			}
		}
		if = {
			limit = { NOT = { has_reform = ruthenian_factional_empire_reform } }
			remove_government_reform = ruthenian_factional_empire_princes_hetmans_reform
			remove_government_reform = ruthenian_factional_empire_boyars_hetmans_reform
			add_government_reform = ruthenian_factional_empire_reform
		}
	}
	else_if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			OR = {
				AND = {
					faction_in_power = kru_knyazi
					faction_influence = { faction = kru_knyazi influence = 60 }
					has_country_flag = kru_previous_dominant_starshyna
				}
				AND = {
					faction_in_power = kru_starshyna
					faction_influence = { faction = kru_starshyna influence = 60 }
					has_country_flag = kru_previous_dominant_knyazi
				}
			}
		}
		if = {
			limit = { NOT = { has_reform = ruthenian_factional_empire_princes_hetmans_reform } }
			remove_government_reform = ruthenian_factional_empire_reform
			remove_government_reform = ruthenian_factional_empire_boyars_hetmans_reform
			add_government_reform = ruthenian_factional_empire_princes_hetmans_reform
		}
	}
	else_if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			OR = {
				AND = {
					faction_in_power = kru_boyary
					faction_influence = { faction = kru_boyary influence = 60 }
					has_country_flag = kru_previous_dominant_starshyna
				}
				AND = {
					faction_in_power = kru_starshyna
					faction_influence = { faction = kru_starshyna influence = 60 }
					has_country_flag = kru_previous_dominant_boyary
				}
			}
		}
		if = {
			limit = { NOT = { has_reform = ruthenian_factional_empire_boyars_hetmans_reform } }
			remove_government_reform = ruthenian_factional_empire_reform
			remove_government_reform = ruthenian_factional_empire_princes_hetmans_reform
			add_government_reform = ruthenian_factional_empire_boyars_hetmans_reform
		}
	}
	else = {
		if = {
			limit = { NOT = { has_reform = ruthenian_factional_empire_reform } }
			remove_government_reform = ruthenian_factional_empire_princes_hetmans_reform
			remove_government_reform = ruthenian_factional_empire_boyars_hetmans_reform
			add_government_reform = ruthenian_factional_empire_reform
		}
	}
}

# Only the faction that dominated before the current one sits in the coalition
# as a junior partner. Its influence and the current faction's influence are
# separate: this effect never drifts either value.
kru_apply_junior_blocs_effect = {
	kru_clear_faction_modifiers_effect = yes
	if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			has_country_flag = kru_previous_dominant_knyazi
			NOT = { faction_in_power = kru_knyazi }
			faction_influence = { faction = kru_knyazi influence = 38 }
		}
		add_country_modifier = { name = KRU_junior_knyazi duration = -1 }
	}
	else_if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			has_country_flag = kru_previous_dominant_boyary
			NOT = { faction_in_power = kru_boyary }
			faction_influence = { faction = kru_boyary influence = 38 }
		}
		add_country_modifier = { name = KRU_junior_boyary duration = -1 }
	}
	else_if = {
		limit = {
			has_country_flag = kru_factional_dominance_active
			has_country_flag = kru_previous_dominant_starshyna
			NOT = { faction_in_power = kru_starshyna }
			faction_influence = { faction = kru_starshyna influence = 38 }
		}
		add_country_modifier = { name = KRU_junior_starshyna duration = -1 }
	}
}

kru_refresh_factional_court_effect = {
	kru_record_dominant_faction_effect = yes
	kru_sync_reform_to_faction_effect = yes
	kru_apply_junior_blocs_effect = yes
}

kru_update_dominant_pair_effect = {
	kru_faction_state_drift_effect = yes
	kru_refresh_factional_court_effect = yes
}

kru_confederation_recognised_effect = {
	if = {
		limit = { NOT = { has_country_flag = kru_confederation_recognised } }
		set_country_flag = kru_confederation_recognised

		# The former overlords have to answer first - they are the ones losing
		# the palatinates.
		if = {
			limit = { exists = LIT }
			LIT = { country_event = { id = svydrigaylo.300 days = 5 } }
		}
		if = {
			limit = { exists = PLC }
			PLC = { country_event = { id = svydrigaylo.301 days = 5 } }
		}
		if = {
			limit = { exists = POL NOT = { exists = PLC } }
			POL = { country_event = { id = svydrigaylo.301 days = 5 } }
		}

		# Moscow holds the rival claim to the Rus succession.
		if = {
			limit = { exists = RUS }
			RUS = { country_event = { id = svydrigaylo.302 days = 10 } }
		}
		if = {
			limit = { exists = MOS NOT = { exists = RUS } }
			MOS = { country_event = { id = svydrigaylo.302 days = 10 } }
		}

		# The southern powers weigh a Ruthenian state against Poland.
		if = {
			limit = { exists = TUR }
			TUR = { country_event = { id = svydrigaylo.303 days = 15 } }
		}
		if = {
			limit = { exists = CRI }
			CRI = { country_event = { id = svydrigaylo.304 days = 15 } }
		}
		if = {
			limit = { exists = MOL }
			MOL = { country_event = { id = svydrigaylo.305 days = 20 } }
		}
		if = {
			limit = { exists = WAL }
			WAL = { country_event = { id = svydrigaylo.306 days = 20 } }
		}

		# Everyone else settles into one of two positions without an event:
		# the Orthodox world accepts Kyiv as the senior seat of the Rus, and
		# whoever already favours us backs the confederation politically.
		every_known_country = {
			limit = {
				NOT = { tag = ROOT }
				religion = orthodox
				NOT = { tag = MOS }
				NOT = { tag = RUS }
				NOT = { tag = ZAZ }
				NOT = { tag = HET }
				NOT = { tag = CHR }
				NOT = { tag = KIE }
			}
			add_opinion = {
				who = ROOT
				modifier = kyivan_confederation_successor
			}
		}
		every_known_country = {
			limit = {
				NOT = { tag = ROOT }
				NOT = { religion = orthodox }
				NOT = { tag = LIT }
				NOT = { tag = POL }
				NOT = { tag = PLC }
				NOT = { tag = MOS }
				NOT = { tag = RUS }
				NOT = { tag = TUR }
				NOT = { tag = CRI }
				NOT = { tag = MOL }
				NOT = { tag = WAL }
				has_opinion = {
					who = ROOT
					value = 50
				}
			}
			add_opinion = {
				who = ROOT
				modifier = supports_ruthenian_confederation
			}
		}
	}
}
