
# NOTE: this file is not currently called from anywhere (no bracket call, no

defined_text = {
	name = GetCossackHostName

	text = {
		trigger = {
			tag = ZAZ
			OR = {
				has_reform = zaz_sich_brotherhood_reform
				has_reform = zaz_kosh_elections_reform
				has_reform = zaz_free_host_reform
				has_reform = zaz_chaiky_trade_reform
			}
		}
		localisation_key = ZAPOROZHIAN_HOST_NAME
	}
	
	text = {
		trigger = {
			tag = HET
			has_reform = het_mazepist_autocracy_reform
		}
		localisation_key = HETMANATE_HOST_NAME
	}

	text = {
		trigger = {
			OR = {
				has_reform = cossacks_reform
				has_reform = cossacks_legacy_reform
				has_reform = rip_cossacks_reform
			}
			any_owned_province = {
				area = lower_don_area
			}
		}
		localisation_key = DON_HOST_NAME
	}
	
	
	
	
	text = {
		# Simplified to the shared trigger rather than re-enumerating reforms
		# a second time - is_cossack_polity is the single list to maintain.
		trigger = {
			is_cossack_polity = yes
		}
		localisation_key = COSSACK_HOST_GENERIC
	}
	
	text = {
		localisation_key = COUNTRY_NAME_DEFAULT
	}
}
