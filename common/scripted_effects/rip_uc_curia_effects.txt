# Once per old save; repeated startup does not erase a new paid benefit.
# The migration flag is never cleared: switching religion cannot reset timers.
rip_uc_curia_migration_effect = {
	if = {
		limit = { religion = greek_catholic }
		rip_uc_clear_the_bands_effect = yes
		# Old petitions used Catholic IDs. Remove only while Greek Catholic;
		# Catholics and their native papal actions are never touched.
		remove_country_modifier = papal_sanction_for_church_taxes
		remove_country_modifier = papal_blessing
		remove_country_modifier = papal_indulgence
		remove_country_modifier = usury_forgiven
		remove_country_modifier = papal_legate
		remove_country_modifier = papal_sanction_for_holy_war
		if = {
			limit = { NOT = { has_country_flag = rip_uc_curia_v2_migrated } }
			rip_uc_clear_church_benefits_effect = yes
			set_country_flag = rip_uc_curia_v2_migrated
		}
	}
	else_if = {
		limit = { has_country_flag = rip_uc_curia_v2_migrated }
		rip_uc_clear_church_benefits_effect = yes
		rip_uc_clear_the_bands_effect = yes
	}
}
