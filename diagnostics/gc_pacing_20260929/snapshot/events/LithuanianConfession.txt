# ========================================================================
# The confessional split inside the Grand Duchy, and how it diverges from
# the Crown's.
# Authenticity: A - documented. Follows recorded institutions and people.
# Sources:
#   the Horodlo privilege reserving council seats for Catholic lords, and
#   its long erosion; the metropolitan of Kyiv seated in the Grand Duchy
#   against Moscow's claim to all Rus; Skaryna's Slavonic press in Vilnius;
#   the Radziwill court at Vilnius and Lithuanian Calvinism, which went
#   further than the Crown's; the Jesuit academy at Vilnius; the Statute's
#   own confessional guarantees beside the Warsaw Confederation; the two
#   parallel hierarchies left in the Grand Duchy after Brest.
#
# This chain is the Grand Duchy's half of a story the mod already tells for
# the Commonwealth as a whole in events/CommonwealthReformation.txt and
# events/UniateChurch.txt. It reads their flags - pl_magnate_reformation,
# pl_warsaw_confederation, pl_catholic_recovery, uniate_union_proposed - and
# asks the question those chains do not: what happens when the court and the
# country are not of the same faith.
# ========================================================================

namespace = lit_confession

# --- 1. A Catholic court over a country that is not.
country_event = {
	id = lit_confession.1
	title = lit_confession.1.t
	desc = lit_confession.1.d
	picture = RELIGION_eventPicture

	fire_only_once = yes

	trigger = {
		lit_rules_an_orthodox_country = yes
		NOT = { is_year = 1520 }
	}

	mean_time_to_happen = {
		months = 60
	}

	option = {
		name = lit_confession.1.a
		ai_chance = { factor = 50 }
		add_country_modifier = {
			name = lit_ruthenian_lords_content
			duration = 9125
		}
		add_papal_influence = -20
		add_legitimacy = -5
	}

	option = {
		name = lit_confession.1.b
		ai_chance = { factor = 50 }
		add_papal_influence = 20
		add_legitimacy = 5
		every_owned_province = {
			limit = {
				religion = orthodox
				is_state_core = ROOT
			}
			add_unrest = 2
		}
	}
}
