	}

# A parliamentary seat represents its land directly. Other states qualify
# through a developed provincial centre with a recognised culture.
uzh_federal_land = {
	region = carpathia_region
	is_state = yes
	OR = {
		culture = ROOT
		has_owner_accepted_culture = yes
	}
	OR = {
		development = 10
		has_seat_in_parliament = yes
	}
}

uzh_loyal_subject_network = {
	NOT = {
		any_subject_country = {
			liberty_desire = 60
		}
	}
}

uzh_identity_core_culture = {
	OR = {
		AND = {
			owner = { has_country_flag = chose_ruthenian_identity }
			OR = {
				culture = ruthenian
				culture = ruthenian_new
				culture = ruthenian_west_ruthenian
				culture = ruthenian_west_slavic
			}
		}
		AND = {
			owner = { has_country_flag = chose_rusyn_identity }
			OR = {
				culture = rusyn
				culture = rusyn_new
				culture = rusyn_west_ruthenian
				culture = rusyn_west_slavic
			}
		}
		AND = {
			owner = { has_country_flag = chose_uhro_rusyn_identity }
			OR = {
				culture = rusyn_new_new
			}
		}
		AND = {
			owner = {
				NOT = { has_country_flag = chose_ruthenian_identity }
				NOT = { has_country_flag = chose_rusyn_identity }
				NOT = { has_country_flag = chose_uhro_rusyn_identity }
			}
			OR = {
				culture = rusyn
				culture = rusyn_new
				culture = rusyn_west_ruthenian
				culture = rusyn_west_slavic
			}
		}
	}
}
