# =============================================================================
# The Boyar Republic - a third fork off hlc_galician_voivodeship_reform,
# alongside the Poland and Austria paths (decisions/HLCRepublicPaths.txt).
# Real anchor: the Galician boyars repeatedly deposed and installed princes
# of their own choosing (Volodyslav Kormylchych briefly reigned as a boyar in
# the 1210s, and the council invited Andrew of Hungary and Coloman over the
# native line - see hlc_hlc_boyar_council_mechanic's own flavor). This path
# takes that constraint to its conclusion: the council stops confirming
# princes and starts electing a term-limited voivode outright.
# =============================================================================

hlc_boyar_republic_reform = {

	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no

	rulers_can_be_generals = yes
	allow_convert = no
	potential = {
		OR = {
			has_reform = hlc_boyar_republic_reform
			west_ukraine_hlc_origin_country = yes
		}
	}
	trigger = {
		OR = {
			has_reform = hlc_boyar_republic_reform
			AND = {
				west_ukraine_hlc_origin_country = yes
				custom_trigger_tooltip = {
					tooltip = hlc_path_boyar_req_tt
					hlc_has_boyar_path = yes
				}
			}
		}
	}
	modifiers = {
		republican_tradition = 0.25

			AND = {
				west_ukraine_hlc_origin_country = yes

				custom_trigger_tooltip = {
					tooltip = hlc_path_confirmed_metropolitanate_req_tt
					hlc_has_confirmed_metropolitanate_path = yes
				}
			}
		}
	}
	modifiers = {
		tolerance_own = 1
		global_missionary_strength = 0.01
		prestige = 1
		max_absolutism = -3
	}
	custom_attributes = {
		locked_government_type = yes
	}
	legacy_equivalent = theocratic_government
	# Real anchor: Constantinople opened the Halych see in 1303, closed it
	# within a few years, reopened it under Yurii II Boleslav around 1332, and
	# closed it again - until Patriarch Philotheos Kokkinos's 1370-71 tomos,
	# pressed for by Casimir III and Louis I, confirmed Antony as Metropolitan
	# of Halych against Kyiv's and Moscow's standing claim to the whole Rus
	# church. This reform is that confirmation, re-earned rather than assumed.
}
