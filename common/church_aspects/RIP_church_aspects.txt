# ===========================================================================
# CHURCH ASPECTS FOR THE MOD'S CHURCH-POWER RELIGIONS
# ---------------------------------------------------------------------------
# Vanilla's only church-power religion is Protestant, so vanilla never had to
# gate its aspects by faith. This mod adds two more (russian_orthodox and
# greek_catholic), which means every aspect here must say which faith it
# belongs to - otherwise the Uniate reform list offers the Muscovite
# inquisition and the Protestant one offers both.
#
# EU4 requires `cost` on every aspect. Entries that lacked it were never
# selectable and are given the standard 100 here.
# ===========================================================================


# --- shared by both eastern-rite faiths -------------------------------------

embrace_eastern_rite = {
	cost = 100
	potential = {
		OR = {
			religion = russian_orthodox
			religion = greek_catholic
		}
	}
	modifier = {
		tolerance_own = 1
		church_power_modifier = 0.1
		legitimacy = 0.5
