namespace = pdl_border_policy

# =============================================================================
# pdl_border_policy - Recurring Border Policy of Podillia
#
# Podillia occupies the vulnerable steppe and river frontier between Poland,
# the Grand Duchy of Lithuania, the Ottomans, Moldavia, and the Crimean Tatars.
#
# The country regularly faces a critical governance choice:
# 1. Fund Border Defense (Kamianets/Dniester fortress bastions or Cossack watches)
# 2. Support Settlement (Slobodas, colonization, and economic growth)
# 3. Delegate Power to Local Elite (Otamans, Boyars, or Magnates)
#
# Each of Podillia's three constitutional paths modifies the cost and effects:
# - Frontier Path (pdl_frontier_path): Relies on Cossacks and horse outposts.
# - Carpathian Path (pdl_carpathian_path): Relies on stone fortresses and local boyars.
# - Magnate Path (pdl_magnate_path): Relies on private latifundia, sejmiks, and commercial towns.
# =============================================================================

country_event = {
	id = pdl_border_policy.1
	title = pdl_border_policy.1.t
	desc = pdl_border_policy.1.desc
	picture = BORDER_TENSION_eventPicture

	is_triggered_only = yes

	# Option A: Fund Border Defense
	option = {
		name = pdl_border_policy.1.a
		add_country_modifier = {
			name = pdl_recent_border_policy
			duration = 1825
		}
		remove_country_modifier = pdl_frontier_defense_funded
		remove_country_modifier = pdl_carpathian_defense_funded
		remove_country_modifier = pdl_magnate_defense_funded
		remove_country_modifier = pdl_defense_funded
		remove_country_modifier = pdl_frontier_settlement_supported
		remove_country_modifier = pdl_carpathian_settlement_supported
		remove_country_modifier = pdl_magnate_settlement_supported
		remove_country_modifier = pdl_settlement_supported
		remove_country_modifier = pdl_frontier_elites_delegated
		remove_country_modifier = pdl_carpathian_elites_delegated
		remove_country_modifier = pdl_magnate_elites_delegated
		remove_country_modifier = pdl_elites_delegated

		if = {
			limit = { has_country_flag = pdl_frontier_path }
			add_treasury = -40
			add_mil_power = -35
			add_manpower = -1
			add_army_tradition = 5
			add_country_modifier = {
				name = pdl_frontier_defense_funded
				duration = 1825
			}
			if = {
				limit = { has_estate = estate_cossacks }
				add_estate_loyalty = {
					estate = estate_cossacks
					loyalty = 10
				}
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_carpathian_path }
			add_treasury = -80
			add_mil_power = -25
			add_country_modifier = {
				name = pdl_carpathian_defense_funded
				duration = 1825
			}
			random_owned_province = {
				limit = {
					OR = {
						province_id = 281
						province_id = 1944
						has_building = fort_15th
					}
				}
				add_garrison = 500
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_magnate_path }
			add_treasury = -100
			add_army_tradition = 5
			add_country_modifier = {
				name = pdl_magnate_defense_funded
				duration = 1825
			}
			if = {
				limit = { has_estate = estate_nobles }
				add_estate_loyalty = {
					estate = estate_nobles
					loyalty = 5
				}
			}
		}
		else = {
			add_treasury = -60
			add_mil_power = -30
			add_country_modifier = {
				name = pdl_defense_funded
				duration = 1825
			}
		}
	}

	# Option B: Support Settlements and Slobodas
	option = {
		name = pdl_border_policy.1.b
		add_country_modifier = {
			name = pdl_recent_border_policy
			duration = 1825
		}
		remove_country_modifier = pdl_frontier_defense_funded
		remove_country_modifier = pdl_carpathian_defense_funded
		remove_country_modifier = pdl_magnate_defense_funded
		remove_country_modifier = pdl_defense_funded
		remove_country_modifier = pdl_frontier_settlement_supported
		remove_country_modifier = pdl_carpathian_settlement_supported
		remove_country_modifier = pdl_magnate_settlement_supported
		remove_country_modifier = pdl_settlement_supported
		remove_country_modifier = pdl_frontier_elites_delegated
		remove_country_modifier = pdl_carpathian_elites_delegated
		remove_country_modifier = pdl_magnate_elites_delegated
		remove_country_modifier = pdl_elites_delegated

		if = {
			limit = { has_country_flag = pdl_frontier_path }
			add_treasury = -30
			add_dip_power = -40
			add_country_modifier = {
				name = pdl_frontier_settlement_supported
				duration = 1825
			}
			random_owned_province = {
				limit = {
					OR = {
						province_id = 1944
						province_id = 281
						area = podolia_volhynia_area
					}
				}
				add_base_manpower = 1
			}
			if = {
				limit = { has_estate = estate_cossacks }
				add_estate_loyalty = {
					estate = estate_cossacks
					loyalty = 10
				}
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_carpathian_path }
			add_treasury = -50
			add_adm_power = -40
			add_country_modifier = {
				name = pdl_carpathian_settlement_supported
				duration = 1825
			}
			random_owned_province = {
				limit = {
					OR = {
						province_id = 281
						province_id = 1944
						area = podolia_volhynia_area
					}
				}
				add_base_production = 1
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_magnate_path }
			add_treasury = -40
			add_adm_power = -30
			add_dip_power = -30
			add_country_modifier = {
				name = pdl_magnate_settlement_supported
				duration = 1825
			}
			random_owned_province = {
				limit = {
					OR = {
						province_id = 281
						province_id = 1944
						area = podolia_volhynia_area
					}
				}
				add_base_tax = 1
			}
			if = {
				limit = { has_estate = estate_burghers }
				add_estate_loyalty = {
					estate = estate_burghers
					loyalty = 10
				}
			}
		}
		else = {
			add_treasury = -40
			add_adm_power = -35
			add_country_modifier = {
				name = pdl_settlement_supported
				duration = 1825
			}
		}
	}

	# Option C: Delegate Power to Local Elites
	option = {
		name = pdl_border_policy.1.c
		add_country_modifier = {
			name = pdl_recent_border_policy
			duration = 1825
		}
		remove_country_modifier = pdl_frontier_defense_funded
		remove_country_modifier = pdl_carpathian_defense_funded
		remove_country_modifier = pdl_magnate_defense_funded
		remove_country_modifier = pdl_defense_funded
		remove_country_modifier = pdl_frontier_settlement_supported
		remove_country_modifier = pdl_carpathian_settlement_supported
		remove_country_modifier = pdl_magnate_settlement_supported
		remove_country_modifier = pdl_settlement_supported
		remove_country_modifier = pdl_frontier_elites_delegated
		remove_country_modifier = pdl_carpathian_elites_delegated
		remove_country_modifier = pdl_magnate_elites_delegated
		remove_country_modifier = pdl_elites_delegated

		if = {
			limit = { has_country_flag = pdl_frontier_path }
			add_prestige = -5
			if = {
				limit = { absolutism = 2 }
				add_absolutism = -2
			}
			add_manpower = 2
			add_country_modifier = {
				name = pdl_frontier_elites_delegated
				duration = 1825
			}
			if = {
				limit = { has_estate = estate_cossacks }
				add_estate_loyalty = {
					estate = estate_cossacks
					loyalty = 15
				}
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_carpathian_path }
			add_adm_power = -20
			add_treasury = 40
			add_country_modifier = {
				name = pdl_carpathian_elites_delegated
				duration = 1825
			}
			if = {
				limit = { has_estate = estate_nobles }
				add_estate_loyalty = {
					estate = estate_nobles
					loyalty = 10
				}
			}
		}
		else_if = {
			limit = { has_country_flag = pdl_magnate_path }
			if = {
				limit = { absolutism = 5 }
				add_absolutism = -5
			}
			add_treasury = 75
			add_country_modifier = {
				name = pdl_magnate_elites_delegated
				duration = 1825
			}
			if = {
				limit = { has_estate = estate_nobles }
				add_estate_loyalty = {
					estate = estate_nobles
					loyalty = 15
				}
				add_estate_influence_modifier = {
					estate = estate_nobles
					desc = pdl_delegated_elites_influence_desc
					influence = 10
					duration = 1825
				}
			}
		}
		else = {
			add_treasury = 30
			add_country_modifier = {
				name = pdl_elites_delegated
				duration = 1825
			}
		}
	}

	# Option D: Maintain Current Stance
	option = {
		name = pdl_border_policy.1.d
		add_prestige = 5
	}
}
