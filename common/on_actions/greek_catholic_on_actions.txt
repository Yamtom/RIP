	# lives in rip_uc_union_spread_effect in

	rip_uc_union_spread_effect = yes

	# What the preaching actually converts is bought with religious unity: a
	# development budget, spent one parish at a time, remainder lost. Both this

	# common/scripted_effects/rip_faith_spread_effects.txt.
	if = {
		limit = { religion = greek_catholic }

		8 = uniate_church.19  # Greek Catholic Ecumenical Council
		8 = uniate_church.20  # Greek Catholic Saints and Martyrs
		750 = 0 # Nothing
	}
}
