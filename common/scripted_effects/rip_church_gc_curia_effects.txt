	remove_country_modifier = rip_church_gc_petition_saint

	remove_country_modifier = rip_church_gc_petition_holy_war

rip_church_gc_pay_curia_petition_effect = {
	add_treasury = -100
	change_variable = { which = rip_church_papal_standing value = -$STANDING$ }
	add_country_modifier = { name = $MODIFIER$ duration = 3650 }
}

# Neither transaction below creates a cardinal, casts a vote, or changes
# Catholic Curia control. A deputation changes only bilateral diplomacy.
rip_church_gc_depute_to_curia_effect = {
	if = {
		limit = { rip_church_gc_can_depute_to_curia = yes }
		add_treasury = -50
		add_patriarch_authority = -0.1
		change_variable = { which = rip_church_papal_standing value = 5 }
		if = {
			limit = { check_variable = { which = rip_church_papal_standing value = 100 } }
			set_variable = { which = rip_church_papal_standing value = 100 }
		}
		PAP = { add_opinion = { who = ROOT modifier = rip_church_opinion_gc_deputation } }
		set_country_flag = rip_church_gc_deputation_sent
	}
}
rip_church_gc_donate_effect = {
	if = {
		limit = { rip_church_gc_can_donate = yes }
		add_treasury = -100
		change_variable = { which = rip_church_papal_standing value = 10 }
		if = {
			limit = { check_variable = { which = rip_church_papal_standing value = 100 } }
			set_variable = { which = rip_church_papal_standing value = 100 }
		}
		if = {
			limit = { has_dlc = Emperor }
			add_curia_treasury = 50
			PAP = { add_treasury = 50 }
		}
		else = { PAP = { add_treasury = 100 } }
		PAP = {
			add_opinion = { who = ROOT modifier = rip_church_opinion_gc_donation }
			ROOT = {
				export_to_variable = {
					which = rip_church_papal_opinion
					value = opinion
					with = PREV
				}
			}
		}
		set_country_flag = rip_church_gc_donated
	}
}
