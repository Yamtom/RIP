rip_accord_border_partner = {
	is_subject = no
	alliance_with = ROOT
	has_opinion = { who = ROOT value = 75 }
	any_owned_province = {
		controlled_by = owner
		has_fort_building_trigger = yes
		any_neighbor_province = {
			owned_by = ROOT
			controlled_by = ROOT
		}
	}
}

# Country scope is the sponsor. The opinion modifier identifies the actual
# signatory through tag changes. A second alliance cannot inherit the accord.
rip_sovereign_accord_active = {
	is_subject = no
	has_country_flag = $signed$
	any_ally = {
		has_opinion_modifier = { who = ROOT modifier = $opinion$ }
		$partner$ = yes
	}
}

rip_hadiach_accord_live = {
	rip_sovereign_accord_active = {
		signed = rip_hadiach_accord_signed
		opinion = rip_hadiach_accord_opinion
		partner = rip_accord_hadiach_partner
	}
}

rip_accord_offer_ready = {
	custom_trigger_tooltip = {
		tooltip = rip_accord_offer_ready_tt
		OR = {
			NOT = { has_country_flag = $refused$ }
			had_country_flag = { flag = $refused$ days = 1825 }
		}
	}
}
