# =============================================================================
# rip_ruthenian_factional_court - the ruler's hand on the rada.
#
# The three ruthenian_factional_empire_*_reform ids are ONE government whose
# rada is the vanilla FACTION mechanic (common/factions/RIP_kru_factions.txt,
# enabled by `factions = { kru_knyazi kru_boyary kru_starshyna }` in the
# reforms). Vanilla runs the influence, names the bloc in power and applies its
# modifier, and draws the faction panel. This mechanic adds three court
# buttons and three pairwise estate balances. The balances follow existing
# faction influence and shift it again when either side wins.
#
# kru_update_dominant_pair_effect (common/scripted_effects/kru_effects.txt)
# reads faction_in_power + faction_influence to set this reform id to the ruling
# PAIR and to give the second bloc a partial voice (KRU_junior_*).
#
# Buttons use vanilla Cossack ADM/DIP/MIL ability sprites; the balance powers
# use native government-power bars. No DLC gate is needed because the
# underlying faction mechanic is base-game. Faction portraits use the vanilla
# layered icon/frame convention.
# =============================================================================

ruthenian_factional_court_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	available = {
		OR = {
			has_reform = ruthenian_factional_empire_reform
			has_reform = ruthenian_factional_empire_princes_hetmans_reform
			has_reform = ruthenian_factional_empire_boyars_hetmans_reform
		}
	}

	powers = {
		# 0 favours the first faction; 100 favours the second. Existing
		# faction influence moves each balance, and reaching an end shifts
		# influence before returning the scale to the centre.
		kru_knyazi_boyary_balance = {
			gui = rip_kru_knyazi_boyary_balance_power
			min = 0
			max = 100
			default = 50
			base_monthly_growth = 0
			reset_on_new_ruler = no
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_knyazi influence = 60 }
					NOT = { faction_influence = { faction = kru_boyary influence = 60 } }
				}
				modifier = { monthly_kru_knyazi_boyary_balance = -0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_boyary influence = 60 }
					NOT = { faction_influence = { faction = kru_knyazi influence = 60 } }
				}
				modifier = { monthly_kru_knyazi_boyary_balance = 0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					diplomatic_reputation = 0.5
					vassal_income = 0.05
				}
				start_value = 0
				end_value = 39
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					global_tax_modifier = 0.05
					advisor_cost = -0.05
				}
				start_value = 61
				end_value = 100
			}
			on_min_reached = {
				add_faction_influence = { faction = kru_knyazi influence = 6 }
				add_faction_influence = { faction = kru_boyary influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_knyazi_boyary_balance
					value = 50
				}
			}
			on_max_reached = {
				add_faction_influence = { faction = kru_boyary influence = 6 }
				add_faction_influence = { faction = kru_knyazi influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_knyazi_boyary_balance
					value = 50
				}
			}
		}
		kru_boyary_starshyna_balance = {
			gui = rip_kru_boyary_starshyna_balance_power
			min = 0
			max = 100
			default = 50
			base_monthly_growth = 0
			reset_on_new_ruler = no
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_boyary influence = 60 }
					NOT = { faction_influence = { faction = kru_starshyna influence = 60 } }
				}
				modifier = { monthly_kru_boyary_starshyna_balance = -0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_starshyna influence = 60 }
					NOT = { faction_influence = { faction = kru_boyary influence = 60 } }
				}
				modifier = { monthly_kru_boyary_starshyna_balance = 0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					stability_cost_modifier = -0.05
					trade_efficiency = 0.05
				}
				start_value = 0
				end_value = 39
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					manpower_recovery_speed = 0.1
					land_morale = 0.025
				}
				start_value = 61
				end_value = 100
			}
			on_min_reached = {
				add_faction_influence = { faction = kru_boyary influence = 6 }
				add_faction_influence = { faction = kru_starshyna influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_boyary_starshyna_balance
					value = 50
				}
			}
			on_max_reached = {
				add_faction_influence = { faction = kru_starshyna influence = 6 }
				add_faction_influence = { faction = kru_boyary influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_boyary_starshyna_balance
					value = 50
				}
			}
		}
		kru_knyazi_starshyna_balance = {
			gui = rip_kru_knyazi_starshyna_balance_power
			min = 0
			max = 100
			default = 50
			base_monthly_growth = 0
			reset_on_new_ruler = no
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_knyazi influence = 60 }
					NOT = { faction_influence = { faction = kru_starshyna influence = 60 } }
				}
				modifier = { monthly_kru_knyazi_starshyna_balance = -0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = {
					faction_influence = { faction = kru_starshyna influence = 60 }
					NOT = { faction_influence = { faction = kru_knyazi influence = 60 } }
				}
				modifier = { monthly_kru_knyazi_starshyna_balance = 0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					legitimacy = 0.5
					vassal_forcelimit_bonus = 0.05
				}
				start_value = 0
				end_value = 39
			}
			range_modifier = {
				trigger = { always = yes }
				modifier = {
					discipline = 0.01
					land_maintenance_modifier = -0.05
				}
				start_value = 61
				end_value = 100
			}
			on_min_reached = {
				add_faction_influence = { faction = kru_knyazi influence = 6 }
				add_faction_influence = { faction = kru_starshyna influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_knyazi_starshyna_balance

				}
			}
			on_max_reached = {
				add_faction_influence = { faction = kru_starshyna influence = 6 }
				add_faction_influence = { faction = kru_knyazi influence = -6 }
				set_government_power = {
					mechanic_type = ruthenian_factional_court_mechanic
					power_type = kru_knyazi_starshyna_balance
					value = 50
				}
			}
		}
	}

	interactions = {
		# Court the princes of the blood - honours, precedence, a hearing at
		# every rada. Their standing in the faction rises; the crown's own
		# legitimacy rises with theirs.
		kru_court_the_knyazi = {
			icon = GFX_cossacks_ability_DIP
			trigger = {
				dip_power = 50
				NOT = { faction_influence = { faction = kru_knyazi influence = 70 } }
			}
			effect = {
				add_dip_power = -50
				add_faction_influence = {
					faction = kru_knyazi
					influence = 25
				}
				add_legitimacy_equivalent = { amount = 2 republican_tradition = 1 }
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_knyazi }
					faction_influence = { faction = kru_knyazi influence = 40 }
				}
			}
		}
		# Court the boyar houses and the chartered towns - offices, charters, a
		# share of the customs. Their standing rises; the treasury takes it in.
		kru_court_the_boyary = {
			icon = GFX_cossacks_ability_ADM
			trigger = {
				adm_power = 50
				NOT = { faction_influence = { faction = kru_boyary influence = 70 } }

			effect = {
				add_adm_power = -50
				add_faction_influence = {
					faction = kru_boyary
					influence = 25
				}

			}

			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_boyary }
					faction_influence = { faction = kru_boyary influence = 40 }
				}
			}
		}
		# Court the Cossack starshyna - regimental appointments, land for the

		kru_court_the_starshyna = {

			ai_chance = {
				factor = 1
				modifier = {
					factor = 8
					is_at_war = yes
				}
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_starshyna }
					faction_influence = { faction = kru_starshyna influence = 40 }
				}
			}
		}
	}
}
