# Blessings of the Russian Orthodox church.
#
# These are blessings, not aspects, and the difference is not cosmetic: all ten
# vanilla blessings carry `is_blessing = yes` and none carries a `cost`, while
# all forty-six vanilla church aspects are the inverse. A blessing is unlocked
# by holding and converting a holy site, is free, and is permanent until a site
# is lost. An aspect is bought with church power.
#
# The seven below were written as aspects, which is why nothing could ever
# enact them: the faith declares blessings, and an aspect-shaped entry in a
# blessings list is not a blessing. Payloads are unchanged apart from
# church_power_modifier, which is meaningless now that this faith does not use
# church power.
#
# Seven are offered against six holy sites, so a country holding every site
# still has to leave one behind.

blessing_of_third_rome = {
	is_blessing = yes

	modifier = {
		prestige_decay = -0.01
		legitimacy = 1
		ae_impact = -0.10
	}

	ai_will_do = {
		factor = 10
		modifier = {
			factor = 3
			NOT = { legitimacy = 80 }
		}
	}
}

blessing_of_patriarch_of_moscow = {
	is_blessing = yes

	modifier = {
		church_loyalty_modifier = 0.1
		stability_cost_modifier = -0.15
	}

	ai_will_do = {
		factor = 10
		modifier = {
			factor = 3
			NOT = { stability = 1 }
		}
	}
}

blessing_of_orthodox_conquest = {
	is_blessing = yes

	modifier = {
		core_creation = -0.15
		province_warscore_cost = -0.10
	}

	ai_will_do = {
		factor = 10
		modifier = {
			factor = 3
			is_at_war = yes
		}
	}
}

blessing_of_cultural_dominance = {
	is_blessing = yes

	modifier = {
		culture_conversion_cost = -0.25
		culture_conversion_time = -0.15
	}

	ai_will_do = {
		factor = 8
		modifier = {
			factor = 3
			NOT = {
				num_of_owned_provinces_with = {
					value = 5
					culture = ROOT
				}
			}
		}
	}
}
