	# --- the Host is a camp before it is a capital, and a camp can be moved.
	# zaz_relocate_sich in ZaporizhiaDecisions.txt moves the Sich inside the
	# old country's own steppe. This one is the harder move: a whole Cossack
	# polity carrying its seat somewhere it does not yet hold, which is what
	# the Host did in 1709 and again in 1775.

	cossack_carry_the_host = {

			}
		}
		provinces_to_highlight = {
			is_host_refuge = yes
			NOT = { owned_by = ROOT }
		}
		allow = {
			adm_power = 100
			mil_power = 100
			custom_trigger_tooltip = {
				tooltip = cossack_carry_the_host_tt
				any_known_country = {
					any_owned_province = {
						is_host_refuge = yes
						NOT = { owned_by = ROOT }
					}
					OR = {
						alliance_with = ROOT
						is_subject_of = ROOT
						has_opinion = { who = ROOT value = 100 }
					}
				}
			}
		}
		effect = {
			add_adm_power = -100
			add_mil_power = -100

		}
