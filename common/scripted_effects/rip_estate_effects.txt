# One estate helper, because the same six lines were written eight times.
#
# Every event in this mod that nudges an estate had to guard the call: an
# add_estate_loyalty on a country that does not have that estate is a silent
# no-op at best. So the guard and the call were copied together, over and over.
#
# Parameterised scripted effects are vanilla's own answer to that - see
# add_or_extend_province_modifier_effect in
# common/scripted_effects/00_scripted_effects.txt, which takes $MODIFIER$ and
# $DURATION$ the same way.
#
# Call it as:
#     rip_estate_mood_effect = { ESTATE = estate_magnates LOYALTY = 15 }

rip_estate_mood_effect = {
	if = {
		limit = { has_estate = $ESTATE$ }
		add_estate_loyalty = {
			estate = $ESTATE$
			loyalty = $LOYALTY$
		}
	}
}

# Fired from on_province_culture_converted
# (common/on_actions/west_ukraine_on_actions.txt) while the Chancery's
# Settlement Ordinance is active
# (common/government_mechanics/rip_hlc_austrian_reorganization.txt). Runs in
# province scope (ROOT is the converted province; vanilla's own handlers of the
# same hook switch to `owner` the same way) and reaches the country through
# `owner`. The flag it reads is set only by that mechanic's own interaction, so
# no country-tag guard is needed here.
#
# The hook fires only for a MANUAL conversion, the player's or the AI's use of
# the province culture button: vanilla marks it "province, manually conversion",
# and conversions made by effects fire on_province_culture_changed instead.
#
# Resettled land goes to the nobility, so the nobles are the only estate that
# is steadied, by 2 loyalty. It is bounded twice, so that assimilating province
# after province can never become a source of general approval:
#  - never above 60% loyalty (the nobles are already content);
#  - at most once a year - a hidden country modifier is the timer, the way
#    vanilla times wotr_one_year_timer (events/disaster_ENG_WarOfTheRoses.txt):
#    an empty event modifier added with hidden = yes and a duration, tested by
#    has_country_modifier.
# The timer is only added together with the loyalty, and the limit requires the
# modifier to be absent, so a running timer is never added a second time.
hlc_settlement_ordinance_conversion_effect = {
	if = {
		limit = {
			owner = {
				has_country_flag = hlc_settlement_ordinance_active
				has_estate = estate_nobles
				NOT = { has_country_modifier = hlc_settlement_loyalty_timer }
				NOT = {
					estate_loyalty = {
						estate = estate_nobles
						loyalty = 60
					}
				}
			}
		}
		owner = {
			rip_estate_mood_effect = { ESTATE = estate_nobles LOYALTY = 2 }
			hidden_effect = {
				add_country_modifier = {
					name = hlc_settlement_loyalty_timer
					duration = 365
					hidden = yes
				}
			}
		}
	}
}
