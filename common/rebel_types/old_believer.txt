######################################################################
# OLD BELIEVER RELIGIOUS ZEALOTS - SHARED FACTION
# This is a religion-based faction, not a country-specific rebel type. Its
# Raskol spawn requests are issued by the Russian church events and effects.
# Keep the single engine ID: old_believer_rebels.
# Structure follows vanilla orthodox_rebels; religion predicates target
# old_believer. Country-specific branches are labelled below.
######################################################################
old_believer_rebels = {

	color = { 230 175 175 }

	area = nation 					# Sticks to provinces of their nation
	government = theocracy 			# Desires Theocracy
	religion = old_believer
	defection = none				# Will not defect
	defect_delay = 120				# Retained from the vanilla template; defection is disabled above.
	independence = none				# Does not strive for independence
	unit_transfer = no
	gfx_type = culture_province
	will_relocate = yes

	resilient = no
	reinforcing = yes
	general = yes
	smart = yes

	artillery = 0.0
	infantry = 0.7
	cavalry = 0.3

	morale = 1.1

	# Possible handle actions
	handle_action_negotiate = yes
	handle_action_stability = yes
	handle_action_build_core = yes
	handle_action_send_missionary = yes

	# The rebel type with the highest modifier for this province gets picked
	spawn_chance = {
		factor = 1
		modifier = {
			factor = 0.5
			is_overseas = yes
		}
		modifier = {
			factor = 0.0
			owner = { religion = ROOT }
		}
		modifier = {
			factor = 0.0
			NOT = { religion = old_believer }
		}
		modifier = {
			factor = 0.001
			tolerance_to_this = 5
		}
		modifier = {
			factor = 0.5
			tolerance_to_this = 2
		}
		modifier = {
			factor = 5.0
			owner = { has_disaster = religious_turmoil }
		}
		modifier = {
			factor = 2000
			has_missionary = yes
		}
		modifier = {
			factor = 2
			NOT = { tolerance_to_this = 1 }
		}
		modifier = {
			factor = 2
			NOT = { tolerance_to_this = -1 }
		}
		modifier = {
			factor = 2
			NOT = { tolerance_to_this = -2 }
		}
	}

 	# This is checked for EACH province in the Area of Operations
	movement_evaluation = {
		factor = 1
		modifier = {
			factor = 0.5
			units_in_province = 1
		}
		modifier = {
			factor = 0.001
			controlled_by = REB
		}
		modifier = {
			factor = 0.0
			religion = old_believer
			owner = { religion = old_believer }
		}
		modifier = {
			factor = 2.0
			NOT = { religion = old_believer }
			any_neighbor_province = {
	 			religion = old_believer
	 			NOT = { has_province_modifier = "forced_conversion" }
	 		}
	 	}
		modifier = {
			factor = 5
			has_missionary = yes
			religion = old_believer
		}
		modifier = {
			factor = 1.5
			unrest = 2
		}
		modifier = {
			factor = 1.5
			unrest = 4
		}
		modifier = {
			factor = 1.5
			unrest = 6
		}
	}

	# PAP-only capital protection retained from the vanilla Orthodox template.
	siege_won_trigger = {
		NOT = {
			AND = {
				owned_by = PAP
				is_capital = yes
			}
		}
 	}
	siege_won_effect = {
		custom_tooltip = religious_rebels_seige_effect_tooltip
		hidden_effect = {
			if = {
				limit = {
					OR = {
						religion = old_believer
						is_reformation_center = yes
						owner = { religion = old_believer }
					}
				}
				add_province_modifier = {
					name = "religious_violence"
					duration = 730
				}
			}
			if = {
				limit = {
					NOT = { religion = old_believer }
					is_reformation_center = no
					NOT = { owner = { religion = old_believer } }
					# BYZ exception retained from the vanilla Orthodox template.
					owner = { NOT = { has_country_flag = BYZ_gave_in_to_papal_pressure } }
				}
				change_religion = old_believer
				add_province_modifier = {
					name = "forced_conversion"
					duration = 730
				}
			}
		}
		owner = {
			change_estate_land_share = {
				estate = estate_church
				share = 10.0
				province = ROOT
			}
		}
	}

	# Shared country rules; the rebellion can exist under any owner of an
	# old_believer province, while the intended scripted spawn is Russian.
	can_negotiate_trigger = {
		OR = {
			religion_group = christian
			dominant_religion = old_believer
		}
	}

	# Country scope
	can_enforce_trigger = {
		always = yes
	}

	# Localisation for their demands
	demands_description = "old_believer_rebels_demand"

	# Country Scope
	demands_enforced_effect = {
		hidden_effect = {
			add_country_modifier = {
				name = just_lost_to_rebels
				duration = 10
				hidden = yes
			}
		}
		if = {
			limit = {
				NOT = { religion_group = christian }
			}
			add_prestige = -50
			every_owned_province = {
				limit = {
					has_rebel_faction = REB
				}
				add_local_autonomy = 20
			}
		}
		# Country exception inherited from vanilla Orthodox rebels.
		if = {
			limit = {
				tag = PAP
			}
			add_stability = -1
		}
		else_if = {
			limit = {
				NOT = { religion = old_believer }
				dominant_religion = old_believer
				NOT = { tag = PAP }
			}
			rip_ob_leave_patriarchate_effect = yes
			change_religion = old_believer
			force_converted = yes
		}
		else_if = {
			limit = {
				religion = old_believer
			}
			add_country_modifier = {
				name = "religious_intolerance_enforced"
				duration = 3650
			}
		}
		else_if = {
			limit = {
				NOT = { religion = old_believer }
				NOT = { dominant_religion = old_believer }
				religion_group = christian
				NOT = { has_country_modifier = heretic_tolerance }
			}
			add_country_modifier = {
				name = "heretic_tolerance"
				duration = 3650
			}
		}
		else = { add_stability = -1 } #Should probably never get here
		if = {
			limit = {
				religion = old_believer
				has_estate = estate_church
				NOT = {
					has_estate_influence_modifier = {
						estate = estate_church
						modifier = EST_VAL_RELIGIOUS_ZEALOTS_TRIUMPH
					}
				}
			}
			add_estate_influence_modifier = {
				estate = estate_church
				desc = EST_VAL_RELIGIOUS_ZEALOTS_TRIUMPH
				duration = 3650
				influence = 15
			}
		}
		change_estate_land_share = {
			estate = estate_church
			share = 10.0
		}
	}
}
