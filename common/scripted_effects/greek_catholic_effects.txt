
convert_to_greek_catholic_effect = {
	change_religion = greek_catholic
	add_province_modifier = {
		name = greek_catholic_diocese
		duration = 5475
	}
	if = {
		limit = { has_building = temple }
		add_base_tax = 1
	}
	if = {
		limit = { has_building = cathedral }
		add_base_tax = 2
	}
}

establish_greek_catholic_center_effect = {
	change_religion = greek_catholic
	add_province_modifier = {
		name = greek_catholic_center
		duration = -1
	}
	add_base_tax = 2
	add_base_production = 1
	add_base_manpower = 1
	if = {
		limit = {
			NOT = { has_building = temple }
			NOT = { has_building = cathedral }
		}
		add_building = temple
	}
}

spread_greek_catholic_to_neighbors = {
	every_neighbor_province = {
		limit = {
			OR = {
				religion = orthodox
				religion = catholic
			}
			owned_by = ROOT
		}
		random = {
			chance = 30
			change_religion = greek_catholic
			add_province_modifier = {
				name = greek_catholic_diocese
				duration = 3650
			}
		}
	}
}

# ---------------------------------------------------------------- the Curia
#
# Rome as the union actually met it: a court you have standing at, not a college you
# sit in. rip_uc_standing runs 0 to 100. It decays, because a court that hears
# nothing from a province forgets it, and it is raised by sending money - which is
# what the Ruthenian sees did, and complained about doing, for two centuries.

rip_uc_standing_tick_effect = {
	if = {
		limit = {
			religion = greek_catholic
			check_variable = { which = rip_uc_standing value = 3 }
		}
		subtract_variable = {
			which = rip_uc_standing
			value = 2
		}
	}
	# a legate resident at the court keeps the name in front of it
	if = {
		limit = {
			religion = greek_catholic
			has_country_modifier = papal_legate
			NOT = { check_variable = { which = rip_uc_standing value = 100 } }
		}
		change_variable = {
			which = rip_uc_standing
			value = 3
		}
	}
	# Same guard as the Sobor's: on_bi_yearly_pulse is global, and unguarded this
	# put "The Curia: unknown at Rome" on every country on the map. The else
	# sheds the band for a country that leaves the faith.
	if = {
		limit = { religion = greek_catholic }
		rip_uc_show_the_standing_effect = yes
	}
	else = {
		rip_uc_clear_the_bands_effect = yes
	}
}

# Every band off, and none back on.
rip_uc_clear_the_bands_effect = {
	hidden_effect = {
		remove_country_modifier = rip_uc_curia_unknown
		remove_country_modifier = rip_uc_curia_heard
		remove_country_modifier = rip_uc_curia_favoured
		remove_country_modifier = rip_uc_curia_protected
	}
}

rip_uc_show_the_standing_effect = {
	hidden_effect = {
		remove_country_modifier = rip_uc_curia_unknown
		remove_country_modifier = rip_uc_curia_heard
		remove_country_modifier = rip_uc_curia_favoured
		remove_country_modifier = rip_uc_curia_protected
		if = {
			limit = { check_variable = { which = rip_uc_standing value = 75 } }
			add_country_modifier = { name = rip_uc_curia_protected duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = rip_uc_standing value = 50 } }
			add_country_modifier = { name = rip_uc_curia_favoured duration = -1 }
		}
		else_if = {
			limit = { check_variable = { which = rip_uc_standing value = 25 } }
			add_country_modifier = { name = rip_uc_curia_heard duration = -1 }
		}
		else = {
			add_country_modifier = { name = rip_uc_curia_unknown duration = -1 }
		}
	}
}

# A donation. The see pays, the court remembers, and it will not be asked twice in
# the same three years.
rip_uc_donation_effect = {
	add_treasury = -150
	change_variable = {
		which = rip_uc_standing
		value = 12
	}
	if = {
		limit = { check_variable = { which = rip_uc_standing value = 101 } }
		set_variable = {
			which = rip_uc_standing
			value = 100
		}
	}
	add_country_modifier = {
		name = rip_uc_donation_recently_sent
		duration = 1095
	}
	rip_uc_show_the_standing_effect = yes
}

# A petition spends the standing it needs. The court does a favour once and then
# expects to be reminded who asked for it.
rip_uc_spend_the_standing_effect = {
	subtract_variable = {
		which = rip_uc_standing
		value = 15
	}
	rip_uc_show_the_standing_effect = yes
}
