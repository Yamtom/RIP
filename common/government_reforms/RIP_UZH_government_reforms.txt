# The Palanok captaincy: an elective frontier state in which the castle
# captains and the Sejm of merchant magnates share the realm. Its second layer
# is the Res Publica "scale" (states_general_mechanic) with two custom sides,
# partia_soymu (left end, -1) and partia_kapitaniv (right end, +1). Vanilla's
# bar is positional: the first block is the left end, the second the right end,
# whatever they are called (militarists/theocrats, revolutionaries/monarchists).
# What moves the bar (every hook goes through
# common/scripted_effects/uzh_states_general_effects.txt and is guarded by this
# reform and by has_states_general_mechanic): a war won against an equal or
# larger foe (on_war_won in common/on_actions/00_on_actions.txt), the three
# Palanok fortress events (events/PalanokFortress.txt) and three missions
# (missions/Zakarpatta_Missions.txt). The term election itself moves it by 0.33
# either way: vanilla's elections.720 fires from on_new_term_election for every
# holder of a states_general_mechanic.
uzh_palanok_captaincy_reform = {
	icon = "rip_gfx_reform_uzh_palanok_captaincy_icon"
	legacy_equivalent = dutch_republic_legacy
	potential = {
		OR = {
			has_reform = uzh_palanok_captaincy_reform
			tag = UZH
		}
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	lock_level_when_selected = yes
	# Dynastic fields. Vanilla never combines has_term_election with
	# queen = yes: every term-election reform sets queen = no and heir = no
	# (states_general_reform, livonian_naval_monarchy, dutch_theocracy_reform).
	# The only vanilla rulers who are chosen rather than born and still marry
	# are the soldier kings (commander_king_reform, admiral_king_reform,
	# livonian_mercenary_monarchy): queen = yes, heir = no, royal_marriage = yes.
	# Palanok follows that second pattern. royal_marriage stays on because the
	# monarchy branch of uz_resist_habsburgs asks for a royal marriage
	# (missions/Zakarpatta_Missions.txt). Not observed in a played campaign:
	# whether an elected captain's consort survives the next election.
	queen = yes
	heir = no
	duration = 20
	has_term_election = yes
	rulers_can_be_generals = yes
	royal_marriage = yes
	allow_convert = no
	trigger = {
		OR = {
			has_reform = uzh_palanok_captaincy_reform
			tag = UZH
		}
	}
	modifiers = {
		defensiveness = 0.1
		fort_maintenance_modifier = -0.1
		# The captaincy's structural drag, constant whichever side is in
		# power; the swing sits on the two sides below. Kept literal
		# (check_government_reforms.py) so the burden never becomes a no-op
		# under one party. Net land_morale with the sides: Sejm in power
		# -5%, Captains in power +5% (-5% here, +10% on partia_kapitaniv).
		land_morale = -0.05
		global_unrest = 1
	}
	custom_attributes = {
		locked_government_type = yes
		enables_plutocratic_idea_group = yes
		enables_aristocratic_idea_group = no
	}
	# The scale needs Res Publica or Winds of Change, like every vanilla
	# states_general_mechanic block; without either the reform is the same
	# elective captaincy with its structural modifiers and no bar. Side
	# modifiers avoid republican_tradition and legitimacy because the reform
	# is offered in both a monarchy (M1) and a republic (R1) tier.
	conditional = {
		allow = {
			OR = {
				has_dlc = "Res Publica"
				has_dlc = "Winds of Change"
			}
		}
		states_general_mechanic = {
			# Left end: the Sejm of merchant magnates and guild elders.
			partia_soymu = {
				trade_efficiency = 0.10
				global_trade_power = 0.10
				land_forcelimit_modifier = -0.10
			}

			# Right end: the castle captains and their haiduk levies.
			partia_kapitaniv = {
				land_morale = 0.10
				global_manpower_modifier = 0.10
				trade_efficiency = -0.10
			}
		}
	}
}

# The Eparchial Road - a genuine theocracy exit for UZH, not another add-on
# to a secular crown. decisions/UzhEparchyPath.txt fires change_government =
# theocracy and hands this reform over once
# common/scripted_triggers/uzh_eparchy_triggers.txt is satisfied. Built on
# the Union of Uzhhorod (uzh_union_synod_reform above) rather than
# duplicating it: that reform never registers under theocracy's
# reform_levels (common/governments/00_governments.txt), so change_government
# clears it the same way it already clears uzh_komitat_system_reform and
# its siblings on a type switch (events/Uzh.txt, uzh.204). It also stays
# clear of the Greek Catholic religion's own hierarchy mechanics
# (common/religions/zz_greek_catholic.txt, decisions/GreekCatholicDecisions.txt)
# - those are read, never edited, here. Same celibate-elective shape as
# kyivan_metropolitanate_reform and zaz_sacred_host_order_reform: a
# consecrated bishop cannot found a dynasty, so queen/heir/royal_marriage
# stay off.
#
# Restored 2026-09-27: e3d2c80d deleted this block ("simplification and
# integration of theocracy mechanics") while leaving every reference to it
# intact (decisions/UzhEparchyPath.txt still grants it, common/governments/
# 00_governments.txt still lists it in a tier, common/government_names/
# 000_RIP_names.txt still names its ranks) - check_government_reforms.py
# failed (114 expected, 118 found) the moment that happened. No replacement
# mechanic exists anywhere in the church-redesign files for the same role
# (head of state via consecrated eparch), so this restores the working
# reform rather than guessing at an unfinished integration.
# =============================================================================

# Two further chapters for the eparchy, on the model of HLC's own two-fork
# continuation (hlc_halych_synodal_confirmation_reform,
# hlc_lviv_uniate_synod_reform - RIP_HLC_government_reforms.txt): each sits in
# its own reform_levels category (common/governments/00_governments.txt) so
# both stack on top of uzh_mukachevo_eparchy_reform instead of replacing it,
# and neither requires the other.
#
# Chronology check performed before writing these: decisions/UzhEparchyPath.txt
# already gates uzh_mukachevo_eparchy_reform's own grant on is_year = 1771 -
# the historical bull Eximia Regalium Principum that freed Mukachevo from
# Eger's Latin jurisdiction for good (the reform's own English desc already
# says as much: "Freed from Eger's tutelage"). Bishop Vasyl Tarasovych's 1646
# fight for the same jurisdictional independence therefore predates this
# mod's own Th1 gate and cannot be a later tier without making Th1
# unreachable before it. Rather than reopen the just-restored Th1 decision,
# both new tiers are anchored to real, later chapters of the same see:
#
# Th2 - Bishop Andriy Bachynskyi (1772-1809) moved the bishopric's own
# residence and chancery from Mukachevo Castle into Uzhhorod itself, founding
# the town's theological seminary in 1778. The see's institutional life
# finally catches up to the country's own name.
# Th3 - in 1818 Rome carved the Presov eparchy for the northern Rusyn
# parishes out of Mukachevo's own territory - the mature see outgrowing a
# single cathedra, not a second union.
# =============================================================================
