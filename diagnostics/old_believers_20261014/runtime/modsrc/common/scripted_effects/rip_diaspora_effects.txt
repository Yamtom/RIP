# Effects behind the Ruthenian Diaspora settler decisions
# (decisions/RIP_RuthenianDiaspora.txt), fired from within the small events
# in events/RIP_RuthenianDiaspora.txt rather than silently from the decision
# itself. Each finds one random qualifying province held by the foreign
# partner identified by the matching trigger in rip_diaspora_triggers.txt and
# brings it over to our own primary culture - the "faith" variants also
# bring it to our religion, the costlier option offered to the player.
#
# $REGION$ must match the region/colonial_region used by the calling
# trigger; $THRESHOLD$ must match the opinion floor already checked before
# the event fired, so random_country cannot land on an unqualified partner.

rip_diaspora_convert_colonial_culture_effect = {
	random_country = {
		limit = {
			any_owned_province = { colonial_region = $REGION$ }
			OR = {
				AND = {
					is_colonial_nation = yes
					has_opinion = { who = ROOT value = $THRESHOLD$ }
					overlord = { has_opinion = { who = ROOT value = $THRESHOLD$ } }
				}
				AND = {
					NOT = { is_colonial_nation = yes }
					NOT = { tag = ROOT }
					has_opinion = { who = ROOT value = $THRESHOLD$ }
				}
			}
		}
		random_owned_province = {
			limit = {
				colonial_region = $REGION$
				NOT = { rip_diaspora_eligible_culture_province = yes }
			}
			change_culture = ROOT
		}
	}
}
