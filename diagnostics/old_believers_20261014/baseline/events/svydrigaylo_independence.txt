country_event = {
	id = svydrigaylo.001
	title = "svydrigaylo.001.t"
	desc = "svydrigaylo.001.d"
	picture = PROTESTANT_CATHOLIC_DEBATE_eventPicture
	
	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					war_exhaustion = 5
					NOT = { army_strength = { who = FROM value = 1 } }
				}
				set_country_flag = overlord_accepts_independence
			}
			else = {
				set_country_flag = overlord_denies_independence
			}
		}
	}
	
	option = {
		name = "svydrigaylo.001.a"
		trigger = { has_country_flag = overlord_accepts_independence }
		
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = ruthenian_independence_accepted
				years = 20
			}
		}
		add_prestige = 20
		add_stability = 1
	}
	
	option = {
		name = "svydrigaylo.001.b"
		trigger = { has_country_flag = overlord_denies_independence }
		
		FROM = {
			add_casus_belli = {
				type = cb_vassalize_mission
				target = ROOT
				months = 120
			}
		}
	}
}

country_event = {
	id = svydrigaylo.300
	title = "svydrigaylo.300.t"
	desc = "svydrigaylo.300.d"
	picture = SEJM_eventPicture

	is_triggered_only = yes

	option = { # Recognise the confederation
		name = "svydrigaylo.300.a"

		add_opinion = {
			who = FROM
			modifier = lithuania_recognizes_ruthenian_independence
		}
		add_prestige = -15
		add_stability = 1

		ai_chance = {
			factor = 40
			modifier = {
				factor = 3
				is_at_war = yes
			}
			modifier = {
				factor = 0.3
				army_strength = { who = FROM value = 2 }
			}
		}
	}

	option = { # The palatinates are ours by charter
		name = "svydrigaylo.300.b"

		add_casus_belli = {
			target = FROM
			type = cb_vassalize_mission
			months = 120
		}
		add_prestige = 10

		ai_chance = {
			factor = 60
			modifier = {
				factor = 0.2
				is_at_war = yes
			}
		}
	}
}
