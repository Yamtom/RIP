namespace = prl_flavor

country_event = {
	id = prl_flavor.1
	title = "prl_flavor.1.t"
	desc = "prl_flavor.1.d"
	picture = DIPLOMACY_eventPicture
	fire_only_once = yes
	trigger = {
		prl_has_legacy_missions = yes
		NOT = { has_country_flag = pereiaslav_treaty_signed }
		NOT = { has_country_flag = rip_pereiaslav_council_declined }
		owns = 2407
		is_year = 1654
		NOT = {
			is_year = 1656
		}
		is_at_war = yes
		OR = {
			war_with = PLC
			war_with = POL
		}
		OR = {
			AND = {
				exists = RUS
				RUS = {
					is_subject = no
					NOT = {
						war_with = ROOT
					}
				}
			}
			AND = {
				exists = MOS
				MOS = {
					is_subject = no
					NOT = {
						war_with = ROOT
					}
				}
			}
		}
		NOT = {
			is_subject = yes
		}
	}
	mean_time_to_happen = {
		months = 6
		modifier = {
			factor = 0.25
			has_country_flag = prl_pereiaslav_council_prepared
		}
	}
	option = {
		name = "prl_flavor.1.a"
		set_country_flag = prl_pereiaslav_oath
		set_country_flag = pereiaslav_treaty_signed
		if = {
			limit = {
				exists = RUS
				RUS = {
					is_subject = no
					NOT = {
						war_with = ROOT
					}
				}
			}
			RUS = {
				create_subject = {
					subject_type = vassal
					subject = ROOT
				}
				add_opinion = {
					who = ROOT
					modifier = het_pereiaslav_oath
				}
			}
		}
		if = {
			limit = {
				OR = {
					NOT = {
						exists = RUS
					}
					RUS = {
						OR = {
							is_subject = yes
							war_with = ROOT
						}
					}
				}
				exists = MOS
				MOS = {
					is_subject = no
					NOT = {
						war_with = ROOT
					}
				}
			}
			MOS = {
				create_subject = {
					subject_type = vassal
					subject = ROOT
				}
				add_opinion = {
					who = ROOT
					modifier = het_pereiaslav_oath
				}
			}
		}
		add_country_modifier = {
			name = prl_russian_protection
			duration = -1
		}
		add_prestige = -5
		clr_country_flag = prl_pereiaslav_council_prepared
	}
	option = {
		name = "prl_flavor.1.b"
		set_country_flag = rip_pereiaslav_council_declined
		add_prestige = 15
		add_legitimacy_equivalent = {
			amount = 10
			republican_tradition = 5
		}
		add_war_exhaustion = -1
		add_country_modifier = {
			name = prl_defiant_council
			duration = 3650
		}
		clr_country_flag = prl_pereiaslav_council_prepared
	}
}

country_event = {
	id = prl_flavor.5
	title = "prl_flavor.5.t"
	desc = "prl_flavor.5.d"
	picture = TRADE_eventPicture
	is_triggered_only = yes

	option = {
		name = "prl_flavor.5.a"
		add_country_modifier = {
			name = prl_river_control
			duration = -1
		}
		add_navy_tradition = 15
		add_mercantilism = 2
		2407 = {
			if = {
				limit = { owned_by = ROOT }
				if = {
					limit = { has_building = marketplace }
					add_building = trade_depot
				}
				else_if = {
					limit = { has_building = trade_depot }
					add_building = stock_exchange
				}
				else_if = {
					limit = { NOT = { has_building = stock_exchange } }
					add_building = marketplace
				}
				else = {
					add_base_production = 2
				}
			}
		}
	}

	option = {
		name = "prl_flavor.5.b"
		add_country_modifier = {
			name = prl_river_control
			duration = 9125
		}
		add_mil_power = 75
		east_dniepr_area = {
			limit = { owned_by = ROOT }
			add_base_production = 1
		}
		2407 = {
			if = {
				limit = { owned_by = ROOT }
				if = {
					limit = { has_building = workshop }
					add_building = counting_house
				}
				else_if = {
					limit = { NOT = { has_building = counting_house } }
					add_building = workshop
				}
				else = {
					add_base_production = 2
				}
			}
		}
	}
