# Podillian path reforms remain available after PDL forms Volhynia, but only
# the institutional road selected by decision is shown. A subject keeps the
# same access: these are internal settlements, not independence checks.

pdl_magnate_reforms_visible = {
	OR = {
		has_reform = pdl_aristocratic_assembly_reform
		has_reform = pdl_magnate_republic_reform
		has_reform = pdl_magnate_dominion_reform
		has_reform = pdl_absolute_dominion_reform
		AND = {
			west_ukraine_pdl_origin_country = yes
			has_country_flag = pdl_magnate_path
		}
	}
}
