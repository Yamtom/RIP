# The mod's first diplomatic actions.
#
# RIP had none at all. WFAS showed the shape: an action is a category, a
# visibility trigger, an allow trigger, and an on_accept effect - and it can
# change a subject's standing without an event or a decision.
#
# Both of these are the mod's own subject rather than borrowed content. The
# first is the offer the Commonwealth actually made to the Cossacks and then
# failed to ratify; the second is the overlord-side counterpart, and it is the
# WFAS idea - one-sided, no acceptance needed, a subject's standing changed by
# the overlord's own hand.

# --- Hadiach: the third member.
#
# In 1658 Vyhovsky and the Crown agreed a Grand Duchy of Ruthenia standing
# beside Poland and Lithuania - its own chancery, its own courts, its own army,
# its own seat at the diet. The Sejm ratified a thinner version and the war
# resumed. Here it is an offer that can actually be made and actually accepted.
rip_offer_hadiach_terms = {
	category = influence

	require_acceptance = yes

	is_visible = {
		OR = {
			tag = PLC
			tag = POL
			tag = LIT
		}
		is_subject = no
		FROM = {
			is_subject = no
			NOT = { war_with = ROOT }
			OR = {
				religion = orthodox
				religion = russian_orthodox
				religion = greek_catholic
			}
			any_owned_province = {
				region = ruthenia_region
			}
		}
		NOT = { has_country_flag = rip_hadiach_offered }
	}

	is_allowed = {
		is_at_war = no
		adm_power = 100
		dip_power = 100
		FROM = {
			is_at_war = no
			NOT = { war_with = ROOT }
			has_opinion = {
				who = ROOT
				value = 25
			}
		}
	}

	# Accepting is not the end of it. The articles were agreed at Hadiach and
	# then had to pass a Sejm, which ratified a thinner version - and the war
	# resumed. That is the part a diplomatic action cannot express, so it hands
	# off to events/Hadiach.txt here.
	on_accept = {
		set_country_flag = rip_hadiach_offered
		add_adm_power = -100
		add_dip_power = -100
		create_subject = {
			subject_type = princedom
			subject = FROM
		}
		FROM = {
			save_event_target_as = hadiach_ruthenia
			add_prestige = 10
		}
		country_event = { id = hadiach.1 days = 20 }
	}

	on_decline = {
		add_prestige = -5
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = lit_union_grievance
			}
		}
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 5
			num_of_rebel_armies = 1
		}
		modifier = {
			factor = 3
			is_in_war = {
				attacker_leader = ROOT
			}
		}
	}

	ai_acceptance = {
		factor = 20
		modifier = {
			factor = 40
			has_opinion = {
				who = FROM
				value = 100
			}
		}
		modifier = {
			factor = -50
			army_size_percentage = 1
		}
	}
}

# --- The charter of a princedom.
#
# The overlord side, and the WFAS shape: no acceptance, the overlord simply
# writes down what the princedom may keep. It costs absolutism and buys quiet.
rip_grant_the_princely_charter = {
	category = influence

	require_acceptance = no

	is_visible = {
		overlord_of = FROM
		FROM = {
			is_subject_of_type = princedom
		}
	}

	is_allowed = {
		is_at_war = no
		adm_power = 50
		FROM = {
			is_at_war = no
			NOT = { has_country_modifier = rip_princely_charter }
		}
	}

	on_accept = {
		add_adm_power = -50
		add_absolutism = -5
		FROM = {
			add_country_modifier = {
				name = rip_princely_charter
				duration = -1
			}
			add_liberty_desire = -25
			add_legitimacy = 5
		}
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 4
			FROM = {
				liberty_desire = 50
			}
		}
	}
}

# --- The udil.
#
# The Kyivan Rus reform rules through kinsmen holding land, not through
# governors holding office, and this is the instrument of it: any subject can
# be made an udil - a princedom, which in this mod's subject types is a vassal
# that can never be annexed or integrated, ranked no higher than a duchy, and
# paying half.
#
# It is deliberately one-sided and needs no acceptance. Handing a kinsman a
# province was not a negotiation; it was how the realm was held together, and
# it is also how it came apart.
rip_grant_the_udil = {
	category = influence

	require_acceptance = no

	is_visible = {
		overlord_of = FROM
		OR = {
			has_reform = kyivan_rus_reform
			has_reform = kyivan_cesarstvo_reform
			has_reform = kyivan_shogunate_reform
			has_reform = ruthenian_principality_reform
			has_reform = principality_appanage
		}
		FROM = {
			NOT = { is_subject_of_type = princedom }
		}
	}

	is_allowed = {
		is_at_war = no
		adm_power = 100
		FROM = {
			is_at_war = no
		}
	}

	on_accept = {
		add_adm_power = -100
		add_absolutism = -5
		create_subject = {
			subject_type = princedom
			subject = FROM
		}
		FROM = {
			add_liberty_desire = -20
			add_legitimacy = 5
			add_country_modifier = {
				name = rip_held_as_an_udil
				duration = -1
			}
		}
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = {
				estate = estate_nobles
				loyalty = 10
			}
		}
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 4
			FROM = {
				liberty_desire = 50
			}
		}
		modifier = {
			factor = 0
			num_of_subjects = 8
		}
	}
}
