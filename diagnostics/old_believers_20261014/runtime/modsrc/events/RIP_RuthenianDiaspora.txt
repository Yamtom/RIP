namespace = rip_diaspora

# Fired by decisions/RIP_RuthenianDiaspora.txt's rip_diaspora_colonial_settlers.
# The decision only sets a country flag naming which colonial region to act
# on; the actual province change happens here, not silently in the decision.
country_event = {
	id = rip_diaspora.10
	title = rip_diaspora.10.t
	desc = rip_diaspora.10.d
	picture = COLONIZATION_eventPicture

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_country_flag = rip_diaspora_target_colonial_canada }
			clr_country_flag = rip_diaspora_target_colonial_canada
			rip_diaspora_convert_colonial_culture_effect = { REGION = colonial_canada THRESHOLD = 100 }
		}
		else_if = {
			limit = { has_country_flag = rip_diaspora_target_colonial_eastern_america }
			clr_country_flag = rip_diaspora_target_colonial_eastern_america
			rip_diaspora_convert_colonial_culture_effect = { REGION = colonial_eastern_america THRESHOLD = 100 }
		}
		else_if = {
			limit = { has_country_flag = rip_diaspora_target_colonial_brazil }
			clr_country_flag = rip_diaspora_target_colonial_brazil
			rip_diaspora_convert_colonial_culture_effect = { REGION = colonial_brazil THRESHOLD = 100 }
		}
		else_if = {
			limit = { has_country_flag = rip_diaspora_target_colonial_la_plata }
			clr_country_flag = rip_diaspora_target_colonial_la_plata
			rip_diaspora_convert_colonial_culture_effect = { REGION = colonial_la_plata THRESHOLD = 100 }
		}
		else_if = {
			limit = { has_country_flag = rip_diaspora_target_colonial_australia }
			clr_country_flag = rip_diaspora_target_colonial_australia
			rip_diaspora_convert_colonial_culture_effect = { REGION = colonial_australia THRESHOLD = 100 }
		}
	}

	option = {
		name = rip_diaspora.10.a
		add_prestige = 1
	}
}

# Fired once by the rip_green_wedge mission itself: we already hold a small,
# uniformly Ruthenian or Rusyn pocket of Manchuria, so it is released as our
# own subject rather than merely noted with a modifier.
country_event = {
	id = rip_diaspora.20
	title = rip_diaspora.20.t
	desc = rip_diaspora.20.d
	picture = COLONIZATION_eventPicture

	is_triggered_only = yes

	trigger = {
		NOT = { exists = ZLK }
	}

	immediate = {
		every_owned_province = {
			limit = {
				region = manchuria_region
				rip_diaspora_eligible_culture_province = yes
			}
			add_core = ZLK
		}
	}

	option = {
		name = rip_diaspora.20.a
		release = ZLK
		vassalize = ZLK
		ZLK = {
			change_primary_culture = ROOT
			change_religion = ROOT
		}
		add_prestige = 10
	}
}
