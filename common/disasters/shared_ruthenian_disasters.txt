# Disasters that several countries can enter
# chernihiv_ruina, khmelnytsky_uprising, governmental_disruption.
# Ordered as vanilla orders a disaster: potential, can_start, can_stop,
# progress, can_end, modifier, on_start, on_end, on_monthly.
# CHR HET KIE PRL VOL ZAZ. The Ruin, with Crimea, Muscovy, Poland, Russia and the Porte reacting.
chernihiv_ruina = {
	ended_by_country_breaking_to_rebels = no
	potential = {
		OR = {
			tag = CHR
			tag = HET
			tag = KIE
			tag = PRL
			tag = VLN
			tag = VOL
			tag = HLC
			tag = ZAZ
		}
		normal_or_historical_nations = yes
		NOT = { has_country_flag = survived_the_ruina }
	}
	can_start = {
		has_any_disaster = no
		OR = {
			current_age = age_of_reformation
			current_age = age_of_absolutism
		}
		is_year = 1657
		NOT = { is_year = 1700 }
		NOT = { stability = 1 }
		OR = {
			war_exhaustion = 4
			num_of_loans = 4
			any_owned_province = {
				is_state_core = ROOT
				devastation = 15
			}
			AND = {
				is_subject = yes
				NOT = { liberty_desire = 50 }
			}
		}
		OR = {
			any_neighbor_country = {
				OR = {
					tag = POL
					tag = MOS
					tag = RUS
					tag = CRI
					tag = TUR
				}
				is_rival = ROOT
			}
			is_in_war = {
				defenders = ROOT
			}
		}
	}
	can_stop = {
		has_any_disaster = yes
	}
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			is_at_war = no
			stability = 1
			NOT = { num_of_rebel_armies = 1 }
		}
	}
	progress = {
		modifier = {
			factor = 1.5
			always = yes
		}
		modifier = {
			factor = 2
			war_exhaustion = 6
		}
		modifier = {
			factor = 1.5
			num_of_loans = 7
		}
		modifier = {
			factor = 1.5
			any_owned_province = {
				is_state_core = ROOT
				devastation = 30
			}
		}
		modifier = {
			factor = 1.5
			NOT = { manpower_percentage = 0.3 }
		}
		modifier = {
			factor = 1.5
			is_subject = yes
			liberty_desire = 50
		}
		modifier = {
			factor = 1
			num_of_rebel_armies = 1
		}
		modifier = {
			factor = -0.5
			stability = 1
		}
		modifier = {
			factor = -0.75
			stability = 2
		}
		modifier = {
			factor = -1
			stability = 3
		}
		modifier = {
			factor = -0.25
			has_country_modifier = chr_baturyn_fortifications
		}
		modifier = {
			factor = -0.5
			has_country_modifier = chr_starshyna_compromise
		}
		modifier = {
			factor = -0.5
			alliance_with = POL
		}
		modifier = {
			factor = -0.5
			alliance_with = MOS
		}
		modifier = {
			factor = -0.5
			alliance_with = RUS
		}
	}
	can_end = {
		stability = 1
		NOT = { war_exhaustion = 2 }
		NOT = { num_of_loans = 3 }
		NOT = {
			any_owned_province = {
				is_state_core = ROOT
				devastation = 10
			}
		}
		NOT = { num_of_rebel_armies = 1 }
	}
	modifier = {
		global_unrest = 3
		manpower_recovery_speed = -0.25
		global_tax_modifier = -0.15
		stability_cost_modifier = 0.35
		legitimacy = -1
		republican_tradition = -0.5
	}
	on_start = ruina_events.1
	on_end = ruina_events.2
	on_monthly = {
		random_events = {
			100 = ruina_events.3
			100 = ruina_events.4
			50 = ruina_events.5
		}
	}
}

# CHR HET KIE PRL VOL ZAZ, and POL LIT PLC while they hold cores in ruthenia_region.
khmelnytsky_uprising = {
	ended_by_country_breaking_to_rebels = no
	potential = {
		OR = {
			AND = {
				OR = {
					tag = POL
					tag = LIT
					tag = PLC
				}
				any_owned_province = {
					region = ruthenia_region
					is_core = ROOT
				}
			}
			OR = {
				tag = CHR
				tag = HET
				tag = KIE
				tag = PRL
				tag = VLN
				tag = VOL
				tag = HLC
				tag = ZAZ
			}
		}
		normal_or_historical_nations = yes
		NOT = { has_country_flag = survived_khmelnytsky_uprising }
	}
	can_start = {
		has_any_disaster = no
		OR = {
			current_age = age_of_reformation
			current_age = age_of_absolutism
		}
		is_year = 1648
		NOT = { is_year = 1660 }
		OR = {
			AND = {
				OR = {
					tag = POL
					tag = LIT
					tag = PLC
				}
				any_owned_province = {
					region = ruthenia_region
					is_core = ROOT
					OR = {
						unrest = 3
						development = 20
					}
				}
				NOT = { stability = 1 }
			}
			AND = {
				OR = {
					tag = CHR
					tag = HET
					tag = KIE
					tag = PRL
					tag = VLN
					tag = VOL
					tag = HLC
					tag = ZAZ
				}
				OR = {
					is_year = 1648
					has_country_modifier = khmelnytsky_revolt_active
				}
			}
		}
	}
	can_stop = {
		NOT = { has_any_disaster = yes }
	}
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			NOT = { num_of_rebel_armies = 1 }
			stability = 1
		}
	}
	progress = {
		modifier = {
			factor = 2
			always = yes
		}
		modifier = {
			factor = 1.5
			any_owned_province = {
				region = ruthenia_region
				unrest = 5
			}
		}
		modifier = {
			factor = 1.5
			num_of_rebel_armies = 2
		}
		modifier = {
			factor = 1.5
			war_exhaustion = 5
		}
		modifier = {
			factor = 1.5
			any_owned_province = {
				region = ruthenia_region
				devastation = 20
			}
		}
		modifier = {
			factor = 1
			NOT = { manpower_percentage = 0.4 }
		}
		modifier = {
			factor = -0.5
			stability = 1
		}
		modifier = {
			factor = -1
			stability = 2
		}
		modifier = {
			factor = -1.5
			stability = 3
		}
		modifier = {
			factor = -0.25
			OR = {
				tag = POL
				tag = LIT
				tag = PLC
			}
			army_size = 30
		}
		modifier = {
			factor = -0.5
			OR = {
				tag = CHR
				tag = HET
				tag = KIE
				tag = PRL
				tag = VLN
				tag = VOL
				tag = HLC
				tag = ZAZ
			}
		}
	}
	can_end = {
		stability = 1
		NOT = { war_exhaustion = 2 }
		OR = {
			AND = {
				OR = {
					tag = POL
					tag = LIT
					tag = PLC
				}
				NOT = {
					any_owned_province = {
						region = ruthenia_region
						unrest = 2
					}
				}
				NOT = { num_of_rebel_armies = 1 }
			}
			AND = {
				OR = {
					tag = CHR
					tag = HET
					tag = KIE
					tag = PRL
					tag = VLN
					tag = VOL
					tag = HLC
					tag = ZAZ
				}
				is_subject = no
			}
		}
	}
	modifier = {
		global_unrest = 2
		stability_cost_modifier = 0.30
		global_tax_modifier = -0.20
		manpower_recovery_speed = -0.30
		legitimacy = -1
	}
	on_start = khmelnytsky_events.1
	on_end = khmelnytsky_events.2
	on_monthly = {
		random_events = {
			100 = khmelnytsky_events.3
			75 = khmelnytsky_events.4
			50 = khmelnytsky_events.5
			25 = khmelnytsky_events.6
		}
	}
}

# KIE and KRU both.
governmental_disruption = {
	potential = {
		OR = {
			tag = KRU
			tag = KIE
		}
		normal_or_historical_nations = yes
	}
	can_start = {
		has_any_disaster = no
		has_reform = representation_monarchy_reform
		OR = {
			current_age = age_of_absolutism
			current_age = age_of_revolutions
		}
	}
	can_stop = {
		OR = {
			has_any_disaster = yes
			has_regency = no
			legitimacy_equivalent = 75
		}
		OR = {
			has_reform = ruthenian_principality_reform
			has_reform = kyivan_rus_reform
		}
		is_subject = yes
	}
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			stability = 2
			NOT = { num_of_rebel_controlled_provinces = 1 }
		}
	}
	progress = {
		modifier = {
			factor = 1
			always = yes
		}
		modifier = {
			factor = 1.5
			has_country_flag = su_path_frontier_centralization
		}
		modifier = {
			factor = -0.25
			has_country_flag = su_path_frontier_liberties
		}
		modifier = {
			factor = -0.25
			has_country_modifier = su_frontier_statutes
		}
		modifier = {
			factor = -0.25
			has_country_modifier = ruina_recovery_south
		}
	}
	can_end = {
		AND = {
			stability = 2
			NOT = { num_of_rebel_controlled_provinces = 1 }
		}
	}
	modifier = {
		stability_cost_modifier = 0.50
		war_exhaustion = 0.3
	}
	on_start = euro_monarchy.1
	on_end = euro_monarchy.3
	on_monthly = {
		events = {
		}
		random_events = {
			1000 = 0
			100 = euro_monarchy.2
		}
	}
}
