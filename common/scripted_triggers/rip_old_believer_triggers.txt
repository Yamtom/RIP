# Old Believer communities and the policy toward them.
#
# A community is a province flag (rip_ob_community), not a religion: the
# province keeps its faith, and only the rip_ob.2 event or a rebel victory ever
# turns a majority into old_believer. There is deliberately no trigger here that
# ends the Raskol - that exit is the patriarch's authority falling below 60% in
# common/disasters/rip_faith_disasters.txt.

# Province scope. Ordinary Orthodox parishes outside the Muscovite
# jurisdiction are never treated as Old Believers just for being Orthodox.
rip_ob_seed_eligible = {
	is_city = yes
	controlled_by = owner
	is_core = owner
	NOT = { has_province_flag = rip_ob_community }
	owner = {
		rip_church_ro_jurisdiction = yes
		has_country_flag = rip_ro_stage_raskol
	}
	OR = {
		religion = russian_orthodox
		AND = {
			religion = orthodox
			region = russia_region
		}
	}
}

# Province scope. The Raskol's conversion wave moves only a Greek/Kyivan Orthodox
# city that its owner holds, in a realm of the Muscovite jurisdiction.
rip_ob_wave_eligible = {
	is_city = yes
	religion = orthodox
	controlled_by = owner
	owner = { rip_church_ro_jurisdiction = yes }
}

# Province scope. The Muscovite culture, in both of the mod's culture sets.
rip_ob_wave_muscovite = {
	OR = {
		culture = russian
		culture = russian_new
	}
}

# Province scope. East Slavic, Uralic (the Ugric group) and Tatar cultures. The
# mod keeps part of the East Slavic cultures in the slavic and carpathian groups,
# so they are listed by name.
rip_ob_wave_kin = {
	OR = {
		culture_group = east_slavic
		culture_group = finno_ugric
		culture_group = tartar
		culture = novgorodian_new
		culture = ryazanian_new
		culture = byelorussian_new
		culture = ruthenian_new
		culture = rusyn_new
		culture = rusyn_new_new
	}
}

# Country scope.
rip_ob_has_communities = {
	any_owned_province = {
		OR = {
			has_province_flag = rip_ob_community
			religion = old_believer
		}
	}
}

# Country scope. The policy is a once-in-ten-years choice that costs 50 ADM.
rip_ob_can_choose_policy = {
	OR = {
		rip_church_ro_jurisdiction = yes
		religion = old_believer
	}
	rip_ob_has_communities = yes
	is_at_war = no
	stability = 0
	adm_power = 50
	OR = {
		NOT = { has_country_flag = rip_ob_policy_changed }
		had_country_flag = { flag = rip_ob_policy_changed days = 3650 }
	}
}
