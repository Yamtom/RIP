# Old Believer communities and the ruler's policy toward them.
#
# Origin: the council of 1666. rip_church_ro_formal_raskol_effect
# (rip_church_ro_history_effects.txt) calls rip_ob_begin_schism_effect, which
# marks up to three communities. The policy is chosen in the options of
# russian_orthodox.6 and rip_faith_disaster.1, or later by the decisions in
# decisions/RIP_OldBelievers.txt.
#
# The Raskol disaster does not create, change or remove a community:
# rip_faith_disaster.2 only sets rip_ro_raskol_settled. Its exit is the patriarch
# losing authority (common/disasters/rip_faith_disasters.txt); the policy only
# changes how quickly that happens.

# Province scope. Province effects never change the majority when creating a
# minority community.
rip_ob_mark_community_effect = {
	if = {
		limit = { NOT = { has_province_flag = rip_ob_community } }
		set_province_flag = rip_ob_community
		add_province_modifier = {
			name = rip_ob_local_community
			duration = -1
		}
	}
}

# Country scope. Picks one eligible province at random.
rip_ob_seed_one_effect = {
	random_owned_province = {
		limit = { rip_ob_seed_eligible = yes }
		rip_ob_mark_community_effect = yes
	}
}

# Country scope. Once per country: up to three communities.
rip_ob_begin_schism_effect = {
	enable_religion = old_believer
	if = {
		limit = { NOT = { has_country_flag = rip_ob_schism_initialized } }
		set_country_flag = rip_ob_schism_initialized
		rip_ob_seed_one_effect = yes
		rip_ob_seed_one_effect = yes
		rip_ob_seed_one_effect = yes
	}
}

# Country scope. Toleration and repression exclude each other.
rip_ob_tolerate_effect = {
	remove_country_modifier = suppression_old_believers
	clr_country_flag = rip_ob_repression
	set_country_flag = rip_ob_toleration
	set_country_flag = rip_ob_policy_changed
	remove_country_modifier = rip_ob_repressive_policy
	add_country_modifier = {
		name = rip_ob_tolerant_policy
		duration = -1
	}
	every_owned_province = { rip_ob_refresh_refuge_effect = yes }
}

rip_ob_repress_effect = {
	remove_country_modifier = old_believer_underground
	clr_country_flag = rip_ob_toleration
	set_country_flag = rip_ob_repression
	set_country_flag = rip_ob_policy_changed
	remove_country_modifier = rip_ob_tolerant_policy
	add_country_modifier = {
		name = rip_ob_repressive_policy
		duration = -1
	}
	every_owned_province = { rip_ob_refresh_refuge_effect = yes }
}

# Country scope. Called at start, every year, on a change of religion and by the
# policy decisions. A conquered minority and an Old Believer majority both
# survive the end of the Raskol; the local consequences are recomputed for the
# CURRENT owner.
rip_ob_upkeep_effect = {
	if = {
		limit = {
			has_country_flag = rip_ro_stage_raskol
			is_year = 1666
			rip_church_ro_jurisdiction = yes
		}
		rip_ob_begin_schism_effect = yes
	}
	# A policy belongs to a ruler of the Muscovite jurisdiction or of the old
	# rite. A country that has left both (a conversion, a lost war) keeps neither
	# the policy nor its modifiers: they last for ever and nothing else removes them.
	if = {
		limit = {
			NOT = { rip_church_ro_jurisdiction = yes }
			NOT = { religion = old_believer }
		}
		clr_country_flag = rip_ob_toleration
		clr_country_flag = rip_ob_repression
		remove_country_modifier = rip_ob_tolerant_policy
		remove_country_modifier = rip_ob_repressive_policy
	}
	every_owned_province = {
		if = {
			limit = {
				religion = old_believer
				NOT = { has_province_flag = rip_ob_community }
			}
			rip_ob_mark_community_effect = yes
		}
		remove_province_modifier = rip_ob_local_toleration
		remove_province_modifier = rip_ob_local_repression
		if = {
			limit = {
				has_province_flag = rip_ob_community
				owner = { has_country_flag = rip_ob_toleration }
			}
			add_province_modifier = {
				name = rip_ob_local_toleration
				duration = -1
			}
		}
		if = {
			limit = {
				has_province_flag = rip_ob_community
				owner = { has_country_flag = rip_ob_repression }
			}
			add_province_modifier = {
				name = rip_ob_local_repression
				duration = -1
			}
		}
	}
}
