# Fired once by the rip_green_wedge mission itself: we already hold a small
# pocket of Manchuria in which at least two provinces have Ruthenian or Rusyn
# communities. The sovereign decides whether to grant them autonomy as a
# vassal, a frontier march, or keep them under direct crown administration.

country_event = {
	id = rip_diaspora.20
	title = rip_diaspora.20.t
	desc = rip_diaspora.20.d
	picture = COLONIZATION_eventPicture
	is_triggered_only = yes
	trigger = {
		NOT = {
			exists = ZLK
		}
	}
	option = {
		name = rip_diaspora.20.a
		ai_chance = {
			factor = 40
		}
		every_owned_province = {
			limit = {
				region = manchuria_region
				rip_diaspora_eligible_culture_province = yes
			}
			add_core = ZLK
		}
		release = ZLK
		vassalize = ZLK
		ZLK = {
			change_primary_culture = ROOT
			change_religion = ROOT
		}
		add_prestige = 10
	}
	option = {
		name = rip_diaspora.20.b
		ai_chance = {
			factor = 30
		}
		every_owned_province = {
			limit = {
				region = manchuria_region
				rip_diaspora_eligible_culture_province = yes
			}
			add_core = ZLK
		}
		release = ZLK
		create_march = ZLK
		ZLK = {
			change_primary_culture = ROOT
			change_religion = ROOT
		}
		add_army_tradition = 5
		add_prestige = 5
	}
	option = {
		name = rip_diaspora.20.c
		ai_chance = {
			factor = 30
		}
		every_owned_province = {
			limit = {
				region = manchuria_region
				rip_diaspora_eligible_culture_province = yes
			}
			add_province_modifier = {
				name = rip_diaspora_crown_frontier
				duration = 7300
			}
		}
		add_adm_power = 25
	}
}
