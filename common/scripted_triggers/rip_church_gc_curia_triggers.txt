# Curia access is Roman diplomacy; it never grants a Greek Catholic cardinal.
rip_church_gc_rome_present = {
	exists = PAP
	PAP = { religion = catholic }
}
rip_church_gc_curia_contact = {
	religion = greek_catholic
	rip_church_gc_rome_present = yes
	NOT = { war_with = PAP }
}
rip_church_gc_has_curia_petition = {
	OR = {
		has_country_modifier = rip_church_gc_petition_church_tax
		has_country_modifier = rip_church_gc_petition_blessing
		has_country_modifier = rip_church_gc_petition_indulgence
		has_country_modifier = rip_church_gc_petition_saint
		has_country_modifier = rip_church_gc_petition_usury
		has_country_modifier = rip_church_gc_petition_holy_war
		has_country_modifier = rip_church_gc_petition_legate
		has_country_modifier = rip_church_gc_petition_monopoly
	}
}
rip_church_gc_has_local_synod = {
	OR = {
		has_country_modifier = rip_church_gc_infrastructure
		has_country_modifier = rip_church_gc_coexistence
	}
}
