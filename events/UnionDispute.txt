# ========================================================================
# The quarrel between Vilnius and Krakow, from the interregnum to Lublin.
# Authenticity: A - documented, with the outcome left open.
# Sources:
#   the union lapses on 10 November 1444 with Wladyslaw at Varna, and the
#   mod's bookmark is the next day; Kazimierz is Grand Duke in Vilnius
#   before he is King in Krakow, and is crowned there only in 1447;
#   the long quarrel over the Volhynian and Podillian castles;
#   separate treasuries, separate armies and separate offices under the
#   union; the act of Lublin in 1569, which moved Podlasie, Volhynia,
#   Bratslav and Kyiv from the Grand Duchy to the Crown at a diet, with the
#   Lithuanian delegation absent from the room.
# ========================================================================

namespace = union_dispute

# --- 1. The interregnum. Krakow has no king; Vilnius has a grand duke.
country_event = {
	id = union_dispute.1
	title = union_dispute.1.t
	desc = union_dispute.1.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		tag = LIT
		lit_at_the_union_question = yes
		NOT = { is_year = 1462 }
		exists = POL
	}

	mean_time_to_happen = {
		months = 36
	}

	option = {
		name = union_dispute.1.a
		ai_chance = { factor = 55 }
		lit_open_union_path_effect = yes
		add_dip_power = 50
		random_country = {
			limit = { tag = POL }
			add_opinion = {
				who = ROOT
				modifier = lit_lords_courted
			}
			country_event = { id = union_dispute.2 days = 10 }
		}
	}

	option = {
		name = union_dispute.1.b
		ai_chance = { factor = 35 }
		lit_open_separate_path_effect = yes
		add_legitimacy = 10
		random_country = {
			limit = { tag = POL }
			add_opinion = {
				who = ROOT
				modifier = lit_union_grievance
			}
			country_event = { id = union_dispute.2 days = 10 }
		}
	}

	option = {
		name = union_dispute.1.c
		ai_chance = { factor = 10 }
		add_dip_power = 25
		add_adm_power = 25
	}
}

# --- 3. The castles of Volhynia and Podillia, claimed by both chanceries for a
#        century and a half.
country_event = {
	id = union_dispute.3
	title = union_dispute.3.t
	desc = union_dispute.3.d
	picture = FORT_eventPicture

	fire_only_once = yes

	trigger = {
		tag = LIT
		is_year = 1450
		NOT = { is_year = 1540 }
		lit_holds_any_ruthenian_palatinate = yes
		exists = POL
	}

	mean_time_to_happen = {
		months = 96
	}

	option = {
		name = union_dispute.3.a
		ai_chance = { factor = 45 }
		add_mil_power = -50
		every_owned_province = {
			limit = {
				OR = {
					province_id = 279	# Volhynia
					province_id = 1943	# Bratslav
					province_id = 280	# Kiev
				}
			}
			add_province_modifier = {
				name = lit_garrisoned_castle
				duration = -1
			}
		}
	}

	option = {
		name = union_dispute.3.b
		ai_chance = { factor = 30 }
		add_dip_power = -50
		random_country = {
			limit = { tag = POL }
			add_opinion = {
				who = ROOT
				modifier = lit_lords_courted
			}
		}
	}

	option = {
		name = union_dispute.3.c
		ai_chance = { factor = 25 }
		add_country_modifier = {
			name = lit_ruthenian_lords_content
			duration = 9125
		}
		add_prestige = -5
	}
}
