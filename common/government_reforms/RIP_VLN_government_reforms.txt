vln_voivodeship_reform = {
	icon = "assembly_hall"
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	potential = {
		OR = {
			tag = VLN
			has_reform = vln_voivodeship_reform
		}
	}
	trigger = {
		OR = {
			has_reform = vln_voivodeship_reform
			AND = {
				tag = VLN
			}
		}
	}
	lock_level_when_selected = yes
	queen = yes
	heir = yes
	royal_marriage = yes
	allow_convert = no
	government_abilities = {
		cossacks_mechanic
	}
	modifiers = {
		governing_capacity = 10
		reform_progress_growth = 0.10
		diplomatic_reputation = 0.5
		monthly_reform_progress = 0.10
	}
	effect = {
		add_country_modifier = {
			name = vln_voivodeship_assembly
			duration = -1
		}
	}
	removed_effect = {
		remove_country_modifier = vln_voivodeship_assembly
	}
	custom_attributes = {
		locked_government_type = yes
	}
}
