# Decisions of the faith-and-border reforms (October 2026).
# Rates are design values, not campaign-validated balance.

country_decisions = {

	# CHR - the grant of slobody to the Old Believers. Each grant spends one
	# refugee (admitted by rip_vetka.1) and plants a settlement of freedoms in
	# woodland: more development and trade, and one more voice for the
	# metropolitan to complain of (rip_vetka.2). Once in five years.
	chr_grant_sloboda_to_old_believers = {
		potential = {
			has_reform = chr_vetka_asylum_reform
		}

		provinces_to_highlight = {
			owned_by = ROOT
			chr_sloboda_site = yes
		}

		allow = {
			adm_power = 25
			stability = 0
			custom_trigger_tooltip = {
				tooltip = chr_sloboda_refugees_tt
				check_variable = { which = chr_ob_refugees value = 1 }
			}
			any_owned_province = { chr_sloboda_site = yes }
			OR = {
				NOT = { has_country_flag = chr_sloboda_recent }
				had_country_flag = { flag = chr_sloboda_recent days = 1825 }
			}
		}

		effect = {
			add_adm_power = -25
			change_variable = { which = chr_ob_refugees value = -1 }
			change_variable = { which = chr_ob_slobody value = 1 }
			set_country_flag = chr_sloboda_recent
			random_owned_province = {
				limit = { chr_sloboda_site = yes }
				add_permanent_province_modifier = { name = chr_old_believer_sloboda duration = -1 }
				add_base_production = 1
				add_base_tax = 1
			}
		}

		ai_will_do = {
			factor = 1
		}
	}

	# PDL - the charter of the four communities of Kamianets. The walls are
	# manned by Ruthenians, Poles, Armenians and the kahal, and nobody is
	# converted by force. Once, and it stays while the reform does.
	pdl_charter_of_four_communities = {
		potential = {
			has_reform = pdl_kamianets_concordat_reform
			NOT = { has_country_flag = pdl_four_communities_chartered }
		}

		provinces_to_highlight = {
			province_id = 281
		}

		allow = {
			owns = 281
			is_at_war = no
			stability = 1
			adm_power = 50
			dip_power = 50
		}

		effect = {
			add_adm_power = -50
			add_dip_power = -50
			set_country_flag = pdl_four_communities_chartered
			281 = {
				add_permanent_province_modifier = { name = pdl_four_communities duration = -1 }
			}
			add_country_modifier = { name = pdl_armenian_caravans duration = 7300 }
		}

		ai_will_do = {
			factor = 1
		}
	}
}
