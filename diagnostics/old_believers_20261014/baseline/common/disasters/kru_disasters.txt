# KRU - the Kyivan Confederation
# ruthenian_court_crisis.
# Ordered as vanilla orders a disaster: potential, can_start, can_stop,
# progress, can_end, modifier, on_start, on_end, on_monthly.
# KRU. The factional empire's court decays past the point the reform can carry it.
ruthenian_court_crisis = {
	potential = {
		tag = KRU
		normal_or_historical_nations = yes
	}
	can_start = {
		has_any_disaster = no
		OR = {
			has_reform = ruthenian_factional_empire_reform
			has_reform = ruthenian_factional_empire_princes_hetmans_reform
			has_reform = ruthenian_factional_empire_boyars_hetmans_reform
		}
		check_variable = { which = ruthenian_decadence value = 90 }
	}
	can_stop = {
		has_any_disaster = yes
	}
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			NOT = { check_variable = { which = ruthenian_decadence value = 75 } }
		}
	}
	progress = {
		modifier = {
			factor = 6
			check_variable = { which = ruthenian_decadence value = 90 }
		}
		modifier = {
			factor = 1.5
			has_country_flag = su_path_frontier_centralization
		}
		modifier = {
			factor = -1
			has_country_flag = su_path_frontier_liberties
		}
		modifier = {
			factor = -1
			has_country_modifier = ruina_recovery_south
		}
	}
	can_end = {
		stability = 1
		NOT = { check_variable = { which = ruthenian_decadence value = 60 } }
	}
	modifier = {
		global_unrest = 3
		stability_cost_modifier = 0.2
		advisor_cost = 0.2
		discipline = -0.03
	}
	on_start = ruthenian_decadence.10
	on_end = ruthenian_decadence.11
	on_monthly = {
		random_events = {
			100 = ruthenian_decadence.12
			80 = ruthenian_decadence.13
			60 = ruthenian_decadence.14
		}
	}
}
