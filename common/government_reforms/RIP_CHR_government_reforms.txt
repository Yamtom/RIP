chr_grain_directorate_reform = {
	icon = "paper_money_map"
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	rulers_can_be_generals = yes
	potential = {
		OR = {
			tag = CHR
			has_reform = chr_grain_directorate_reform
		}
	}
	trigger = {
		OR = {
			has_reform = chr_grain_directorate_reform
			AND = {
				tag = CHR
				custom_trigger_tooltip = {
					tooltip = chr_agro_hypercorp_complete_req_tt
					chr_agro_hypercorp_completed = yes
				}
			}
		}
	}
	modifiers = {
		global_unrest = -1
		global_trade_goods_size_modifier = 0.05
		production_efficiency = 0.05
		burghers_loyalty_modifier = 0.05
	}
	custom_attributes = {
		locked_government_type = yes
	}
	effect = {
		custom_tooltip = chr_grain_directorate_reform_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					trade_goods = grain
				}
				add_province_modifier = {
					name = chr_grain_directorate_province
					duration = -1
					hidden = yes
				}
			}
		}
	}
	removed_effect = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					has_province_modifier = chr_grain_directorate_province
				}
				remove_province_modifier = chr_grain_directorate_province
			}
		}
	}
	ai = {
		factor = 1
	}
}
