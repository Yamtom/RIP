# RIP - Galician (HLC) estate privileges, Austrian administrative path.
#
# Real prototype: estate_nobles_strong_duchies
# (common/estate_privileges/02_noble_privileges.txt) - loyalty = 0.05, a
# +5% loyalty equilibrium, gated on holding subjects of vassal/march/personal-
# union/incorporated type. HLC's Austria fork never attaches
# estate_magnates (see the note in rip_hlc_boyar_council.txt: that estate
# only reaches HLC through the Poland-path hlc_magnate_assembly_reform), so
# this is written for the always-present estate_nobles instead.
#
# Unlike the vanilla prototype, which accepts any vassal, is_valid and
# can_select ask for an INCORPORATED vassal - the appanage the Austria path's
# own diplomatic action creates
# (common/new_diplomatic_actions/RIP_HLC_diplomatic_actions.txt). That is what
# ties the chain together: the Austrian bureaucracy opens the action, the
# action makes the incorporated vassal, and the nobles' privilege is the
# reward for holding one. When the last incorporated vassal is annexed or
# lost, is_valid fails and the privilege lapses with it.

estate_nobles_hlc_strong_appanages = {
	icon = privilege_royal_authority
	max_absolutism = -10
	loyalty = 0.05
	influence = 0.1

	is_valid = {
		has_estate = estate_nobles
		west_ukraine_hlc_origin_country = yes
		hlc_has_austria_path = yes
		any_subject_country = {
			is_subject_of_type = incorporated_vassal
		}
	}
	can_select = {
		has_estate = estate_nobles
		west_ukraine_hlc_origin_country = yes
		custom_trigger_tooltip = {
			tooltip = estate_nobles_hlc_strong_appanages_trigger_tooltip
			hlc_has_austria_path = yes
			any_subject_country = {
				is_subject_of_type = incorporated_vassal
			}
		}
	}

	penalties = {
		land_maintenance_modifier = 0.05
	}
	benefits = {
		diplomatic_upkeep = 2
		reduced_liberty_desire = 10
	}

	ai_will_do = {
		factor = 10
	}
}
