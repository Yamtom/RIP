# The Confederation of Bar. Started by the Dissident question (events/RIP_shared_commonwealth_RuthenianBranches.txt, plc_bar.1).

plc_confederation_of_bar = {
	potential = {
		tag = PLC
		normal_or_historical_nations = yes
		NOT = { has_country_flag = survived_bar_confederation }
	}
	can_start = {
		has_any_disaster = no
		is_year = 1768
		NOT = { is_year = 1775 }
		OR = {
			has_country_flag = plc_dissidents_settled
			has_country_flag = plc_dissidents_refused
		}
	}
	can_stop = {
		NOT = { has_any_disaster = yes }
	}
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			NOT = { num_of_rebel_armies = 1 }
			stability = 1
		}
	}
	progress = {
		modifier = {
			factor = 2
			always = yes
		}
		modifier = {
			factor = 1.5
			num_of_rebel_armies = 1
		}
		modifier = {
			factor = 1.5
			war_exhaustion = 4
		}
		modifier = {
			factor = -0.5
			stability = 1
		}
		modifier = {
			factor = -1
			stability = 2
		}
	}
	can_end = {
		is_year = 1772
		stability = 1
		NOT = { num_of_rebel_armies = 1 }
		NOT = { war_exhaustion = 3 }
	}
	modifier = {
		global_unrest = 2
		stability_cost_modifier = 0.2
		manpower_recovery_speed = -0.2
		legitimacy = -1
	}
	on_start = plc_bar.2
	on_end = plc_bar.3
}
