# =============================================================================
# RIP_UZH_great_projects
# UZH - the Huszt March / Transcarpathia. One monument: palanok_fortress.
# Ordered as vanilla orders a monument: start, date, time, build_cost,
# can_be_moved, move_days_per_unit_distance, starting_tier, type, build_trigger,
# on_built, on_destroyed, can_use_modifiers_trigger, can_upgrade_trigger,
# keep_trigger, tier_0 .. tier_3.
# =============================================================================

# Palanok, the castle on the volcanic hill above Mukachevo. It guarded the
# mountain passes, sheltered Transcarpathian diets, and under the Rakoczi
# princes became the anchor of anti-Habsburg resistance in the Upper Tisza.
#
# start = 1952 is Maramaros, UZH's fixed_capital, not Mukachevo itself - the
# mod has no separate Mukachevo province, so the castle sits in the capital.
# The area payload therefore reaches transylvania_area (157 158 1952 4127).
#
# date = 1400.01.01 is before every bookmark the mod ships, so the castle is
# placed at start and upgraded from tier 0; build_trigger is therefore reached
# only on a rebuild after destruction. It still has to carry the culture gate,
# or an owner could pay 1800 for a castle whose modifiers it can never use.
# Spelled with province_is_or_accepts_culture, the monument idiom in vanilla
# (130 uses); has_owner_accepted_culture has none.
#
# Ladder and payloads stay inside the envelope measured across all 138 vanilla
# monuments: costs 1000 / 2500 / 5000, times 120 / 240 / 480 months, at most
# six country modifiers a tier, no combat pips, prestige never above 1.

palanok_fortress = {
	start = 1952	# Maramaros
	date = 1400.01.01
	time = {
		months = 36
	}
	build_cost = 1800
	can_be_moved = no
	move_days_per_unit_distance = 1
	starting_tier = 0
	type = monument

	build_trigger = {
		custom_trigger_tooltip = {
			tooltip = palanok_fortress_culture_tt
			OR = {
				culture = ruthenian
				culture = rusyn
				culture = hungarian
				culture = slovak
				culture = romanian
			}
			province_is_or_accepts_culture = yes
		}
	}

	on_built = {
	}

	on_destroyed = {
	}

	can_use_modifiers_trigger = {
		custom_trigger_tooltip = {
			tooltip = palanok_fortress_culture_tt
			OR = {
				culture = ruthenian
				culture = rusyn
				culture = hungarian
				culture = slovak
				culture = romanian
			}
			province_is_or_accepts_culture = yes
		}
	}

	can_upgrade_trigger = {
		custom_trigger_tooltip = {
			tooltip = palanok_fortress_upgrade_tt
			province_is_or_accepts_culture = yes
		}
	}

	keep_trigger = {
	}

	tier_0 = {
		upgrade_time = {
			months = 0
		}
		cost_to_upgrade = {
			factor = 0
		}
		province_modifiers = {
		}
		area_modifier = {
		}
		country_modifiers = {
		}
		on_upgraded = {
		}
	}

	tier_1 = {
		upgrade_time = {
			months = 120
		}
		cost_to_upgrade = {
			factor = 1000
		}
		province_modifiers = {
			local_defensiveness = 0.2
			local_friendly_movement_speed = 0.05
		}
		area_modifier = {
			local_manpower_modifier = 0.05
		}
		country_modifiers = {
			siege_ability = 0.05
			fort_maintenance_modifier = -0.05
		}
		on_upgraded = {
			owner = {
				country_event = { id = uzh_palanok.1 days = 3 }
			}
		}
	}

	tier_2 = {
		upgrade_time = {
			months = 240
		}
		cost_to_upgrade = {
			factor = 2500
		}
		province_modifiers = {
			local_defensiveness = 0.3
			local_hostile_attrition = 1
		}
		area_modifier = {
			local_manpower_modifier = 0.075
			local_production_efficiency = 0.05
		}
		country_modifiers = {
			defensiveness = 0.1
			fort_maintenance_modifier = -0.1
		}
		on_upgraded = {
			owner = {
				country_event = { id = uzh_palanok.2 days = 3 }
			}
		}
	}

	tier_3 = {
		upgrade_time = {
			months = 480
		}
		cost_to_upgrade = {
			factor = 5000
		}
		province_modifiers = {
			local_defensiveness = 0.4
			local_hostile_attrition = 1
			local_friendly_movement_speed = 0.1
		}
		area_modifier = {
			local_manpower_modifier = 0.1
			local_regiment_cost = -0.1
		}
		country_modifiers = {
			defensiveness = 0.15
			siege_ability = 0.1
			prestige = 1
		}
		on_upgraded = {
			owner = {
				country_event = { id = uzh_palanok.3 days = 3 }
			}
		}
	}
}
