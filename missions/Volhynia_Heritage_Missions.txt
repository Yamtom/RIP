# HLC, VOL, VLN | Galician and Volhynian heritage missions.
#
# This file sorts between Volhynia_Conquest_Missions.txt and
# Volhynia_Missions.txt on purpose. The engine places the series of one column
# one after another in load order, each mission on its declared row or just
# below the previous one (whichever is lower). Declared rows here are therefore
# only minimums:
#   * HLC alone has nothing above these series, so they sit at the top of their
#     columns, directly under the trunk (column 4 rows 3-5, column 3 row 5).
#   * A VOL loads the conquest series first, so the same series drop into the
#     gaps the VOL tree leaves: Frontier Crown, Confessional Settlement and
#     Statutes at column 4 rows 5-7 (between VOL_restore_ruthenia and
#     VOL_far_north), the Halych Compact at column 3 row 8, Provision the
#     Frontier and Conquer Crimea at column 5 rows 6-7 (between
#     VOL_invest_in_eastern_rus_mission and VOL_choose_development_path).
# Do not move VOL series that load before this file without rechecking
# tests/check_volhynia_missions.py: a later row shifted here shifts every row
# loaded after it in that column.
#
# Both halves of the frontier are open to every VOL whatever its origin: the
# Galician series reads fortresses, the Volhynian series reads the hinterland.

VOL_missions_4_HLC = {
	slot = 4
	generic = no
	ai = yes
	potential = { west_ukraine_hlc_heritage_missions = yes }
	has_country_shield = yes

	hlc_carpathian_bulwark = {
		icon = mission_conquer_50_development
		required_missions = { }
		position = 3
		provinces_to_highlight = {
			# Fixed Carpathian approaches: Halych, Peremyshl, Drohobych.
			OR = { province_id = 261 province_id = 2424 province_id = 4541 }
			OR = {
				NOT = { owned_by = ROOT }
				NOT = { controlled_by = ROOT }
				NOT = {
					OR = { has_building = fort_15th has_building = fort_16th has_building = fort_17th has_building = fort_18th }
				}
			}
		}
		trigger = {
			red_ruthenia_area = { type = all country_or_non_sovereign_subject_holds = ROOT }
			num_of_owned_provinces_with = {
				value = 3
				OR = { province_id = 261 province_id = 2424 province_id = 4541 }
				controlled_by = ROOT
				OR = { has_building = fort_15th has_building = fort_16th has_building = fort_17th has_building = fort_18th }
			}
		}
		effect = {
			add_country_modifier = { name = rip_galicia_carpathian_shield duration = 3650 }
			if = {
				limit = {
					NOT = { hlc_has_confessional_settlement = yes }
				}
				country_event = { id = rip_galicia.2 }
			}
		}
	}

	hlc_confessional_compact = {
		icon = mission_build_up_to_force_limit
		required_missions = { hlc_carpathian_bulwark }
		position = 4
		trigger = {
			OR = {
				custom_trigger_tooltip = {
					tooltip = hlc_confessional_dualism_confessional_req_tt
					hlc_has_confessional_settlement = yes
				}
				custom_trigger_tooltip = {
					tooltip = hlc_confessional_fallback_req_tt
					AND = {
						religious_unity = 0.75
						stability = 1
						num_of_owned_provinces_with = {
							value = 5
							development = 10
						}
					}
				}
			}
		}
		effect = {
			# Renew whichever settlement the synod chose. The choice is read from
			# the flags rip_galicia.2 sets, not from the modifier alone: the
			# modifier lasts ten years and may already have lapsed.
			if = {
				limit = {
					OR = {
						has_country_flag = rip_galicia_synod_held
						has_country_modifier = rip_galicia_uniate_synod
					}
					NOT = { has_country_flag = rip_galicia_orthodox_held }
					NOT = { has_country_modifier = rip_galicia_orthodox_bastion }
				}
				add_country_modifier = { name = rip_galicia_uniate_synod duration = 3650 }
			}
			if = {
				limit = {
					OR = {
						has_country_flag = rip_galicia_orthodox_held
						has_country_modifier = rip_galicia_orthodox_bastion
					}
					NOT = { has_country_flag = rip_galicia_synod_held }
					NOT = { has_country_modifier = rip_galicia_uniate_synod }
				}
				add_country_modifier = { name = rip_galicia_orthodox_bastion duration = 3650 }
			}
			if = {
				limit = {
					OR = {
						has_country_flag = rip_galicia_synod_held
						has_country_modifier = rip_galicia_sejm_uniate
					}
				}
				add_country_modifier = { name = rip_galicia_sejm_uniate duration = 3650 }
			}
			if = {
				limit = {
					NOT = { hlc_has_constitution_settlement = yes }
				}
				country_event = { id = rip_galicia.3 }
			}
			add_prestige = 10
		}
	}

	hlc_constitutional_settlement = {
		icon = mission_have_two_subjects
		required_missions = { hlc_confessional_compact }
		position = 5
		trigger = {
			OR = {
				custom_trigger_tooltip = {
					tooltip = hlc_constitution_civic_req_tt
					hlc_has_constitution_civic_outcome = yes
					2961 = {
						owned_by = ROOT
						development = 25
						has_trade_building_trigger = yes
					}
				}
				custom_trigger_tooltip = {
					tooltip = hlc_constitution_martial_req_tt
					hlc_has_constitution_martial_outcome = yes
					army_size_percentage = 0.9
					army_tradition = 30
				}
				custom_trigger_tooltip = {
					tooltip = hlc_constitution_fallback_req_tt
					AND = {
						hlc_has_constitution_civic_outcome = no
						hlc_has_constitution_martial_outcome = no
						adm_tech = 14
						government_rank = 2
						stability = 2
					}
				}
			}
		}
		effect = {
			add_prestige = 10
			change_government_reform_progress = 15
			if = {
				limit = {
					OR = {
						hlc_has_constitution_civic_outcome = yes
						has_country_modifier = rip_galicia_constitution
					}
				}
				add_adm_power = 50
				add_dip_power = 25
				add_country_modifier = { name = rip_galicia_constitution duration = 7300 }
			}
			else_if = {
				limit = {
					OR = {
						hlc_has_constitution_martial_outcome = yes
						has_country_modifier = rip_galicia_crown_and_sabre
					}
				}
				add_mil_power = 50
				add_army_tradition = 5
				add_country_modifier = { name = rip_galicia_crown_and_sabre duration = 7300 }
			}
			else = {
				add_adm_power = 25
				add_dip_power = 25
				add_mil_power = 25
			}
		}
	}
}

				OR = {

			}
		}

		effect = {
			add_country_modifier = {
				name = vol_carpathian_bulwark_modifier

		}
