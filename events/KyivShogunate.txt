# The senior line has failed to leave an heir and no longer even holds Kyiv
# itself - one of the senior udily does. Fired from kie_on_actions.txt :
# on_monarch_death through rip_kyiv_seniorate_line_broken_trigger (common/
# scripted_triggers/rip_kyivan_seniorate_triggers.txt). No vanilla rule of this
# exact shape exists to copy (checked events/Japan.txt, flavorJAP.txt,
# decisions/JapaneseNation.txt, and every common/scripted_trigger and
# common/scripted_effect file for "shogun"/"Kyoto"/"mantle" - the shogunate's
# own succession-by-holding-the-capital is flavour, not a scripted rule); this
# reuses the mechanic's own ladder-of-thrones logic instead. All the real work
# happens in immediate, guarded by exists = event_target - the option is an
# acknowledgement only, so a failed lookup never leaves the event without a
# valid option.
country_event = {
	id = kyiv_shogunate.2
	title = kyiv_shogunate.2.t
	desc = kyiv_shogunate.2.d
	picture = COURT_eventPicture

	is_triggered_only = yes

	trigger = {
		has_reform = kyivan_shogunate_reform
	}

	immediate = {
		random_subject_country = {
			limit = {
				is_subject_of_type = senior_udil
				owns = 280
			}
			save_event_target_as = rip_kyiv_new_senior_prince
		}
		if = {
			limit = { exists = event_target:rip_kyiv_new_senior_prince }
			free_vassal = event_target:rip_kyiv_new_senior_prince
			# Every senior udil still standing answers to Kyiv's new holder
			# now, not to a house that could not keep the city.
			every_subject_country = {
				limit = { is_subject_of_type = senior_udil }
				ROOT = {
					free_vassal = PREV
				}
				event_target:rip_kyiv_new_senior_prince = {
					create_subject = {
						subject_type = senior_udil
						subject = PREV
					}
				}
			}
			event_target:rip_kyiv_new_senior_prince = {
				# The lapse this same on_monarch_death just marked him with,
				# on his way out the door as a subject, means nothing now.
				remove_country_modifier = rip_tenure_lapsed
				add_government_reform = kyivan_shogunate_reform
				add_prestige = 10
				add_legitimacy_equivalent = {
					amount = 10
					republican_tradition = 5
				}
				create_subject = {
					subject_type = senior_udil
					subject = ROOT
				}
			}
			# The old house keeps its name and its lands; only the seniority
			# has moved. principality_appanage is the same tenure-and-government
			# reform rip_grant_the_udil hands to any new udil-holder.
			if = {
				limit = {
					government = monarchy
					NOT = { has_reform = principality_appanage }
				}
				add_government_reform = principality_appanage
			}
			add_country_modifier = {
				name = rip_tenure_lapsed
				duration = -1
			}
		}
	}

	option = {
		name = kyiv_shogunate.2.a
	}
}
