# Gates for the Ruthenian Diaspora mission tree (missions/zzzz_RuthenianDiaspora_
# Missions.txt) and the settler decisions it unlocks
# (decisions/RIP_RuthenianDiaspora.txt). The tree opens for any country whose
# primary culture sits in the Ruthenian or Rusyn family - byelorussian is
# deliberately excluded, see tools/data/rip_colonial_names.json
# for the three culture families this mod covers with colonial names.

# Historical commemoration in both layouts. Settlements and the finale have independent gates.
rip_diaspora_can_commemorate_lysianskyi = {
	is_year = 1806
	has_country_flag = circumnavigation_done
}

rip_diaspora_eligible_culture = {
	OR = {
		primary_culture = ruthenian
		primary_culture = ruthenian_new
		primary_culture = ruthenian_west_ruthenian
		primary_culture = ruthenian_west_slavic
		primary_culture = rusyn
		primary_culture = rusyn_new
		primary_culture = rusyn_new_new
		primary_culture = rusyn_west_ruthenian
		primary_culture = rusyn_west_slavic
	}
}

# Province-scope twin of the above: checks the province's own culture, not
# its owner's primary culture.
rip_diaspora_eligible_culture_province = {
	OR = {
		culture = ruthenian
		culture = ruthenian_new
		culture = ruthenian_west_ruthenian
		culture = ruthenian_west_slavic
		culture = rusyn
		culture = rusyn_new
		culture = rusyn_new_new
		culture = rusyn_west_ruthenian
		culture = rusyn_west_slavic
	}
}

# Is there a foreign settler partner already established in a colonial
# region? Two shapes count: a colonial nation has formed there (opinion
# checked with both the nation and its overlord), or a single foreign power
# still owns it directly and no colonial nation has formed yet. $REGION$ is
# a colonial_region key, $THRESHOLD$ the opinion floor required of each
# party that must approve.
rip_diaspora_colonial_partner_trigger = {
	any_country = {
		any_owned_province = { colonial_region = $REGION$ }
		OR = {
			AND = {
				is_colonial_nation = yes
				has_opinion = { who = ROOT value = $THRESHOLD$ }
				overlord = { has_opinion = { who = ROOT value = $THRESHOLD$ } }
			}
			AND = {
				NOT = { is_colonial_nation = yes }
				NOT = { tag = ROOT }
				has_opinion = { who = ROOT value = $THRESHOLD$ }
			}
		}
	}
}

# Province scope. The Green Wedge lay in the Russian Far East: north of the
# Amur, along its lower course and in the Ussuri country down to the Pacific.
# Three vanilla 1.37 areas sit wholly there. central_jilin_area and
# heilongjiang_area also reach across the rivers, so their wedge provinces are
# named one by one: 729 Bohori, 2105 Kuru, 4217 Pijin, 2102 Albazin. 1053
# Hinggan stays out: map/rivers.bmp puts it south of the Amur. Liaodong, Girin,
# Ningguta and Sakhalin are outside too. ZLK's capital (history/countries) is
# 2106 Furdan, inside this set.
rip_green_wedge_province = {
	OR = {
		area = furdan_area
		area = east_heilongjiang_area
		area = ilan_hala_area
		province_id = 729
		province_id = 2105
		province_id = 4217
		province_id = 2102
	}
}

# Province scope. The zone the Grey Wedge charter and its settlers use.
rip_grey_wedge_province = {
	region = central_asia_region
}

# Province scope: a local community under our rule or under a friendly host.
rip_diaspora_established_community = {
	rip_diaspora_eligible_culture_province = yes
	OR = {
		owned_by = ROOT
		owner = { has_opinion = { who = ROOT value = 100 } }
	}
}

# A mainland wedge has no colonial-nation layer above it: a single foreign owner
# on good terms with us is enough. $WEDGE$ is one of the two province triggers
# above, $THRESHOLD$ the opinion floor.
rip_diaspora_mainland_partner_trigger = {
	any_country = {
		NOT = { tag = ROOT }
		any_owned_province = { $WEDGE$ = yes }
		has_opinion = { who = ROOT value = $THRESHOLD$ }
	}
}

# Convertible-province variants used by the settler decisions and choice
# events so regions with only already-Ruthenian (or already-converted-faith)
# provinces neither stay selectable nor keep the decision open when empty.

	any_country = {
		any_owned_province = {
			colonial_region = $REGION$
			NOT = { rip_diaspora_eligible_culture_province = yes }
		}
		OR = {
			AND = {
				is_colonial_nation = yes
				has_opinion = { who = ROOT value = $THRESHOLD$ }
				overlord = { has_opinion = { who = ROOT value = $THRESHOLD$ } }
			}
			AND = {
				NOT = { is_colonial_nation = yes }
				NOT = { tag = ROOT }
				has_opinion = { who = ROOT value = $THRESHOLD$ }
			}
		}
	}
}
