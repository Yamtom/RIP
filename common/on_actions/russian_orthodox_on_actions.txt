
on_yearly_pulse = {
	random_events = {
		25 = russian_orthodox.1  # Third Rome Doctrine
		20 = russian_orthodox.2  # Moscow Patriarchate
		20 = russian_orthodox.8  # Gathering Russian Lands
		600 = 0 # Nothing
	}
}

on_bi_yearly_pulse = {
	# The zeal of the Sobor is the engine's fervour bar now - see
	# `fervor = yes` in common/religions/russian_orthodox.txt. What is left
	# here is the migration: a save started before the change still carries the
	# hand-rolled focus modifiers and the five fake progress-bar modifiers, and
	# nothing else would ever take them off. Both effects are idempotent, so
	# running them on the pulse costs a save that never had them nothing.
	rip_ro_clear_the_bands_effect = yes
	rip_ro_clear_the_zeal_effect = yes

	# The faith travels with the merchants through the real trade node policy
	# Propagate Religion - see common/scripted_effects/rip_faith_spread_effects.txt.
	#
	# The unlock also has to be applied HERE, not only in the religion's
	# on_convert. A country that was already of this faith when the mechanic
	# was added never runs on_convert again, so it would never receive the
	# flags and the policy would stay greyed out for ever. Both effects only
	# set country flags, so re-running them costs nothing.
	if = {
		limit = {
			religion = russian_orthodox
			NOT = { has_country_flag = can_use_propagate_religion }
		}
		rip_faith_unlock_trade_propagation_effect = yes
	}

	# The church keeps consolidating after the schism, at the speed its own
	# unity allows - see rip_ro_consolidation_effect. The budget runs in three
	# steps, 0.75 then 0.50 then 0.25 for ever, so this pulse is fast at first
	# and slow afterwards without ever stopping.
	rip_ro_consolidation_effect = yes

	# THE OIL FOR THE LAMPS.
	# Must run AFTER the consolidation and BEFORE the backlash: the icons cost
	# fervour every pulse, and how many are standing is what the Old Believers
	# are rising about. See common/scripted_effects/rip_ro_icon_effects.txt.
	rip_ro_icon_upkeep_effect = yes

	# WHO OBJECTS.
	# The old rite that would not follow Moscow, and - once the books have been
	# corrected - the Old Believers who would not follow it either. See
	# common/scripted_effects/rip_faith_zeal_effects.txt.
	rip_ro_zealot_backlash_effect = yes

	random_events = {
		15 = russian_orthodox.5  # Orthodox Inquisition
		10 = russian_orthodox.6  # Old Believer Schism
		12 = russian_orthodox.9  # Symphonia
		10 = russian_orthodox.11 # Imperial Orthodox Church
		10 = russian_orthodox.12 # Missionary Success Siberia
		8 = russian_orthodox.13  # Defender of Orthodoxy
		8 = russian_orthodox.14  # Patriarch of All Rus'
		700 = 0 # Nothing
	}
}

# Fires for EVERY country that changes religion, not only this one - EU4 merges
# on_action blocks across files and this hook has no faith of its own. Both
# effects below are written to be harmless where they do not apply.
on_religion_change = {
	# A church that changes obedience starts its own three-step wave rather
	# than inheriting the decay of the one it left.
	rip_faith_reset_stage_effect = yes

	# And it does not keep the previous church's iconostasis standing.
	rip_ro_icon_douse_all_effect = yes
}
