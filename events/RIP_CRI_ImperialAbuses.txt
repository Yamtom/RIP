		name = rip_cri_reckoning.1.a

		country_event = { id = rip_cri_reckoning.2 days = 180 }
	}
	option = {
		name = rip_cri_reckoning.1.b
		ai_chance = { factor = 1 }
		if = {
			limit = { has_estate = estate_nomadic_tribes }
			add_estate_loyalty = { estate = estate_nomadic_tribes loyalty = -10 }
		}
		set_country_flag = rip_cri_kaffa_marts_curbed
		add_dip_power = 25
		add_country_modifier = {
			name = rip_cri_reformed_tamga
			duration = 5475
		}
	}
}

# --- Disaster: rip_cri_yasyr_depopulation_crisis (.3 start, .4 pulse, .5 end)

country_event = {
	id = rip_cri_reckoning.3
	title = rip_cri_reckoning.3.t
	desc = rip_cri_reckoning.3.d
	picture = PLAGUE_eventPicture
	is_triggered_only = yes

	option = {
		name = rip_cri_reckoning.3.a
		ai_chance = { factor = 55 }
		add_horde_unity = -10
		add_treasury = -40
		if = {
			limit = {
				any_owned_province = {
					OR = {
						area = crimea_area
						area = yedisan_area
						area = azov_area
					}
				}
			}
			random_owned_province = {
				limit = {
					OR = {
						area = crimea_area
						area = yedisan_area
						area = azov_area
					}
				}
				add_devastation = 20
				spawn_rebels = {
					type = particularist_rebels
					size = 1
				}
			}
		}
	}
	option = {
		name = rip_cri_reckoning.3.b
		ai_chance = { factor = 45 }
		add_dip_power = -45
		if = {
			limit = { has_estate = estate_nomadic_tribes }
			add_estate_loyalty = { estate = estate_nomadic_tribes loyalty = -15 }
		}
		set_country_flag = rip_cri_kaffa_marts_curbed
	}
}
