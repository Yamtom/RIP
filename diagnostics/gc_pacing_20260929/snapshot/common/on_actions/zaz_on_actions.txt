# ZAZ - the Sich and where it sits.

on_yearly_pulse = {
	if = {
		limit = {
			OR = {
				tag = ZAZ
				is_cossack_polity = yes
			}
			any_owned_province = {
				has_province_modifier = zaporozhian_sich_capital
				NOT = { province_id = capital }
			}
		}
		every_owned_province = {
			limit = {
				has_province_modifier = zaporozhian_sich_capital
				NOT = { province_id = capital }
			}
			remove_province_modifier = zaporozhian_sich_capital
		}
	}
}
