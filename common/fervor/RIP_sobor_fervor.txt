# ==========================================================================
# THE ZEAL OF THE SOBOR
#
# russian_orthodox declares `fervor = yes`, which gives it the Reformed bar and
# the Reformed focus buttons in interface/countryreligionview.gui. The bar is
# the engine's; what it buys is ours.
#
# These three carry the payloads the hand-rolled version used - see the retired
# rip_ro_zeal_war / _trade / _stability in RIP_faith_modifiers.txt - so a save
# that had a focus running keeps the same standing bonus under a new name.
#
# `potential` gates them to this faith. The eight vanilla entries are gated the
# other way in 00_fervor.txt, which this mod overrides for exactly that reason:
# without it a Muscovite Sobor would offer the Dutch Republic's trade focus.
#
# Cost is the vanilla scale (8) so the bar's own fill rate stays meaningful.
# ==========================================================================

sobor_of_war = {
	cost = 8

	potential = {
		FROM = { religion = russian_orthodox }
	}

	trigger = { }

	effect = {
		land_morale = 0.1
		naval_morale = 0.1
		shock_damage = 0.05
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 3
			FROM = { is_at_war = yes }
		}
		modifier = {
			factor = 2
			FROM = { is_in_important_war = yes }
		}
		modifier = {
			factor = 0.4
			FROM = { is_at_war = no }
		}
	}
}

sobor_of_the_market = {
	cost = 8

	potential = {
		FROM = { religion = russian_orthodox }
	}

	trigger = { }

	effect = {
		global_trade_power = 0.1
		trade_efficiency = 0.1
		global_trade_goods_size_modifier = 0.05
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 2
			FROM = { trade_income_percentage = 0.25 }
		}
		modifier = {
			factor = 0.5
			FROM = { is_at_war = yes }
		}
	}
}

sobor_of_the_canons = {
	cost = 8

	potential = {
		FROM = { religion = russian_orthodox }
	}

	trigger = { }

	effect = {
		stability_cost_modifier = -0.15
		global_unrest = -1
		church_loyalty_modifier = 0.1
	}

	ai_will_do = {
		factor = 1
		modifier = {
			factor = 4
			FROM = { NOT = { stability = 0 } }
		}
		modifier = {
			factor = 2
			FROM = { num_of_rebel_armies = 1 }
		}
	}
}
