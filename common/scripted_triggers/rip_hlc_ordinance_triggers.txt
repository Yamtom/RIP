# =============================================================================
# Scripted triggers for the Chancery Ordinances
# (common/government_mechanics/rip_hlc_austrian_reorganization.txt).
#
# Two kinds live here. The slot trigger is the rule the player meets in the
# button tooltips: the crown keeps at most three of its six ordinances in
# force, so a fourth has to wait until one has run its course. The *_wanted
# triggers are only read by the AI's ai_chance blocks in the same mechanic -
# they are the situations in which a ruler would reach for an ordinance.
# Nothing here has an effect.
#
# The country modifier of each ordinance
# (hlc_<name>_ordinance_modifier, ten years) is the single record of "in
# force"; the mechanic's buttons add it and hlc_clear_ordinances_effect
# (common/scripted_effects/rip_hlc_ordinance_effects.txt) removes it. There is
# no flag any more: a flag could outlive its modifier, a modifier cannot.
# =============================================================================

# True while fewer than three ordinances are in force. calc_true_if with plain
# triggers is vanilla's own counting form (num_of_ruler_traits in
# common/scripted_triggers/00_scripted_triggers.txt).
hlc_ordinance_slot_free = {
	NOT = {
		calc_true_if = {
			has_country_modifier = hlc_chancery_ordinance_modifier
			has_country_modifier = hlc_settlement_ordinance_modifier
			has_country_modifier = hlc_cadastral_ordinance_modifier
			has_country_modifier = hlc_urbarial_ordinance_modifier
			has_country_modifier = hlc_toleration_ordinance_modifier
			has_country_modifier = hlc_frontier_ordinance_modifier
			amount = 3
		}
	}
}

# Several owned cities hold a culture that is neither the primary nor an
# accepted one: the settlement ordinance has work to do. has_owner_culture and
# has_owner_accepted_culture are province triggers, used the same way inside
# vanilla's estate agendas (common/estate_agendas/0002_rajput_agendas.txt).
hlc_ordinance_settlement_wanted = {
	num_of_owned_provinces_with = {
		value = 3
		is_city = yes
		has_owner_culture = no
		has_owner_accepted_culture = no
	}
}

# Short of money and with nobles who can bear being counted. Wanted above 40%
# noble loyalty; the survey itself lowers it by five points, so an AI ruler
# does not send the surveyors out and then watch its nobles revolt.
hlc_ordinance_cadastre_wanted = {
	has_estate = estate_nobles
	estate_loyalty = {
		estate = estate_nobles
		loyalty = 40
	}
	OR = {
		num_of_loans = 1
		NOT = { years_of_income = 0.25 }
	}
}

# A restless countryside whose lords can bear a written register of dues: at
# least three owned cities in unrest, and the nobles at half loyalty or more.
hlc_ordinance_urbarial_wanted = {
	has_estate = estate_nobles
	estate_loyalty = {
		estate = estate_nobles
		loyalty = 50
	}
	num_of_owned_provinces_with = {
		value = 3
		is_city = yes
		unrest = 3
	}
}

# Other faiths fill the towns: at least four owned cities that do not share
# the state religion. has_owner_religion is a province trigger, used the same
# way by vanilla's advisor and ancestor definitions.
hlc_ordinance_toleration_wanted = {
	num_of_owned_provinces_with = {
		value = 4
		is_city = yes
		has_owner_religion = no
	}
}

# The marches are in arms: only a frontier at war calls up watchmen.
hlc_ordinance_frontier_wanted = {
	is_at_war = yes
}
