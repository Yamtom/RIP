### Crimea - azov_area
286  = "Azaq"                          # the town at the Don mouth, Genoese Tana in 1444 and an Ottoman fortress from 1475

### Crimea - crimea_area
284  = "Qirim"                         # Solhat, the Horde governor's seat that lent its name to the whole peninsula and to the khanate's own chancery style
285  = "Kefe"                          # the Ottoman form of Caffa, seat of the sancak and later eyalet that ran the slave market and the strait
2410 = "Mankup"                        # Mangup-Kale, the Gothic princes' rock, taken 1475 and kept as an Ottoman kadilik under that name
2447 = "Taman"                         # the fortress and ferry head on the far side of the strait from Kerch, the Genoese Matrega after 1475

### Crimea - lower_don_area
1971 = "Sari Su"                       # the yellow water of the Volga-Don portage, the Turkic hydronym behind the later Russian town name
2409 = "Tuzla"                         # the salt pans of the Donets that Nogai and Cossack parties worked and fought over in the same seasons

### Crimea - yedisan_area
282  = "Ozu"                           # the fortress at the Dnieper liman that also names the river in Ottoman and Tatar usage
1974 = "Gazi Kerman"                   # the fortress raised in the 1490s to command the Tavan crossing of the river
2406 = "Yedisan"                       # the horde of the seven clans pastured between the Bug and the Inhul; the chancery names the pasture, not the ford

### Crimea - zaporizhia_area
283  = "Desti Qipcaq"                  # the standing chancery term for the open steppe above the rapids, land held by grazing and not by garrison
2412 = "Camboyluq"                     # the Jamboyluk horde wintered between the lower Dnieper and the Molochna toward the Azov shore

### Carpathia - moldavia_area
268  = "Bogdan"                        # the voivodate as the Porte and the khans named it, after its founding voivode
1756 = "Aqkerman"                      # the white fortress on the Dniester liman, taken 1484 and the anchor of the Budjak horde
4529 = "Yas"                           # the Ottoman rendering of the voivode's residence in the north of Bogdan

### Carpathia - wallachia_area
161  = "Eflak"                         # the southern voivodate under the same chancery convention as Bogdan
4531 = "Yergogu"                       # the Ottoman fortress on the Danube crossing that Tatar contingents used going south

### Caucasia - circassia_area
463  = "Cerkes"                        # the Kuban slope named for its people in the khanate's tribute registers
2195 = "Natuhay"                       # the coastal district of the Natukhai, the Circassian communities the Kefe market drew on
2197 = "Kabartay"                      # the eastern Circassian princes' country, named so in accounts of the khans' campaigns beyond the Kuban

### Caucasia - imereti_area
