# The rota is recomputed yearly as well as on every grant and revocation, because
# an udil can stop being one without either action being used - it wins its
# independence war, it is conquered, or somebody else annexes it. Gated on the
# country already carrying a tier, so it costs nothing for everyone else.
on_yearly_pulse = {
	effect = {
		if = {
			limit = {
				OR = {
					has_country_modifier = rip_the_rota_1
					has_country_modifier = rip_the_rota_2
					has_country_modifier = rip_the_rota_3
					has_country_modifier = rip_the_rota_4
					has_country_modifier = rip_the_rota_5
				}
			}
			rip_recalculate_the_rota_effect = yes
		}
	}
}
