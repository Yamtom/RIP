# Scripted effects for the West-Ruthenian and West-Slavic cultural maneuver
# Supports Belarusian (byelorussian), Ruthenian (ruthenian), and Rusyn (rusyn)
# independently via the three-circle collective consensus and global province
# adoption indices (Stage 2 >= 60% S1/WR, Stage 4 >= 85% S3/WS).

# ------------------------------------------------------------------------
# Shared helper: replace obsolete accepted_culture slot without wasting a slot
# ------------------------------------------------------------------------

wr_refresh_byelorussian_accepted_slot_wr_effect = {
	if = {
		limit = {
			NOT = { wr_is_byelorussian_country = yes }
			any_owned_province = {
				culture = byelorussian_west_ruthenian
			}
			NOT = { accepted_culture = byelorussian_west_ruthenian }
		}
		add_accepted_culture = byelorussian_west_ruthenian
	}
	if = {
		limit = {
			accepted_culture = byelorussian
			NOT = { any_owned_province = { culture = byelorussian } }
		}
		remove_accepted_culture = byelorussian
	}
	if = {
		limit = {
			accepted_culture = byelorussian_new
			NOT = { any_owned_province = { culture = byelorussian_new } }
		}
		remove_accepted_culture = byelorussian_new
	}
}

# Runs in province scope on owned target provinces
wr_apply_stage1_charter_to_province_effect = {
	if = {
		limit = {
			has_province_modifier = wr_west_slavic_shared_script_space
		}
		remove_province_modifier = wr_west_slavic_shared_script_space
	}
	if = {
		limit = {
			NOT = { has_province_modifier = wr_chancery_prosta_mova_tradition }
		}
		add_province_modifier = {
			name = wr_chancery_prosta_mova_tradition
			duration = -1
		}
	}
}

# Runs in province scope on owned target or West-Slavic provinces
wr_apply_stage3_affinity_to_province_effect = {
	if = {
		limit = {
			wr_is_target_province = yes
			has_province_modifier = wr_chancery_prosta_mova_tradition
		}
		remove_province_modifier = wr_chancery_prosta_mova_tradition
	}
	if = {
		limit = {
			NOT = { has_province_modifier = wr_west_slavic_shared_script_space }
		}
		add_province_modifier = {
			name = wr_west_slavic_shared_script_space
			duration = -1
		}
	}
}

# Runs in province scope on owner / culture change / conversion
wr_sync_single_province_charter_effect = {
	if = {
		limit = {
			wr_is_target_province = yes
			is_core = owner
			owner = {
				wr_can_run_maneuver_base = yes
				OR = {
					has_country_flag = wr_distinctiveness_recognized
					has_country_flag = wr_west_ruthenian_group_formed
				}
				NOT = { has_country_flag = wr_west_slavic_affinity_active }
				NOT = { has_country_flag = wr_west_slavic_shift_completed }
			}
		}
		wr_apply_stage1_charter_to_province_effect = yes
	}
	else_if = {
		limit = {
			wr_is_target_province = yes
			is_core = owner
			owner = {
				wr_can_run_maneuver_base = yes
				OR = {
					has_country_flag = wr_west_slavic_affinity_active
					has_country_flag = wr_west_slavic_shift_completed
				}
				NOT = { has_country_flag = wr_distinctiveness_recognized }
				NOT = { has_country_flag = wr_west_ruthenian_group_formed }
			}
		}
		wr_apply_stage3_affinity_to_province_effect = yes
	}
}

# Runs in country scope on yearly pulse or rally
wr_sync_owned_charter_provinces_effect = {
	if = {
		limit = {
			OR = {
				has_country_flag = wr_distinctiveness_recognized
				has_country_flag = wr_west_ruthenian_group_formed
			}
			NOT = { has_country_flag = wr_west_slavic_affinity_active }
			NOT = { has_country_flag = wr_west_slavic_shift_completed }
		}
		every_owned_province = {
			limit = {
				is_core = owner
				OR = {
					AND = {
						wr_is_byelorussian_province = yes
						owner = { wr_byelorussian_initiator_stake = yes }
					}
					AND = {
						wr_is_ruthenian_province = yes
						owner = { wr_ruthenian_initiator_stake = yes }
					}
					AND = {
						wr_is_rusyn_province = yes
						owner = { wr_rusyn_initiator_stake = yes }
					}
				}
			}
			wr_apply_stage1_charter_to_province_effect = yes
		}
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = wr_west_slavic_affinity_active
				has_country_flag = wr_west_slavic_shift_completed
			}
			NOT = { has_country_flag = wr_distinctiveness_recognized }
			NOT = { has_country_flag = wr_west_ruthenian_group_formed }
		}
		every_owned_province = {
			limit = {
				is_core = owner
				OR = {
					AND = {
						wr_is_byelorussian_province = yes
						owner = { wr_byelorussian_initiator_stake = yes }
					}
					AND = {
						wr_is_ruthenian_province = yes
						owner = { wr_ruthenian_initiator_stake = yes }
					}
					AND = {
						wr_is_rusyn_province = yes
						owner = { wr_rusyn_initiator_stake = yes }
					}
					culture_group = west_slavic
				}
			}
			wr_apply_stage3_affinity_to_province_effect = yes
		}
	}
}

# Computes global province shares (0..100) for all 3 peoples on current country scope.
wr_recalculate_maneuver_indices_effect = {
	save_event_target_as = wr_index_target
	set_variable = { which = wr_byelorussian_total value = 0 }
	set_variable = { which = wr_byelorussian_s1_cnt value = 0 }
	set_variable = { which = wr_byelorussian_s3_cnt value = 0 }
	set_variable = { which = wr_byelorussian_s1_pct value = 0 }
	set_variable = { which = wr_byelorussian_s3_pct value = 0 }
	set_variable = { which = wr_ruthenian_total value = 0 }
	set_variable = { which = wr_ruthenian_s1_cnt value = 0 }
	set_variable = { which = wr_ruthenian_s3_cnt value = 0 }
	set_variable = { which = wr_ruthenian_s1_pct value = 0 }
	set_variable = { which = wr_ruthenian_s3_pct value = 0 }
	set_variable = { which = wr_rusyn_total value = 0 }
	set_variable = { which = wr_rusyn_s1_cnt value = 0 }
	set_variable = { which = wr_rusyn_s3_cnt value = 0 }
	set_variable = { which = wr_rusyn_s1_pct value = 0 }
	set_variable = { which = wr_rusyn_s3_pct value = 0 }
	every_province = {
		limit = {
			is_city = yes
			wr_is_target_province = yes
		}
		if = {
			limit = { wr_is_byelorussian_province = yes }
			event_target:wr_index_target = { change_variable = { which = wr_byelorussian_total value = 1 } }
			if = {
				limit = {
					OR = {
						has_province_modifier = wr_chancery_prosta_mova_tradition
						culture = byelorussian_west_ruthenian
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_byelorussian_s1_cnt value = 1 } }
			}
			else_if = {
				limit = {
					OR = {
						has_province_modifier = wr_west_slavic_shared_script_space
						culture = byelorussian_west_slavic
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_byelorussian_s3_cnt value = 1 } }
			}
		}
		else_if = {
			limit = { wr_is_ruthenian_province = yes }
			event_target:wr_index_target = { change_variable = { which = wr_ruthenian_total value = 1 } }
			if = {
				limit = {
					OR = {
						has_province_modifier = wr_chancery_prosta_mova_tradition
						culture = ruthenian_west_ruthenian
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_ruthenian_s1_cnt value = 1 } }
			}
			else_if = {
				limit = {
					OR = {
						has_province_modifier = wr_west_slavic_shared_script_space
						culture = ruthenian_west_slavic
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_ruthenian_s3_cnt value = 1 } }
			}
		}
		else_if = {
			limit = { wr_is_rusyn_province = yes }
			event_target:wr_index_target = { change_variable = { which = wr_rusyn_total value = 1 } }
			if = {
				limit = {
					OR = {
						has_province_modifier = wr_chancery_prosta_mova_tradition
						culture = rusyn_west_ruthenian
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_rusyn_s1_cnt value = 1 } }
			}
			else_if = {
				limit = {
					OR = {
						has_province_modifier = wr_west_slavic_shared_script_space
						culture = rusyn_west_slavic
					}
				}
				event_target:wr_index_target = { change_variable = { which = wr_rusyn_s3_cnt value = 1 } }
			}
		}
	}
	if = {
		limit = { check_variable = { which = wr_byelorussian_total value = 1 } }
		set_variable = { which = wr_byelorussian_s1_pct which = wr_byelorussian_s1_cnt }
		multiply_variable = { which = wr_byelorussian_s1_pct value = 100 }
		divide_variable = { which = wr_byelorussian_s1_pct which = wr_byelorussian_total }
		set_variable = { which = wr_byelorussian_s3_pct which = wr_byelorussian_s3_cnt }
		multiply_variable = { which = wr_byelorussian_s3_pct value = 100 }
		divide_variable = { which = wr_byelorussian_s3_pct which = wr_byelorussian_total }
	}
	if = {
		limit = { check_variable = { which = wr_ruthenian_total value = 1 } }
		set_variable = { which = wr_ruthenian_s1_pct which = wr_ruthenian_s1_cnt }
		multiply_variable = { which = wr_ruthenian_s1_pct value = 100 }
		divide_variable = { which = wr_ruthenian_s1_pct which = wr_ruthenian_total }
		set_variable = { which = wr_ruthenian_s3_pct which = wr_ruthenian_s3_cnt }
		multiply_variable = { which = wr_ruthenian_s3_pct value = 100 }
		divide_variable = { which = wr_ruthenian_s3_pct which = wr_ruthenian_total }
	}
	if = {
		limit = { check_variable = { which = wr_rusyn_total value = 1 } }
		set_variable = { which = wr_rusyn_s1_pct which = wr_rusyn_s1_cnt }
		multiply_variable = { which = wr_rusyn_s1_pct value = 100 }
		divide_variable = { which = wr_rusyn_s1_pct which = wr_rusyn_total }
		set_variable = { which = wr_rusyn_s3_pct which = wr_rusyn_s3_cnt }
		multiply_variable = { which = wr_rusyn_s3_pct value = 100 }
		divide_variable = { which = wr_rusyn_s3_pct which = wr_rusyn_total }
	}
}

wr_refresh_all_sphere_indices_effect = {
	every_country = {
		limit = {
			wr_is_maneuver_relevant_country = yes
		}
		wr_recalculate_maneuver_indices_effect = yes
	}
}

# ------------------------------------------------------------------------
# Stage 1: Recognition of West-Ruthenian cultural distinctiveness (<= 50%)
# ------------------------------------------------------------------------

wr_dispatch_three_circle_congress_stage1_effect = {
	every_country = {
		limit = {
			NOT = { tag = ROOT }
			wr_can_run_maneuver_base = yes
			NOT = { has_country_flag = wr_congress_s1_responded }
			NOT = { has_country_flag = wr_west_slavic_affinity_active }
			NOT = { has_country_flag = wr_west_slavic_shift_completed }
			OR = {
				AND = {
					wr_byelorussian_congress_member = yes
					ROOT = { wr_byelorussian_initiator_stake = yes }
				}
				AND = {
					wr_ruthenian_congress_member = yes
					ROOT = { wr_ruthenian_initiator_stake = yes }
				}
				AND = {
					wr_rusyn_congress_member = yes
					ROOT = { wr_rusyn_initiator_stake = yes }
				}
			}
		}
		if = {
			limit = {
				wr_is_carrier_country = yes
				NOT = { has_country_flag = wr_distinctiveness_recognized }
			}
			country_event = {
				id = wr_cultural_maneuver.41
				days = 3
			}
		}
		else_if = {
			limit = {
				wr_has_accepted_target_culture = yes
				NOT = { has_country_flag = wr_distinctiveness_recognized }
			}
			country_event = {
				id = wr_cultural_maneuver.51
				days = 4
			}
		}
		else_if = {
			limit = {
				wr_has_unaccepted_target_provinces = yes
			}
			country_event = {
				id = wr_cultural_maneuver.61
				days = 5
			}
		}
	}
}

wr_apply_stage1_distinctiveness_effect = {
	set_country_flag = wr_distinctiveness_recognized
	add_country_modifier = {
		name = wr_west_ruthenian_distinctiveness
		duration = -1
	}
	add_prestige = 10
	if = {
		limit = {
			wr_byelorussian_initiator_stake = yes
			NOT = { wr_is_byelorussian_country = yes }
			any_owned_province = {
				culture = byelorussian
			}
			NOT = { accepted_culture = byelorussian }
		}
		add_accepted_culture = byelorussian
	}
	if = {
		limit = {
			wr_ruthenian_initiator_stake = yes
			NOT = { wr_is_ruthenian_country = yes }
			any_owned_province = {
				culture = ruthenian
			}
			NOT = { accepted_culture = ruthenian }
		}
		add_accepted_culture = ruthenian
	}
	if = {
		limit = {
			wr_rusyn_initiator_stake = yes
			NOT = { wr_is_rusyn_country = yes }
			any_owned_province = {
				culture = rusyn
			}
			NOT = { accepted_culture = rusyn }
		}
		add_accepted_culture = rusyn
	}
	if = {
		limit = {
			wr_byelorussian_initiator_stake = yes
		}
		every_owned_province = {
			limit = {
				wr_is_byelorussian_province = yes
			}
			wr_apply_stage1_charter_to_province_effect = yes
		}
	}
	if = {
		limit = {
			wr_ruthenian_initiator_stake = yes
		}
		every_owned_province = {
			limit = {
				wr_is_ruthenian_province = yes
			}
			wr_apply_stage1_charter_to_province_effect = yes
		}
	}

		limit = {
			wr_rusyn_initiator_stake = yes
		}
		every_owned_province = {
			limit = {
				wr_is_rusyn_province = yes
			}
			wr_apply_stage1_charter_to_province_effect = yes
		}
	}
	wr_dispatch_three_circle_congress_stage1_effect = yes
	wr_refresh_all_sphere_indices_effect = yes
}

				wr_is_target_province = yes
			}
		}

			limit = {
				wr_is_target_province = yes
			}
			spawn_rebels = {
				type = particularist_rebels
				size = 1
			}
		}
	}
	else = {
		random_owned_province = {
			spawn_rebels = {
				type = particularist_rebels
				size = 1
			}
		}
	}
}
