##############################################################
# The two disasters of the faith mechanics.
#
# rip_ro_raskol and rip_uc_brotherhoods, both in
# common/disasters/rip_faith_disasters.txt. The Raskol starts because the
# Muscovite church succeeded; the Brotherhoods start because the Uniate one
# did not. Everything they read - the three-step wave, the iconostasis, the
# standing at Rome - is the same machinery the pulses run on.
##############################################################

namespace = rip_faith_disaster

# ---------------------------------------------------------------- THE RASKOL

country_event = {
	id = rip_faith_disaster.1
	title = rip_faith_disaster.1.t
	desc = rip_faith_disaster.1.d
	picture = RELIGIOUS_TURMOIL_eventPicture

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# The Old Believers now exist. rip_ro_zealot_backlash_effect reads
			# this flag on every pulse - before it is set there is nothing for
			# anyone to be an old believer about.
			set_country_flag = rip_ro_stage_raskol
			set_global_flag = raskol_schism
		}
	}

	option = {	# Correct the books. The church decides what is correct.
		name = rip_faith_disaster.1.a
		ai_chance = { factor = 55 }

		add_patriarch_authority = 0.05
		add_stability = -1
		every_owned_province = {
			limit = {
				is_city = yes
				religion = russian_orthodox
			}
			add_unrest = 2
		}
		random_owned_province = {
			limit = {
				is_city = yes
				religion = russian_orthodox
			}
			spawn_rebels = {
				type = heretic_rebels
				size = 2
			}
		}
	}

	option = {	# Let the old rite stand where it stands.
		name = rip_faith_disaster.1.b
		ai_chance = { factor = 45 }

		add_patriarch_authority = -0.15
		add_prestige = -10
		add_country_modifier = {
			name = old_believer_underground
			duration = 10950
		}
	}
}
