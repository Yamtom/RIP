# ========================================================================
# Palanok, the castle above Mukachevo.
# Authenticity: B - documented castle, invented council scenes.
# Sources:
#   the castle on the volcanic hill commanding the Verecke and Uzhok
#   passes; the Solotvyno salt coming down the Upper Tisza past the only
#   lockable roof on the road; the Transcarpathian counties that met in
#   its great hall; the Rakoczi princes who held it against Vienna.
# The monument itself is common/great_projects/RIP_UZH_great_projects.txt.
# Hooked from tier on_upgraded, not on_built: date = 1400.01.01 puts the
# castle on the map at every bookmark the mod ships, so on_built is never
# reached. No vanilla monument fires an event from either hook, so this is
# UNVERIFIED IN GAME: if the three events never appear after an upgrade, the
# fallback is on_great_project_constructed or a pulse gated on has_great_project.
#
# Each option also moves the Palanok scale (states_general_mechanic of
# uzh_palanok_captaincy_reform) 15 points toward the party it favours: walls,
# levies and the arsenal for the Captains' Party; the town's wharf, the home
# harvest and the seated diet for the Sejm Party. See
# common/scripted_effects/uzh_states_general_effects.txt.
# ========================================================================

namespace = uzh_palanok

# --- 1. Tier 1: the passes are closed.
country_event = {
	id = uzh_palanok.1
	title = uzh_palanok.1.t
	desc = uzh_palanok.1.d
	picture = FORT_eventPicture

	is_triggered_only = yes

	option = {
		name = uzh_palanok.1.a
		uzh_sg_captains_15_effect = yes
		ai_chance = { factor = 60 }
		add_country_modifier = {
			name = uz_palank_border_watch
			duration = -1
		}
	}

		name = uzh_palanok.1.b
		uzh_sg_sejm_15_effect = yes
		ai_chance = { factor = 40 }
		add_mil_power = 50
		add_army_tradition = 5
	}
}
