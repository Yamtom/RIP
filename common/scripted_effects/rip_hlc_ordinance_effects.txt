# =============================================================================
# Clean-up for the Chancery Ordinances
# (common/government_mechanics/rip_hlc_austrian_reorganization.txt).
#
# Each ordinance is one country modifier of ten years, so it ends by itself;
# what has to be done by hand is to take them out when the crown authority
# that enacted them is gone, and to move onto the ten-year term the permanent
# ordinances (and their flags) that a save made before the six-ordinance
# mechanic still holds. Two places do that, and both call these effects:
#
#  - hlc_crown_authority_reform.removed_effect
#    (common/government_reforms/RIP_HLC_government_reforms.txt) - the normal
#    way out, the reform being dropped;
#  - hlc_ordinance_sweep_effect, from on_yearly_pulse
#    (common/on_actions/RIP_HLC_on_actions.txt) - the cases a reform's
#    removed_effect cannot see: a tag change, a government change made by an
#    event or a revolution, a save made before this clean-up existed.
#
# Only if-blocks are used. Inside an else_if/else body of a scripted effect the
# engine drops some effects without an error (diagnostics/engine_probes_20260929).
# =============================================================================

# Take every ordinance out of force, whatever its state. Flags (the record of
# the older, permanent ordinances) and modifiers are cleared separately so that
# a stray modifier with no flag, or the reverse, goes too. The hidden timer of
# the settlement ordinance is left to run out by itself: it has no effect of
# its own.
hlc_clear_ordinances_effect = {
	clr_country_flag = hlc_chancery_ordinance_active
	clr_country_flag = hlc_settlement_ordinance_active
	clr_country_flag = hlc_cadastral_ordinance_active
	if = {
		limit = { has_country_modifier = hlc_chancery_ordinance_modifier }
		remove_country_modifier = hlc_chancery_ordinance_modifier
	}
	if = {
		limit = { has_country_modifier = hlc_settlement_ordinance_modifier }
		remove_country_modifier = hlc_settlement_ordinance_modifier
	}
	if = {
		limit = { has_country_modifier = hlc_cadastral_ordinance_modifier }
		remove_country_modifier = hlc_cadastral_ordinance_modifier
	}
	if = {
		limit = { has_country_modifier = hlc_urbarial_ordinance_modifier }
		remove_country_modifier = hlc_urbarial_ordinance_modifier
	}
	if = {
		limit = { has_country_modifier = hlc_toleration_ordinance_modifier }
		remove_country_modifier = hlc_toleration_ordinance_modifier
	}
	if = {
		limit = { has_country_modifier = hlc_frontier_ordinance_modifier }
		remove_country_modifier = hlc_frontier_ordinance_modifier
	}
}

# An older save holds an ordinance as a flag and a modifier with no end. The
# flag is never set now and is not tested any more either (the script-layer
# check would call that "tested but never set"): one run per country, marked
# by hlc_ordinances_migrated, clears the old flags and gives every ordinance
# modifier of the older three that is still held a ten-year term, so the
# button's own cooldown and the slot count apply to it. Removing and adding is
# the way to change a running modifier's term: the engine does not refresh a
# modifier it already holds (eu4-add-province-modifier-no-refresh in the
# project notes). The three newer ordinances have always had a term.
hlc_ordinance_migrate_effect = {
	if = {
		limit = { NOT = { has_country_flag = hlc_ordinances_migrated } }
		set_country_flag = hlc_ordinances_migrated
		clr_country_flag = hlc_chancery_ordinance_active
		clr_country_flag = hlc_settlement_ordinance_active
		clr_country_flag = hlc_cadastral_ordinance_active
		if = {
			limit = { has_country_modifier = hlc_chancery_ordinance_modifier }
			remove_country_modifier = hlc_chancery_ordinance_modifier
			add_country_modifier = {
				name = hlc_chancery_ordinance_modifier
				duration = 3650
			}
		}
		if = {
			limit = { has_country_modifier = hlc_settlement_ordinance_modifier }
			remove_country_modifier = hlc_settlement_ordinance_modifier
			add_country_modifier = {
				name = hlc_settlement_ordinance_modifier
				duration = 3650
			}
		}
		if = {
			limit = { has_country_modifier = hlc_cadastral_ordinance_modifier }
			remove_country_modifier = hlc_cadastral_ordinance_modifier
			add_country_modifier = {
				name = hlc_cadastral_ordinance_modifier
				duration = 3650
			}
		}
	}
}

# Run once a year for every country, so it is written to cost next to nothing
# outside the Austrian crown: every limit is made only of flag, modifier and
# reform look-ups (no iterated scope, no counting trigger, no variable), and
# the effects behind them run only when a limit passes - for a country that
# never held the crown, none of the six look-ups in the first limit passes and
# the others need the crown authority or the Austrian bureaucracy.
hlc_ordinance_sweep_effect = {
	if = {
		limit = {
			OR = {
				has_country_modifier = hlc_chancery_ordinance_modifier
				has_country_modifier = hlc_settlement_ordinance_modifier
				has_country_modifier = hlc_cadastral_ordinance_modifier
				has_country_modifier = hlc_urbarial_ordinance_modifier
				has_country_modifier = hlc_toleration_ordinance_modifier
				has_country_modifier = hlc_frontier_ordinance_modifier
			}
			NOT = { has_reform = hlc_crown_authority_reform }
		}
		hlc_clear_ordinances_effect = yes
	}
	if = {
		limit = {
			has_reform = hlc_crown_authority_reform
			NOT = { has_country_flag = hlc_ordinances_migrated }
		}
		hlc_ordinance_migrate_effect = yes
	}
	# Dates the Austrian bureaucracy for the AI: hlc_incorporate_appanage waits
	# ten years after it (common/new_diplomatic_actions/RIP_HLC_diplomatic_actions.txt).
	# Set here, not in the reform, so that a save which already holds the reform
	# starts its clock at the next pulse instead of never.
	if = {
		limit = {
			has_reform = hlc_austrian_bureaucracy_reform
			NOT = { has_country_flag = hlc_austrian_bureaucracy_dated }
		}
		set_country_flag = hlc_austrian_bureaucracy_dated
	}
}
