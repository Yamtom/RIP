# Hooks of the faith-and-border reforms. EU4 merges on_action blocks across
# files, so these add to the hooks of orthodox_crusade_on_actions.txt and the
# other files rather than replacing them.

# Zeal of the Host (zaz_orthodox_knighthood_reform).
on_battle_won_country = {
	rip_zaz_zeal_battle_effect = yes
}

on_siege_won_country = {
	rip_zaz_zeal_siege_effect = yes
}

# The houses of the order are re-counted once a year; a lost province takes its
# monastery and its hospital bonus with it.
on_yearly_pulse = {
	if = {
		limit = { has_reform = zaz_orthodox_knighthood_reform }
		rip_zaz_sync_monasteries_effect = yes
	}
}
