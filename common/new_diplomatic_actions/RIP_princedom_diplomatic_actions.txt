# The princedom, from the overlord's side.
#
# Neither of these is gated on a tag. They belong to whoever holds the reforms
# and the subjects: a princedom in this mod is a vassal that can never be
# annexed or integrated, ranked no higher than a duchy, and paying half.
#
# Both are one-sided and need no acceptance, which is the WFAS shape - a
# subject's standing changed by the overlord's own hand, with no event and no
# decision in between.

# --- The charter of a princedom.
#
# The overlord side, and the WFAS shape: no acceptance, the overlord simply
# writes down what the princedom may keep. It costs absolutism and buys quiet.
rip_grant_the_princely_charter = {
	category = influence

	require_acceptance = no

	is_visible = {
		overlord_of = FROM
		FROM = {
			is_subject_of_type = princedom
		}
	}

	is_allowed = {
		is_at_war = no
		prestige = 5
		FROM = {
			is_at_war = no
			NOT = { has_country_modifier = rip_princely_charter }
		}
	}

	on_accept = {
		# add_absolutism was the stated price and it is arithmetically nothing
		# before 1620, while the reforms that reach these actions are live from
		# 1444. A charter is a thing written down in front of the nobility, so
		# the nobility is what it costs: they gain ground the crown gives up.
		add_prestige = -5
		add_estate_influence_modifier = {
			estate = estate_nobles
			desc = EST_VAL_RIP_PRINCELY_CHARTER
			influence = 5
			duration = 9125
		}
		add_favors = {
			who = FROM
			amount = 10
		}
		FROM = {
			add_country_modifier = {
				name = rip_princely_charter
				duration = -1
			}
			add_liberty_desire = -25
			add_legitimacy_equivalent = { amount = 5 republican_tradition = 3 }
		}
	}

	# A trigger, not a weight block - see the note in the PLC file.
	ai_will_do = {
		FROM = {
			liberty_desire = 50
		}
	}
}

# --- The udil.
#
# The Kyivan Rus reform rules through kinsmen holding land, not through
# governors holding office, and this is the instrument of it: any subject can
# be made an udil - a princedom, which in this mod's subject types is a vassal
# that can never be annexed or integrated, ranked no higher than a duchy, and
# paying half.
#
# It is deliberately one-sided and needs no acceptance. Handing a kinsman a
# province was not a negotiation; it was how the realm was held together, and
# it is also how it came apart.
rip_grant_the_udil = {
	category = influence

	require_acceptance = no

	is_visible = {
		overlord_of = FROM
		OR = {
			has_reform = kyivan_rus_reform
			has_reform = kyivan_cesarstvo_reform
			has_reform = kyivan_shogunate_reform
			has_reform = ruthenian_principality_reform
			has_reform = principality_appanage
		}
		FROM = {
			NOT = { is_subject_of_type = princedom }
		}
	}

	is_allowed = {
		is_at_war = no
		prestige = 10
		FROM = {
			is_at_war = no
		}
	}

	on_accept = {
		# The cost is the rota itself, and it stacks: one rip_the_rota for every
		# udil standing, each -0.25 legitimacy, +0.5 unrest and -5% vassal force
		# limit. The first is cheap. The fifth is what the comment above means by
		# "it is also how it came apart", and it is felt from 1444 rather than
		# from 1620, which is all add_absolutism ever managed.
		add_prestige = -10
		add_country_modifier = {
			name = rip_the_rota
			duration = -1
		}
		create_subject = {
			subject_type = princedom
			subject = FROM
		}
		FROM = {
			add_liberty_desire = -20
			add_legitimacy_equivalent = { amount = 5 republican_tradition = 3 }
			add_country_modifier = {
				name = rip_held_as_an_udil
				duration = -1
			}
		}
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = {
				estate = estate_nobles
				loyalty = 10
			}
		}
	}

	# A trigger, not a weight block. The eight-subject brake was written as a
	# zero-weight modifier, which this slot does not read; it is a NOT now.
	ai_will_do = {
		FROM = {
			liberty_desire = 50
		}
		NOT = { num_of_subjects = 8 }
	}
}

# --- Taking it back.
#
# There was no orderly way to un-give an udil, which is the point: the senior
# prince who tried it was making an enemy of a kinsman with a claim, and every
# other kinsman watched him do it. So this is cheap in coin and expensive in
# everything else - the ex-udil gets a claim on you, every other princedom hears
# about it, and the nobility that witnessed the grant remembers the revocation.
rip_revoke_the_udil = {
	category = influence

	require_acceptance = no

	is_visible = {
		overlord_of = FROM
		FROM = {
			is_subject_of_type = princedom
			has_country_modifier = rip_held_as_an_udil
		}
	}

	is_allowed = {
		is_at_war = no
		prestige = 20
		FROM = {
			NOT = { liberty_desire = 75 }
		}
	}

	on_accept = {
		add_prestige = -20
		add_country_modifier = {
			name = rip_the_udil_revoked
			duration = 7300
		}
		remove_country_modifier = rip_the_rota
		add_estate_loyalty = {
			estate = estate_nobles
			loyalty = -15
		}
		FROM = {
			remove_country_modifier = rip_held_as_an_udil
			add_liberty_desire = 40
			add_casus_belli = {
				target = ROOT
				type = cb_disloyal_vassal
				months = 300
			}
			add_opinion = {
				who = ROOT
				modifier = lit_union_grievance
			}
		}
		every_subject_country = {
			limit = {
				is_subject_of_type = princedom
				NOT = { tag = FROM }
			}
			add_liberty_desire = 15
		}
	}

	# A trigger, not a weight block.
	ai_will_do = {
		FROM = {
			liberty_desire = 70
		}
	}
}
