# ========================================================================
# The union taken by the crown.
# Authenticity: C - counterfactual. Tier: SPECULATIVE. The mod's own
# alternate history, and it says so where the player can read it: the
# opening event and the capstone both carry the alternate-history line.
#
# Sources and the shape of the counterfactual:
#   Every union that was actually signed - Brest 1596, Uzhhorod 1646,
#   Peremyshl 1691, Lviv 1700, Lutsk 1702 - was signed by bishops living
#   under a Catholic crown, and Rome's whole interest was that it did not
#   have to latinise anybody. That road is already in the mod, as
#   convert_to_greek_catholic_decision in decisions/GreekCatholicDecisions.txt
#   and the twenty flavour events of namespace uniate_church. The bishops
#   act; the crown follows.
#
#   This chain is the mirror image, and it did not happen. It is built on
#   the English settlement instead: Henry did not take the church out of
#   conviction, he took it because Rome would not give him what he wanted,
#   and he kept nearly every outward form. Here a Ruthenian sovereign wants
#   something Rome can give and Constantinople cannot - a dispensation for
#   the marriage that settles the succession, or the word kingdom written by
#   a hand the western courts read - and pays for it in the obedience of a
#   metropolitanate that was never asked. The rite does not move. The chain
#   of command does.
#
#   Note what is deliberately NOT set here: pursuing_uniate_union and
#   union_of_brest_happened. Those flags belong to the bishops' union and
#   gate its flavour. A church taken by the palace has not been to Brest and
#   should not inherit Brest's story; what it inherits is everything gated on
#   religion = greek_catholic, which is most of the later chain anyway.
#
#   The cost is the clergy estate, the parishes, and one permanent country
#   modifier, rip_ucr_church_of_the_palace, that says out loud who the
#   bishops answer to. All payloads are in
#   common/event_modifiers/RIP_faith_modifiers.txt.
# ========================================================================


namespace = uniate_crown


# ------------------------------------------------------------------ 1
# The reason of state. No church is touched here; the chancery is only
# reading its own list out loud, and the metropolitan is not in the room.
country_event = {
	id = uniate_crown.1
	title = uniate_crown.1.t
	desc = uniate_crown.1.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		religion = orthodox
		is_ruthenian_culture_area = yes
		# The whole point of this entry is that it is NOT the Brest entry: no
		# Catholic overlord, no Catholic neighbour leaning on the sees. An
		# independent crown with a want of its own.
		is_subject = no
		NOT = { has_country_flag = rip_ucr_reason_of_state }
		NOT = { has_country_flag = rip_ucr_declined }
		NOT = { has_country_flag = union_of_brest_happened }
		NOT = { has_country_flag = pursuing_uniate_union }
		NOT = { tag = MOS }
		NOT = { tag = RUS }
		OR = {
			primary_culture = ruthenian
			primary_culture = byelorussian
			primary_culture = rusyn
			capital_scope = { region = ruthenia_region }
		}
		is_year = 1500
		NOT = { is_year = 1700 }
		# A crown strong enough to act over the head of its own synod. A prince
		# who cannot make his own sejm sit will not make his own bishops sign.
		stability = 1
		OR = {
			legitimacy = 90
			absolutism = 30
			republican_tradition = 85
		}
		# Somebody has to be able to deliver what is being bought.
		exists = PAP
		any_known_country = {
			religion = catholic
			is_great_power = yes
			NOT = { war_with = ROOT }
		}
		# and the crown has to want something it cannot get at home
		OR = {
			NOT = { has_heir = yes }
			NOT = { government_rank = 2 }
			any_rival_country = {
				NOT = { religion = orthodox }
				NOT = { religion = russian_orthodox }
			}
		}
	}

	mean_time_to_happen = {
		months = 300
		modifier = {
			factor = 0.6
			NOT = { has_heir = yes }
		}
		modifier = {
			factor = 0.7
			absolutism = 50
		}
		modifier = {
			factor = 0.7
			any_known_country = {
				religion = catholic
				is_great_power = yes
				alliance_with = ROOT
			}
		}
		# A crown whose own clergy already run the country is a crown with less
		# reason to look for a second master.
		modifier = {
			factor = 1.5
			estate_influence = {
				estate = estate_church
				influence = 60
			}
		}
	}

	option = { # Name the title as the price
		name = uniate_crown.1.a
		ai_chance = {
			factor = 40
			modifier = {
				factor = 3
				NOT = { government_rank = 2 }
			}
		}

		set_country_flag = rip_ucr_reason_of_state
		set_country_flag = rip_ucr_want_title

		add_dip_power = -25
		# Asking to be called something is a small humiliation before it is a
		# large promotion.
		add_prestige = -5
	}

	option = { # Name the succession as the price
		name = uniate_crown.1.b
		ai_chance = {
			factor = 40
			modifier = {
				factor = 4
				NOT = { has_heir = yes }
			}
		}

		set_country_flag = rip_ucr_reason_of_state
		set_country_flag = rip_ucr_want_dispensation

		add_dip_power = -25
		add_legitimacy = -5
	}

	option = { # Name nothing
		name = uniate_crown.1.c
		ai_chance = {
			factor = 20
			modifier = {
				factor = 3
				estate_influence = {
					estate = estate_church
					influence = 60
				}
			}
		}

		set_country_flag = rip_ucr_declined

		add_prestige = 5
		add_patriarch_authority = 0.05
		rip_estate_mood_effect = { ESTATE = estate_church LOYALTY = 10 }
	}
}

				estate = estate_church

			}
		}

			limit = {

			name = rip_ucr_metropolitan_in_exile

				OR = {
					religion = orthodox
					religion = russian_orthodox
				}

			}

				who = ROOT
				modifier = rip_ucr_opinion_took_the_church
			}
		}

		country_event = { id = uniate_crown.4 days = 365 }
	}

		trigger = { always = yes }
