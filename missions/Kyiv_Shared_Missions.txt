# KIE, KRU, UKR | Kyivan missions also available to the Kyivan Confederation and Ruthenia.

KIE_missions_1 = {
	slot = 1
	generic = no
	ai = yes
	potential = {
		AND = {
			OR = {
				tag = KIE
				tag = KRU
				tag = UKR
			}
			NOT = {
				map_setup = map_setup_random
			}
		}
	}
	has_country_shield = yes

	KIE_subjugate_volynians = {
		icon = mission_extent_yasak_to_the_east
		required_missions = { KIE_true_ruler }
		position = 2
		provinces_to_highlight = {
			OR = {
				area = volhynia_area
				area = podlasie_area
			}
			NOT = {
				country_or_non_sovereign_subject_holds = ROOT
			}
		}
		trigger = {
			volhynia_area = {
				type = all
				country_or_non_sovereign_subject_holds = ROOT
			}
			podlasie_area = {
				type = all
				country_or_non_sovereign_subject_holds = ROOT
			}
		}
		effect = {
			# Podlasie is already held at this point (see trigger), so the
			# claim goes to Podillia, the next neighbour to the south.
			podolia_volhynia_area = {
				limit = {
					NOT = {
						is_permanent_claim = ROOT
					}
					NOT = {
						is_core = ROOT
					}
				}
				add_permanent_claim = ROOT
			}
			add_prestige = 10
			add_manpower = 2
		}
	}

	KIE_growing_wealth = {

	KIE_true_ruler = {
		icon = king_of_poland
		required_missions = { }
		position = 1
		trigger = {
			OR = {
				AND = {
					monthly_mil = 10
					army_size_percentage = 1
				}
				AND = {
					monthly_dip = 10
					treasury = 150
				}
				AND = {
					monthly_adm = 10
					stability = 1
				}
			}
		}
		effect = {
			add_prestige = 10
			if = {
				limit = {
					exists = VOL
					VOL = {
						is_free_or_tributary_trigger = yes
						is_neighbor_of = ROOT
					}
				}
				add_casus_belli = {
					type = cb_vassalize_mission
					months = 120
					target = VOL
				}
			}
			else_if = {
				limit = {
					exists = VLN
					VLN = {
						is_free_or_tributary_trigger = yes
						is_neighbor_of = ROOT
					}
				}
				add_casus_belli = {
					type = cb_vassalize_mission
					months = 120
					target = VLN
				}
			}
			else = {
				volhynia_area = {
					limit = {
						NOT = { is_core = ROOT }
						NOT = { is_claim = ROOT }
					}
					add_claim = ROOT
				}
			}
		}
	}

	KIE_brotherhood_schools = {
		icon = petro_mohyla
		position = 4
		required_missions = { KIE_magdeburg_law }
		provinces_to_highlight = {
			province_id = 280
		}
		# Two routes to the same school. The historical Epiphany Brotherhood was
		# founded in 1615 by an Orthodox community under a Catholic king, and in
		# 1620 Hetman Sahaidachnyi enrolled the whole Zaporozhian Host in it, so a
		# realm of another faith can patronise Kyiv's Orthodox community instead
		# of building the whole parish network itself.
		trigger = {
			adm_tech = 10
			280 = {
				owned_by = ROOT
			}
			OR = {
				custom_trigger_tooltip = {
					tooltip = KIE_brotherhood_parish_network_tt
					AND = {
						OR = {
							religion = orthodox
							religion = russian_orthodox
						}
						280 = {
							OR = { has_building = temple has_building = cathedral }
						}
						num_of_owned_provinces_with = {
							value = 3
							OR = { has_building = temple has_building = cathedral }
						}
					}
				}
				custom_trigger_tooltip = {
					tooltip = KIE_brotherhood_patronage_tt
					AND = {
						NOT = { religion = orthodox }
						NOT = { religion = russian_orthodox }
						280 = {
							religion = orthodox
							OR = { has_building = temple has_building = cathedral }
						}
						OR = {
							AND = {
								has_estate = estate_burghers
								estate_influence = {
									estate = estate_burghers
									influence = 40
								}
							}
							mission_completed = KIE_cossack
							alliance_with = ZAZ
						}
					}
				}
			}
		}
		effect = {
			add_country_modifier = {
				name = KIE_brotherhood_schools
				duration = 5475
			}
			add_adm_power = 100
			280 = {
				add_base_tax = 1
			}
		}
	}
