# Country scope; no state conversion and no automatic propagation.
rip_du_catholic_patron = {
 religion = catholic
 has_country_flag = rip_church_supports_union
 NOT = { has_country_flag = rip_church_opposes_union }
 is_year = 1596 is_at_war = no stability = 1
 exists = PAP NOT = { war_with = PAP }
 any_owned_province = { religion = greek_catholic }
}
# Province scope, independent of ROOT. One eparchy is closed to every random
# path (the Eparchial Union decision and khmelnytsky_events.7) while an earlier
# offer stands: rip_du_approached is refreshed by every paid offer, so
# had_province_flag reads "last set at least 3650 days ago"; a refusal keeps the
# resistance modifier for ten years; a dated seat event that has been answered
# (events/RIP_EparchyHistory.txt, docs/EPARCHY_TIMELINE.uk.md) closes its seat for
# good. The dated events also set rip_du_approached when they convert a seat.
rip_du_seat_open = {
 NOT = { has_province_modifier = resistance_to_greek_catholic_spread }
 OR = {
  NOT = { has_province_flag = rip_du_approached }
  had_province_flag = { flag = rip_du_approached days = 3650 }
 }
 NOT = { has_province_flag = rip_eparchy_przemysl_union_answered }
 NOT = { has_province_flag = rip_eparchy_lviv_union_answered }
 NOT = { has_province_flag = rip_eparchy_lutsk_union_answered }
}
# Province scope. Named seats only, each opened no earlier than the date its
# hierarchy actually entered communion with Rome (the map province is a
# territorial approximation of the see, not its exact bounds):
#   1596 Brest (Volodymyr-Brest, Potii), Pinsk (Pelchytsky), Lutsk (Terletsky)
#        and Kholm (Zbirujski) - signatories of the Union of Brest.
#   1691 Peremyshl (Vynnytsky, 23 June 1691).   1700 Lviv (Shumlyansky).
# Mukachevo (1952) is deliberately absent: it has its own 1646 event chain.
rip_du_historical_seat = {
 OR = {
  AND = { is_year = 1596 OR = { province_id = 277 province_id = 278 province_id = 279 province_id = 4538 } }
  AND = { is_year = 1691 province_id = 2424 }
  AND = { is_year = 1700 province_id = 2961 }
 }
}
# Province scope, independent of ROOT (country decision or province event).
rip_du_eligible_diocese = {
 rip_du_historical_seat = yes
 is_city = yes religion = orthodox
 is_core = owner controlled_by = owner
 NOT = { has_missionary = yes }
 rip_du_seat_open = yes
}
# Province scope. The land the Catholic crown may press by order in
# khmelnytsky_events.7: an Orthodox city of ruthenia_region that no earlier
# offer or dated seat event has settled. Trigger and effect share this one test,
# so the event never fires for a province the effect would refuse.
rip_du_crown_union_province = {
 is_city = yes religion = orthodox region = ruthenia_region
 controlled_by = owner
 rip_du_seat_open = yes
}

 }
 any_owned_province = { rip_du_eligible_diocese = yes }
