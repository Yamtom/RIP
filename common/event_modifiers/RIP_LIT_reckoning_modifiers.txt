# --- Branch 1: the Kyiv appanage -------------------------------------------------------------------------------------

# Kyiv under an appointed voivode: a tighter grip on the Ruthenian lands, paid for in the princes' trust.

	global_tax_modifier = 0.05
	global_unrest = 1
