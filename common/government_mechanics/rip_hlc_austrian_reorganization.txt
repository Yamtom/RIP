# =============================================================================
# rip_hlc_austrian_reorganization - the Austrian fork of the Galician
# Voivodeship (hlc_centralized_voivodeship_reform -> hlc_austrian_bureaucracy_
# reform -> hlc_crown_authority_reform) never had a mechanic of its own; the
# Poland fork already did (rip_hlc_boyar_council.txt's hlc_boyar_council_
# mechanic, plus has_polish_election on hlc_magnate_assembly_reform). This is
# the Austria fork's counterpart, built the same way the boyar council was -
# from a real vanilla shape, not a copy of it:
#
# Vanilla has no "toggleable law" object as such. What it has, and what this
# copies the structure of, is perfectionism_mechanic's mutually exclusive
# focuses (common/government_mechanics/08_perfectionism.txt): buttons that
# put a persistent (duration = -1) country modifier in force and others that
# take it out again. That is the shape used here for three ordinances the
# chancery can enact and later rescind:
#
#  - The Chancery Ordinance seconds trained clerks to the chancellor's
#    staff - a standing diplomatic bonus while enacted. Vanilla has no
#    scripted effect that changes a ruler's raw skill number outside of
#    ruler generation and add_ruler_personality (which grants a fixed
#    personality, not arbitrary points, and personalities themselves carry
#    no adm/dip/mil fields - checked in common/ruler_personalities/00_core.
#    txt); the honest substitute is a persistent chancery-flavoured country
#    modifier, not a literal ruler stat.
#  - The Settlement Ordinance answers on_province_culture_converted
#    (common/on_actions/00_on_actions.txt) while enacted: culture conversion
#    is cheaper, and the first manual conversion each year steadies the
#    nobles who receive the land - see hlc_settlement_ordinance_conversion_
#    effect (common/scripted_effects/rip_estate_effects.txt) and the on_action
#    hook in common/on_actions/RIP_HLC_on_actions.txt. Vanilla fires that
#    hook only for a MANUAL conversion.
#  - The Cadastral Ordinance sends the chancery's surveyors through every
#    holding, the way the Josephine land survey of the 1780s did, so the
#    treasury collects on what each estate truly holds - and the nobles,
#    counted for the first time, resent it. A standing tax bonus paid for
#    with noble loyalty.
#
# What makes them a choice and not three free bonuses:
#
#  - PRICE. Enacting costs monarch power of the kind the clerks' trade
#    suggests, 50 as in the sibling mechanics and vanilla's own buttons
#    (rip_vln_magnate_senate.txt, 06_devshirme.txt): diplomatic power for the
#    chancery, administrative power for settlement and the cadastre.
#    `dip_power_cost` / `adm_power_cost` in the trigger block is how vanilla
#    puts the price in the button tooltip and checks it can be paid; in the
#    effect block the same name pays it. Rescinding is free.
#  - TWO SLOTS. At most two of the three can be in force. hlc_ordinance_slot_
#    free (common/scripted_triggers/rip_hlc_ordinance_triggers.txt) counts the
#    flags and every enact button requires it, so the third ordinance can be
#    had only by rescinding one of the others. An ordinance's enact and
#    rescind buttons share one cooldown token (cooldown_desc names it in the
#    cooldown tooltip), so it can be neither rescinded nor enacted again
#    within five years of its last change - a choice is kept, not toggled.
#  - THE AI. Every button has an ai_chance: it enacts what its realm needs
#    (settlement when several owned cities hold a culture that is neither
#    primary nor accepted, the cadastre when short of money and the nobles are
#    loyal, the chancery otherwise) and it rescinds what has turned against it
#    (the cadastre when noble loyalty has fallen, the settlement when no such
#    city is left, anything beyond two). The situations are the *_wanted /
#    *_spent / *_dangerous scripted triggers in the same trigger file.
#  - CLEAN-UP. hlc_crown_authority_reform.removed_effect and a yearly sweep
#    (hlc_ordinance_sweep_effect, common/scripted_effects/rip_hlc_ordinance_
#    effects.txt) take every ordinance out of force once the crown authority
#    that enacted them is gone, so none outlives it.
#
# Layout: SIX buttons, two full rows of three. The top row enacts, the bottom
# row rescinds, and each column is one ordinance (chancery, settlement,
# cadastre). The order of the interactions below is the order of the buttons.
# The bottom row sits below the fold of vanilla's mechanics list, which shows
# one row of three; the wheel scrolls to it. That is a limit of the vanilla
# window which the author has been told about, not something this file can
# change. A later revision folded each enact/rescind pair into one toggle
# button so that all three fitted on the visible row; the
# author asked for six abilities explicitly, so the toggle design was
# deliberately replaced by this one. What was kept of it: the shared-token
# cooldown_desc and the situational AI (every button decides from the
# *_wanted / *_spent / *_dangerous triggers); dropped: the combined button,
# its OR trigger and its if/else effect, which hid the free rescind inside a
# button priced for enacting.
#
# Buttons-only, same as rip_hlc_boyar_council.txt: no powers, no .gui, no DLC
# gate. Each button has its own sprite (interface/rip_ability_icons.gfx),
# drawn by tools/build_government_ability_icons.py.
# =============================================================================

hlc_austrian_reorganization_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	available = {
		has_reform = hlc_crown_authority_reform
	}

	interactions = {
		# Top row: enact.
		hlc_enact_chancery_ordinance = {
			icon = GFX_rip_chancery_enact
			trigger = {
				has_reform = hlc_crown_authority_reform
				custom_trigger_tooltip = {
					tooltip = hlc_chancery_ordinance_not_in_force_tt
					NOT = { has_country_flag = hlc_chancery_ordinance_active }
				}
				custom_trigger_tooltip = {
					tooltip = hlc_ordinance_slot_free_tt
					hlc_ordinance_slot_free = yes
				}
				dip_power_cost = 50
			}
			effect = {
				dip_power_cost = 50
				set_country_flag = hlc_chancery_ordinance_active
				add_country_modifier = {
					name = hlc_chancery_ordinance_modifier
					duration = -1
				}
			}
			cooldown_years = 5
			cooldown_token = hlc_chancery_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 4
				modifier = {
					factor = 0
					has_country_flag = hlc_chancery_ordinance_active
				}
				# The plainest of the three: a realm that needs settlers or
				# a survey reaches for those first.
				modifier = {
					factor = 0.25
					OR = {
						hlc_ordinance_settlement_wanted = yes
						hlc_ordinance_cadastre_wanted = yes
					}
				}
			}
		}
		hlc_enact_settlement_ordinance = {
			icon = GFX_rip_settlement_enact
			trigger = {
				has_reform = hlc_crown_authority_reform
				custom_trigger_tooltip = {
					tooltip = hlc_settlement_ordinance_not_in_force_tt
					NOT = { has_country_flag = hlc_settlement_ordinance_active }
				}
				custom_trigger_tooltip = {
					tooltip = hlc_ordinance_slot_free_tt
					hlc_ordinance_slot_free = yes
				}
				adm_power_cost = 50
			}
			effect = {
				adm_power_cost = 50
				set_country_flag = hlc_settlement_ordinance_active
				add_country_modifier = {
					name = hlc_settlement_ordinance_modifier
					duration = -1
				}
			}
			cooldown_years = 5
			cooldown_token = hlc_settlement_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 1
				modifier = {
					factor = 0
					has_country_flag = hlc_settlement_ordinance_active
				}
				# Nothing to settle, nothing to enact.
				modifier = {
					factor = 0
					NOT = { hlc_ordinance_settlement_wanted = yes }
				}
				# A realm full of unaccepted communities needs it more.
				modifier = {
					factor = 4
					num_of_owned_provinces_with = {
						value = 6
						is_city = yes
						has_owner_culture = no
						has_owner_accepted_culture = no
					}
				}
			}
		}
		hlc_enact_cadastral_ordinance = {
			icon = GFX_rip_cadastre_enact
			trigger = {
				has_reform = hlc_crown_authority_reform
				custom_trigger_tooltip = {
					tooltip = hlc_cadastral_ordinance_not_in_force_tt
					NOT = { has_country_flag = hlc_cadastral_ordinance_active }
				}
				custom_trigger_tooltip = {
					tooltip = hlc_ordinance_slot_free_tt
					hlc_ordinance_slot_free = yes
				}
				adm_power_cost = 50
			}
			effect = {
				adm_power_cost = 50
				set_country_flag = hlc_cadastral_ordinance_active
				add_country_modifier = {
					name = hlc_cadastral_ordinance_modifier
					duration = -1
				}
			}
			cooldown_years = 5
			cooldown_token = hlc_cadastral_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 3
				modifier = {
					factor = 0
					has_country_flag = hlc_cadastral_ordinance_active
				}
				# The survey costs noble loyalty: only a crown that is short of
				# money and whose nobles are loyal sends the surveyors out.
				modifier = {
					factor = 0
					NOT = { hlc_ordinance_cadastre_wanted = yes }
				}
			}
		}
		# Bottom row: rescind. Free of cost, bound by the same cooldown token
		# as the ordinance's enact button.
		hlc_rescind_chancery_ordinance = {
			icon = GFX_rip_chancery_rescind
			trigger = {
				custom_trigger_tooltip = {
					tooltip = hlc_chancery_ordinance_in_force_tt
					has_country_flag = hlc_chancery_ordinance_active
				}
			}
			effect = {
				clr_country_flag = hlc_chancery_ordinance_active
				remove_country_modifier = hlc_chancery_ordinance_modifier
			}
			cooldown_years = 5
			cooldown_token = hlc_chancery_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 10
				# Only to make room for an ordinance the realm needs more, or
				# to bring a realm back to two.
				modifier = {
					factor = 0
					NOT = {
						OR = {
							hlc_ordinance_over_limit = yes
							hlc_ordinance_room_needed = yes
						}
					}
				}
			}
		}
		hlc_rescind_settlement_ordinance = {
			icon = GFX_rip_settlement_rescind
			trigger = {
				custom_trigger_tooltip = {
					tooltip = hlc_settlement_ordinance_in_force_tt
					has_country_flag = hlc_settlement_ordinance_active
				}
			}
			effect = {
				clr_country_flag = hlc_settlement_ordinance_active
				remove_country_modifier = hlc_settlement_ordinance_modifier
			}
			cooldown_years = 5
			cooldown_token = hlc_settlement_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 10
				# Once no unaccepted city is left its benefit no longer applies.
				modifier = {
					factor = 0
					NOT = {
						OR = {
							hlc_ordinance_over_limit = yes
							hlc_ordinance_settlement_spent = yes
						}
					}
				}
			}
		}
		hlc_rescind_cadastral_ordinance = {
			icon = GFX_rip_cadastre_rescind
			trigger = {
				has_reform = hlc_crown_authority_reform
				custom_trigger_tooltip = {
					tooltip = hlc_cadastral_ordinance_in_force_tt
					has_country_flag = hlc_cadastral_ordinance_active
				}
			}
			effect = {
				clr_country_flag = hlc_cadastral_ordinance_active
				remove_country_modifier = hlc_cadastral_ordinance_modifier
			}
			cooldown_years = 5
			cooldown_token = hlc_cadastral_ordinance
			cooldown_desc = hlc_ordinance_cooldown_desc
			ai_chance = {
				factor = 10
				# Noble loyalty has fallen too far to be counted again.
				modifier = {
					factor = 0
					NOT = {
						OR = {
							hlc_ordinance_over_limit = yes
							hlc_ordinance_cadastre_dangerous = yes
						}
					}
				}
			}
		}
	}
}
