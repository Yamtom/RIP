# =============================================================================
# rip_kyivan_seniorate - kyivan_shogunate_reform (localised "Kyivan Seniorate")
# was mapped onto shogunate_mechanic, which is the Japanese daimyo UI and wrong
# for it. This is the Seniorate proper: the senior prince at Kyiv holds junior
# Rurikid princes as senior_udil subjects (common/subject_types/rip_senior_udil.txt)
# - a rotational seat, not a fixed patrimony, owing the levy rather than the
# rent, held at the senior prince's pleasure and lapsing when a new senior
# prince takes the throne (kie_on_actions.txt : on_monarch_death).
#
# Three accumulating powers, one per interaction, the same shape vanilla uses
# for russian_mechanic (common/government_mechanics/04_russian.txt): each power
# takes base_monthly_growth plus monarch_power, so the ruler's own ADM/DIP/MIL
# skill adds to the monthly gain - that scaling is the engine's own behaviour
# once monarch_power is named on the power, not a literal number written here,
# same as vanilla's file. Each interaction's cost_type/cost is its power and
# its threshold at once: below cost the button is unavailable, and progress
# cannot climb past max, so a reign that never spends a power still loses the
# overflow past the ceiling every month. This replaces the old flat mil/dip/adm
# cost plus cooldown_years scheme; the three interactions keep their prior
# names, icons and effects.
#
# Buttons render through the default government interactions list, no custom
# .gui, vanilla Cossack ability sprites for the icons. Gated on Third Rome /
# Domination like kyivan_seniority_mechanic - the UI panel comes with those.
# =============================================================================

kyivan_seniorate_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	available = {
		OR = {
			has_dlc = "Third Rome"
			has_dlc = "Domination"
		}
		has_reform = kyivan_shogunate_reform
	}

	powers = {
		# The levy: the seat's one standing due to Kyiv, so it leans on the
		# same skill that mans it.
		kyiv_princely_levy_power = {
			max = 100
			base_monthly_growth = 0.7
			monarch_power = MIL
			reset_on_new_ruler = no
		}
		# The rota: a diplomatic act, moving a prince up the ladder of
		# thrones, so it leans on DIP rather than MIL or ADM.
		kyiv_seat_rotation_power = {
			max = 100
			base_monthly_growth = 0.6
			monarch_power = DIP
			reset_on_new_ruler = no
		}
		# Resets on succession on purpose: a new senior prince has not yet
		# earned the standing to reconfirm anyone, which is exactly the gap
		# rip_tenure_lapsed (kie_on_actions.txt : on_monarch_death) opens
		# under him that same month.
		kyiv_grant_reconfirmation_power = {
			max = 100
			base_monthly_growth = 1.2
			monarch_power = ADM
			reset_on_new_ruler = yes
		}
	}

	interactions = {
		# Call up the senior udils' full retinues for a campaign - more than the
		# standing forcelimit the seat already owes.
		kyiv_call_the_princely_levy = {
			icon = GFX_cossacks_ability_MIL
			cost_type = kyiv_princely_levy_power
			cost = 100
			trigger = {
				any_subject_country = {
					is_subject_of_type = senior_udil
				}
				NOT = { has_country_modifier = rip_kyiv_princely_levy }
			}
			effect = {
				add_country_modifier = {
					name = rip_kyiv_princely_levy
					duration = 3650
				}
			}
			cooldown_years = 0
			ai_chance = {
				factor = 1
				modifier = {
					factor = 10
					is_at_war = yes
				}
			}
		}
		# Move a junior prince up the ladder of thrones. A rotation the senior
		# prince controls resets the seat-holder's standing toward Kyiv.
		kyiv_rotate_the_seats = {
			icon = GFX_cossacks_ability_DIP
			cost_type = kyiv_seat_rotation_power
			cost = 100
			trigger = {
				any_subject_country = {
					is_subject_of_type = senior_udil
					liberty_desire = 30
				}
			}
			effect = {
				random_subject_country = {
					limit = {
						is_subject_of_type = senior_udil
						liberty_desire = 30
					}
					add_liberty_desire = -40
					add_opinion = {
						who = ROOT
						modifier = kyivan_confirmed_in_the_udil
					}
				}
				add_legitimacy_equivalent = {
					amount = 3
					republican_tradition = 2
				}
			}
			cooldown_years = 0
			ai_chance = {
				factor = 1
				modifier = {
					factor = 10
					any_subject_country = {
						is_subject_of_type = senior_udil
						liberty_desire = 55
					}
				}
			}
		}
		# Reconfirm every seat after a succession has lapsed the grants.
		kyiv_reconfirm_the_grants = {
			icon = GFX_cossacks_ability_ADM
			cost_type = kyiv_grant_reconfirmation_power
			cost = 80
			trigger = {
				any_subject_country = {
					is_subject_of_type = senior_udil
					has_country_modifier = rip_tenure_lapsed
				}
			}
			effect = {
				add_prestige = -5
				every_subject_country = {
					limit = {
						is_subject_of_type = senior_udil
						has_country_modifier = rip_tenure_lapsed
					}
					remove_country_modifier = rip_tenure_lapsed
					add_liberty_desire = -15
					add_trust = {
						who = ROOT
						value = 10
						mutual = yes
					}
				}
			}
			cooldown_years = 0
			ai_chance = {
				factor = 20
			}
		}
	}
}
