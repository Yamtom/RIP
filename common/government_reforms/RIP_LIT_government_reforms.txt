# =============================================================================
# RIP_LIT_government_reforms
# The Grand Duchy of Lithuania, reworked.
#
# Vanilla's grand_duchy_reform is a stub: one modifier, governing capacity 200,
# and nothing that says what the Grand Duchy actually was - a Catholic dynasty
# over a country that was mostly Orthodox, ruled through a council of lords, a
# chancery that wrote in Ruthenian, and a levy of boyars who held land for it.
# This file replaces that stub by key, so anything that already tests
# has_reform = grand_duchy_reform keeps working, and hangs a ladder of six
# Lithuanian institutions off it.
# Version-sensitive override: audited against vanilla 1.37.5. Recompare this
# definition whenever the supported EU4 version changes.
#
# The two tier-1 reforms at the bottom are the ends of the union quarrel in
# decisions/UnionDispute.txt: the Commonwealth kept as two nations, or the
# Grand Duchy under a crown of its own. Each replaces grand_duchy_reform in the
# same tier, the way the Siversk roads replace the veche.
#
# Every reform here is registered in common/governments/00_governments.txt, or
# it would never be offered, and every one carries a potential block, or it
# would be offered to every country in the game.
# =============================================================================

# --- tier 1: the Grand Duchy itself ----------------------------------------

grand_duchy_reform = {
	icon = "king_2"
	allow_normal_conversion = yes
	potential = {
		OR = {
			AND = {
				tag = LIT
				NOT = { map_setup = map_setup_random }
			}
			has_reform = grand_duchy_reform
			AND = {
				is_playing_custom_nation = yes
				have_had_reform = grand_duchy_reform
			}
		}
	}
	legacy_equivalent = despotic_monarchy
	fixed_rank = 1
	valid_for_nation_designer = yes
	nation_designer_cost = 10
	modifiers = {
		governing_capacity = 200
		num_accepted_cultures = 2
		cavalry_power = 0.1
		global_unrest = -1
		diplomatic_reputation = 1
		global_autonomy = 0.05
	}
	custom_attributes = {
		enables_aristocratic_idea_group = yes
	}
}
