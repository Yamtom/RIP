country_event = {
	id = rip_galicia.2
	title = rip_galicia.2.t
	desc = rip_galicia.2.d
	picture = REFORM_eventPicture
	fire_only_once = yes
	trigger = {
		west_ukraine_hlc_heritage_missions = yes
		# The synod follows the fortified frontier: without this gate it sat two
		# years after the path was chosen, and its ten-year modifiers had run out
		# before hlc_carpathian_bulwark / hlc_confessional_compact could renew them.
		# The mission itself fires the event at once, so the gate only stops the
		# MTTH draw from pre-empting it.
		mission_completed = hlc_carpathian_bulwark
		OR = {
			has_country_flag = rip_galicia_path_austria
			has_country_flag = rip_galicia_path_poland
			# A Volhynian-origin VOL never chose a Galician path; its synod is held
			# by the frontier mission itself.
			tag = VOL
		}
		NOT = {
			has_country_flag = rip_galicia_confessional_set
		}
	}
	mean_time_to_happen = {
		months = 24
	}
	option = {
		name = rip_galicia.2.a
		# Uniate Synod
		set_country_flag = rip_galicia_confessional_set
		# The synod sat: later Galician content reads this flag. Alternative
		# history, a game model of the Halych synod. The reward is the
		# recipient's own religious currency; a Greek Catholic state has no
		# papal influence to spend, and gets patriarch_authority instead.
		set_country_flag = rip_galicia_synod_held
		rip_faith_mission_resource_effect = {
			PA = 0.25
			PI = 25
			ADM = 25
		}
		add_country_modifier = {
			name = rip_galicia_uniate_synod
			duration = 3650
		}
		add_country_modifier = {
			name = rip_galicia_sejm_uniate
			duration = 3650
		}
	}
	option = {
		name = rip_galicia.2.b
		# Orthodox Bastion
		set_country_flag = rip_galicia_confessional_set
		# Read by hlc_confessional_compact, which renews the bastion even
		# after the ten-year modifier has expired.
		set_country_flag = rip_galicia_orthodox_held
		add_legitimacy = 10
		if = {
			limit = {
				religion = orthodox
			}
			add_patriarch_authority = 0.05
		}
		add_country_modifier = {
			name = rip_galicia_orthodox_bastion
			duration = 3650
		}
	}
}
