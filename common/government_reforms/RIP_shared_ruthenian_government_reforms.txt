# The udil as a government, not only as a tenure.
#
# Granted by script only, which is what potential = { has_reform = <itself> }
# means - vanilla writes fourteen reforms that way, kalmar_union among them. It
# had exactly one grant site, trade_city_reform inside kyivan_shogunate_reform,
# which is the engine hook that fires when a country turns a province into a
# trading city. That was the original design: Kyivan Rus made a trading city and
# the new subject came out an appanage. It can never fire as written, because
# only a merchant republic makes trading cities and kyivan_shogunate_reform sits
# in the monarchy tier, so no country could ever hold this reform.
#
# The second grant site is rip_grant_the_udil in
# common/new_diplomatic_actions/RIP_princedom_diplomatic_actions.txt, which turns
# an ordinary vassal into an udil - the subject type from
# common/subject_types/Kyivan_princedom.txt and this government together. The
# trade_city_reform line is left where it is: it is the original intent, and it
# costs nothing to keep should a Kyivan republic ever exist.
#
# The payload used to be the trading city's - trade efficiency, burgher influence,
# plutocratic ideas, the free city icon - because that is where it came from. It
# is the udil's now. What an udil is for is putting men in the field for the
# senior prince, and what holds it is the family, so that is what it pays:
# force limit and manpower for the levy, noble influence for the backing, and no
# absolutism at all, because a prince holding of a prince has none. The cheaper
# development is the one thing kept from the trading-city origin, since a portion
# granted to be settled is still a portion granted to be settled.
#
# All five values sit inside the range vanilla uses for the same modifiers in its
# own reforms - force limit 0.15 against a vanilla span of -0.33 to 0.25, manpower
# 0.15 at vanilla's median, noble influence 0.1 under vanilla's 0.15 ceiling - and
# five modifiers is inside vanilla's own median of three and maximum of eight.
principality_appanage = {
	icon = "nobleman"
	allow_normal_conversion = no
	potential = {
		has_reform = principality_appanage
	}
	lock_level_when_selected = yes
	queen = yes
	heir = yes
	royal_marriage = yes
	rulers_can_be_generals = yes
	modifiers = {
		land_forcelimit_modifier = 0.15
		global_manpower_modifier = 0.15
		nobles_influence_modifier = 0.1
		development_cost = -0.1
		max_absolutism = -20
	}
	fixed_rank = 1
	custom_attributes = {
		locked_government_type = yes
		# The other way round from the trading city it used to be. A portion held
		# by a kinsman and kept by his family is the aristocratic case, not the
		# plutocratic one.
		enables_aristocratic_idea_group = yes
		enables_plutocratic_idea_group = no
	}
}
