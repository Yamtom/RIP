# One estate helper, because the same six lines were written eight times.
#
# Every event in this mod that nudges an estate had to guard the call: an
# add_estate_loyalty on a country that does not have that estate is a silent
# no-op at best. So the guard and the call were copied together, over and over.
#
# Parameterised scripted effects are vanilla's own answer to that - see
# add_or_extend_province_modifier_effect in
# common/scripted_effects/00_scripted_effects.txt, which takes $MODIFIER$ and
# $DURATION$ the same way.
#
# Call it as:
#     rip_estate_mood_effect = { ESTATE = estate_magnates LOYALTY = 15 }

rip_estate_mood_effect = {
	if = {
		limit = { has_estate = $ESTATE$ }
		add_estate_loyalty = {
			estate = $ESTATE$
			loyalty = $LOYALTY$
		}
	}
}

# Fired from on_province_culture_converted
# (common/on_actions/west_ukraine_on_actions.txt) while the Chancery's
# Settlement Ordinance is active
# (common/government_mechanics/rip_hlc_austrian_reorganization.txt). Runs in
# province scope and switches to owner before touching estates; the flag it
# reads is set only by that mechanic's own interaction, so no country-tag
# guard is needed here.
hlc_settlement_ordinance_conversion_effect = {
	if = {
		limit = { owner = { has_country_flag = hlc_settlement_ordinance_active } }
		owner = {
			rip_estate_mood_effect = { ESTATE = estate_nobles LOYALTY = 1 }
			rip_estate_mood_effect = { ESTATE = estate_burghers LOYALTY = 1 }
			rip_estate_mood_effect = { ESTATE = estate_clergy LOYALTY = 1 }
			rip_estate_mood_effect = { ESTATE = estate_magnates LOYALTY = 1 }
		}
	}
}
