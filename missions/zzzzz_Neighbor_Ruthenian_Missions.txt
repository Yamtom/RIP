# Additive Ruthenian-culture and Ruthenia-region missions for neighboring base and DLC trees.
# Never replaces vanilla files; loads after vanilla and existing RIP series.
#
# Every series is tied to the last mission of the vanilla column it joins, so each first mission has an arrow and
# no vacuum above it. A parent has to exist in the variant the player sees, hence one series per variant with the
# vanilla potential: Lithuania under Lions of the North (lit_jesuit_academy, row 5) or without it (lit_develop_sluck,
# row 4); the Ottomans of Domination (column 1 of DOM_Ottoman_Missions.txt, ending on
# tur_embrace_the_military_doctrines, row 13) or the older tree that TUR without Domination and RUM use (column 1
# ending on conquer_hungary, row 7). The *_classic series are copies with their own keys, because a mission key
# cannot repeat; tests/check_ruthenian_mission_series.py checks that the copies stay identical.
# Column 1 of Lithuania also carries the Kyiv succession (rip_lit_kyiv_succession, under the princes of Rus) and column 3 the
# Volok survey (rip_lit_volok_survey, under lit_build_manufactories; a root of the free column 3 in the older tree). Both open the
# events of events/RIP_LIT_Reckoning.txt.
# The Tatar and Habsburg series live in zzzz_Ruthenian_Contacts_Horde_Habsburg.txt, which sorts after the vanilla
# horde files; the Dnieper tribute moved there to follow the yarlyks.

rip_lit_ruthenian_missions = {
	slot = 1
	generic = no
	ai = yes
	potential_on_load = {
		has_dlc = "Lions of the North"
	}
	potential = {
		tag = LIT
		has_dlc = "Lions of the North"
		NOT = { map_setup = map_setup_random }
	}

	has_country_shield = yes

	rip_lit_ruthenian_chancery = {
		icon = mission_noble_council
		required_missions = { lit_jesuit_academy }
		position = 6
		provinces_to_highlight = {
			region = ruthenia_region
			rip_diaspora_eligible_culture_province = yes
			OR = {
				NOT = { owned_by = ROOT }
				local_autonomy = 35
			}
		}
		trigger = {
			OR = {
				accepted_culture = ruthenian
				accepted_culture = ruthenian_new
				accepted_culture = ruthenian_west_ruthenian
				accepted_culture = ruthenian_west_slavic
				rip_diaspora_eligible_culture = yes
			}
			num_of_owned_provinces_with = {
				value = 8
				region = ruthenia_region
				rip_diaspora_eligible_culture_province = yes
				NOT = { local_autonomy = 35 }
			}
			OR = {
				adm_tech = 8
				num_of_owned_provinces_with = {
					value = 3
					region = ruthenia_region
					OR = {
						has_building = courthouse
						has_building = town_hall
					}
				}
			}
		}
		effect = {
			add_adm_power = 75
			change_government_reform_progress = 35
			add_country_modifier = {
				name = rip_lit_chancery_statutes
				duration = 9125
			}
		}
	}

	rip_lit_princes_of_rus = {
		icon = mission_eastern_european_city
		required_missions = { rip_lit_ruthenian_chancery }
		position = 7
		provinces_to_highlight = {
			OR = {
				province_id = 280
				AND = {
					region = ruthenia_region
					rip_diaspora_eligible_culture_province = yes
					NOT = { development = 15 }
				}
			}
		}
		trigger = {
			280 = {
				owned_by = ROOT
				development = 20
				rip_diaspora_eligible_culture_province = yes
			}
			num_of_owned_provinces_with = {
				value = 4
				region = ruthenia_region
				rip_diaspora_eligible_culture_province = yes
				development = 15
			}
			stability = 1
		}
		effect = {
			add_prestige = 15
			add_dip_power = 75
			280 = {
				add_base_tax = 1
				add_base_production = 1
				add_base_manpower = 1
			}
			add_country_modifier = {
				name = rip_lit_ruthenian_pillar
				duration = -1
			}
		}
	}

	rip_lit_kyiv_succession = {
		icon = mission_noble_council
		required_missions = { rip_lit_princes_of_rus }
		position = 8
		provinces_to_highlight = {
			province_id = 280
		}
		trigger = {
			is_year = 1471
			280 = { country_or_non_sovereign_subject_holds = ROOT }
			adm_power = 25
		}
		effect = {
			country_event = { id = rip_lit_reckoning.1 }
		}
	}
}

	}

	has_country_shield = yes

	rip_tur_podillian_eyalet = {
		icon = mission_ottoman_diplomacy
		required_missions = { tur_embrace_the_military_doctrines }
		position = 14
		provinces_to_highlight = {
			OR = {
				area = podolia_volhynia_area
				AND = {
					region = ruthenia_region
					rip_diaspora_eligible_culture_province = yes
				}
			}
			NOT = { country_or_non_sovereign_subject_holds = ROOT }
		}
		trigger = {
			OR = {
				AND = {
					281 = { country_or_non_sovereign_subject_holds = ROOT }
					num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
						value = 5
						region = ruthenia_region
						rip_diaspora_eligible_culture_province = yes
					}
				}
				any_subject_country = {
					rip_diaspora_eligible_culture = yes
					num_of_cities = 5
				}
			}
		}
		effect = {
			add_prestige = 15
			add_mil_power = 50
			if = {
				limit = {
					281 = { country_or_non_sovereign_subject_holds = ROOT }
				}
				281 = {
					add_province_modifier = {
						name = rip_tur_kamianets_citadel
						duration = 9125
					}
				}
			}
			add_country_modifier = {
				name = rip_tur_northern_protectorate
				duration = 9125
			}
			podolia_volhynia_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_claim = ROOT }
				}
				add_claim = ROOT
			}
			west_dniepr_area = {
				limit = {
					NOT = { is_core = ROOT }
					NOT = { is_claim = ROOT }
				}
				add_claim = ROOT
			}
		}
	}
}

						region = ruthenia_region
					}
					rip_diaspora_eligible_culture_province = yes
					OR = {
						has_building = temple

			add_adm_power = 50
