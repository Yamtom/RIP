# RIP - vln path state triggers.

west_ukraine_vln_origin_country = {
	OR = {
		tag = VLN
		AND = {
			OR = {
				tag = VOL
				tag = UKR
			}
			has_country_flag = rip_vol_origin_vln
		}
	}
}

west_ukraine_vln_alt_paths = {
	OR = {
		AND = {
			tag = VLN
			exists = HLC
		}
		AND = {
			tag = VOL
			has_country_flag = rip_vol_origin_vln
		}
	}
}

west_ukraine_vln_legacy_missions = {
	OR = {
		west_ukraine_vln_origin_country = yes
		AND = {
			tag = VOL
			OR = {
				mission_completed = VLN_declare_independence
				mission_completed = VLN_unite_galicia
				mission_completed = VLN_integrate_north
				mission_completed = VLN_restore_ruthenia
				mission_completed = VLN_galician_alliance
				mission_completed = VLN_baltic_aspirations
				mission_completed = VLN_constitutional_assembly
				mission_completed = VLN_carpathian_bulwark
				mission_completed = VLN_conquer_crimea
			}
		}
	}
}
