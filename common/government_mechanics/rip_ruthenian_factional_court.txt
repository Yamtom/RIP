# =============================================================================
# rip_ruthenian_factional_court - the ruler's hand on the rada.
#
# The three ruthenian_factional_empire_*_reform ids are one government whose
# rada combines the vanilla three-faction system with the native States General
# scale. The leading faction and the previously dominant faction form the
# coalition; the remaining faction is the opposition. The reform id selects
# that pair, and a single States General bar tracks the coalition's standing.
#
# The faction buttons are direct influence and loyalty levers. The old three
# pairwise powers are retired: they could form a contradictory cycle and let a
# faction keep bonuses after its dominance ended.
#
# The States General is gated by Res Publica or Winds of Change. The native
# faction panel and the buttons remain available without either DLC.
# =============================================================================

ruthenian_factional_court_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	# Plain triggers only: `available` is parsed before the scripted triggers exist (measured in EU4 1.37.5), so kru_holds_rada_reform cannot be used here.
	available = {
		OR = {
			has_reform = ruthenian_factional_empire_reform
			has_reform = ruthenian_factional_empire_princes_hetmans_reform
			has_reform = ruthenian_factional_empire_boyars_hetmans_reform
			has_reform = ruthenian_factional_empire_princes_boyars_reform
			has_reform = ruthenian_factional_empire_hetmans_princes_reform
			has_reform = ruthenian_factional_empire_hetmans_boyars_reform
		}
	}

	interactions = {
		# Court the princes of the blood - honours, precedence, a hearing at
		# every rada. Their standing in the faction rises; the crown's own
		# legitimacy rises with theirs.
		kru_court_the_knyazi = {
			icon = GFX_faction_kru_knyazi
			trigger = {
				dip_power = 50
				NOT = { faction_influence = { faction = kru_knyazi influence = 70 } }
			}
			effect = {
				add_dip_power = -50
				add_faction_influence = {
					faction = kru_knyazi
					influence = 25
				}
				add_legitimacy_equivalent = { amount = 2 republican_tradition = 1 }
				kru_refresh_factional_court_effect = yes
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_knyazi }
					faction_influence = { faction = kru_knyazi influence = 40 }
				}
			}
		}
		# Court the boyar houses and the chartered towns - offices, charters, a
		# share of the customs. Their standing rises; the treasury takes it in.
		kru_court_the_boyary = {
			icon = GFX_faction_kru_boyary
			trigger = {
				adm_power = 50
				NOT = { faction_influence = { faction = kru_boyary influence = 70 } }
			}
			effect = {
				add_adm_power = -50
				add_faction_influence = {
					faction = kru_boyary
					influence = 25
				}
				add_treasury = 50
				kru_refresh_factional_court_effect = yes
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_boyary }
					faction_influence = { faction = kru_boyary influence = 40 }
				}
			}
		}
		# Court the Cossack starshyna - regimental appointments, land for the
		# officers, the register widened. Their standing rises; the host is
		# readier for it.
		kru_court_the_starshyna = {
			icon = GFX_faction_kru_starshyna
			trigger = {
				mil_power = 50
				NOT = { faction_influence = { faction = kru_starshyna influence = 70 } }
			}
			effect = {
				add_mil_power = -50
				add_faction_influence = {
					faction = kru_starshyna
					influence = 25
				}
				add_army_tradition = 5
				kru_refresh_factional_court_effect = yes
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 8
					is_at_war = yes
				}
				modifier = {
					factor = 6
					NOT = { faction_in_power = kru_starshyna }
					faction_influence = { faction = kru_starshyna influence = 40 }
				}
			}
		}
	}
}
