
convert_to_greek_catholic_effect = {
	change_religion = greek_catholic
	add_province_modifier = {
		name = greek_catholic_diocese
		duration = 5475
	}
	if = {
		limit = { has_building = temple }
		add_base_tax = 1
	}
	if = {
		limit = { has_building = cathedral }
		add_base_tax = 2
	}
}

establish_greek_catholic_center_effect = {
	change_religion = greek_catholic
	add_province_modifier = {
		name = greek_catholic_center
		duration = -1
	}
	add_base_tax = 2
	add_base_production = 1
	add_base_manpower = 1
	if = {
		limit = {
			NOT = { has_building = temple }
			NOT = { has_building = cathedral }
		}
		add_building = temple
	}
}

spread_greek_catholic_to_neighbors = {
	every_neighbor_province = {
		limit = {
			OR = {
				religion = orthodox
				religion = catholic
			}
			owned_by = ROOT
		}
		random = {
			chance = 30
			change_religion = greek_catholic
			add_province_modifier = {
				name = greek_catholic_diocese
				duration = 3650
			}
		}
	}
}

# ---------------------------------------------------------------- the Curia
#
# Rome as the union actually met it: a court you have standing at, not a college you
# sit in. rip_uc_standing runs 0 to 100. It decays, because a court that hears
# nothing from a province forgets it, and it is raised by sending money - which is
# what the Ruthenian sees did, and complained about doing, for two centuries.

rip_uc_standing_tick_effect = {
	# The legate resident at the court keeps the see's name in front of it, and
	# that is worth authority at home as much as at Rome. This is the only
	# thing left of the old rip_uc_standing counter: the counter, its decay and
	# its five band modifiers are gone, because the faith has a real bar.
	if = {
		limit = {
			religion = greek_catholic
			has_country_modifier = papal_legate
		}
		add_patriarch_authority = 0.02
	}
}

# Every band off, and none back on.
rip_uc_clear_the_bands_effect = {
	hidden_effect = {
		remove_country_modifier = rip_uc_curia_unknown
		remove_country_modifier = rip_uc_curia_heard
		remove_country_modifier = rip_uc_curia_favoured
		remove_country_modifier = rip_uc_curia_protected
	}
}

# The four rip_uc_curia_* band modifiers stood here, switched by the value of
# rip_uc_standing to fake a progress bar out of tooltips. The faith has a real
# one - patriarch authority, from has_patriarchs - so the bands are retired and
# this effect only sheds them from saves that still carry one.
rip_uc_show_the_standing_effect = {
	rip_uc_clear_the_bands_effect = yes
}

# A donation. The see pays, the court remembers, and it will not be asked twice in
# the same three years.
rip_uc_donation_effect = {
	add_treasury = -150
	# money sent to Rome comes back as standing the metropolitan can spend at
	# home; the engine caps patriarch authority at 1.0 by itself
	add_patriarch_authority = 0.12
	add_country_modifier = {
		name = rip_uc_donation_recently_sent
		duration = 1095
	}
	rip_uc_show_the_standing_effect = yes
}

# A petition spends the standing it needs. The court does a favour once and then
# expects to be reminded who asked for it.
# A petition is not free and it is not paid in coin alone. What it costs is
# the metropolitan's own authority: every letter to Rome is a letter that says
# the see could not settle the matter itself, and the parishes hear about it.
rip_uc_spend_the_standing_effect = {
	add_patriarch_authority = -0.15
}

# ==========================================================================
# STANDING WITH ROME - THE LOOP, CLOSED
#
#   religious unity -> the union spreads -> standing at Rome -> petitions
#
# The union church never sat in the college of cardinals and never voted on
# the throne of Peter, so it has no papal influence and no curia. What its
# metropolitans really had was standing: a court that would read their letters
# because they arrived with something behind them.
#
# What is behind them is the union itself. Every parish that comes over is
# evidence the union works, and evidence is what a petition is argued from -
# so rip_gc_unity_spread_effect calls this on every conversion, and the loop
# is closed: unity buys parishes, parishes buy standing, standing buys
# petitions, and petitions spend it back down.
#
# It is carried on patriarch authority rather than on a variable of its own,
# and that is deliberate. greek_catholic declares has_patriarchs, so the bar
# already exists, is already drawn with a number on it, and already prices the
# icons and province modifiers of this rite. A separate rip_uc_standing
# counter would need its own bar, EU4 gives no way to draw one, and the mod
# has twice tried to fake it out of tooltip modifiers. It is the same
# quantity: what the metropolitan can spend.
# ==========================================================================
rip_uc_gain_standing_effect = {
	add_patriarch_authority = 0.02
}
