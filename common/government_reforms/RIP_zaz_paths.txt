# The four roads out of the Sich.
#
# Zaporizhzhia starts as a host and does not have to stay one. Each of these is
# a tier-1 government type, taken through a decision in
# decisions/ZaporizhiaPaths.txt, and each closes the other three: the flag
# zaz_path_chosen is what makes them exclusive.
#
# Every one is gated by potential, or EU4 lists it in the reform slot of every
# country in the game and merely greys it out. Every one is registered in
# common/governments/00_governments.txt, or it is never offered at all.

# --- 1. Cossackdom. The Swiss road: cantons that sell soldiers and defend
#        their own passes, and conquer nothing.
zaz_cossack_cantons_reform = {
	icon = "fist_in_air"
	legacy_equivalent = united_cantons_legacy
	potential = {
		zaz_path_cantons_visible = yes
	}
	trigger = {
		zaz_path_cantons_visible = yes
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	lock_level_when_selected = yes
	republic = yes
	republican_name = yes
	has_term_election = yes
	duration = 6
	max_states = 100
	government_abilities = {
		cossacks_mechanic
	}
	modifiers = {
		defensiveness = 0.25
		merc_maintenance_modifier = -0.25
		mercenary_manpower = 0.25
		land_morale = 0.1
		global_unrest = -2
		province_warscore_cost = 0.25
		diplomatic_reputation = 1
		max_absolutism = -20
	}
	custom_attributes = {
		locked_government_type = yes
		enables_aristocratic_idea_group = yes
	}
}
