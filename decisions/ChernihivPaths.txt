	# Three roads out of the Siversk Veche, one choice. Each sets
	# chr_path_chosen through chr_take_the_road_effect, which shuts the other
	# two, and each hands over a tier-1 government type that locks its level.
	# The second stage of a road is a statute; the third waits on Great
	# Chernihiv, in common/scripted_triggers/chr/chr_path_triggers.txt.

	# ======================= ROAD A - THE TOWNS =======================
	# Each road's potential now also asks for chr_ready_<reform>, set by
	# common/scripted_effects/chr_path_progress_effects.txt once the matching
	# chr_history capacity crosses its threshold. The old numeric conditions
	# stay in allow unchanged - a country still has to qualify - but it no
	# longer sees the choice before the history behind it exists.

	chr_road_of_the_towns = {
		major = yes
		potential = {
			chr_at_the_crossroads = yes
			NOT = { has_country_flag = chr_road_towns_open }
			has_country_flag = chr_ready_desna_staple
		}
		allow = {
			chr_can_take_towns_road = yes
		}
		effect = {
			set_country_flag = chr_road_towns_open
			# A road that starts in a republic has to start in a republic.
			if = {
				limit = { NOT = { government = republic } }
				change_government = republic
			}
			add_government_reform = chr_desna_staple_reform
			chr_take_the_road_effect = yes
			chr_charter_the_staple_effect = yes
			add_republican_tradition = 10
			country_event = { id = chr_paths.1 days = 3 }
		}
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 3
				republican_tradition = 70
			}
			modifier = {
				factor = 0
				num_of_cities = 12
			}
		}
	}

	# ======================= BORDER COMPACTS =======================
	chr_border_compacts = {
		major = yes
		potential = {
			tag = CHR
		}
		allow = {
			dip_power = 40
			is_at_war = no
			NOT = { has_country_modifier = chr_border_compact_cooldown }
			NOT = { has_country_modifier = chr_border_compact_betrayal }
			any_country = {
				OR = {
					tag = PLC
					tag = POL
					tag = LIT
					tag = MOS
					tag = RUS
				}
				is_at_war = no
				NOT = { war_with = ROOT }
				has_truce = no
				OR = {
					has_guaranteed = ROOT
					ROOT = { alliance_with = PREV }
				}
			}
		}
		effect = {
			add_dip_power = -40
			add_country_modifier = {
				name = chr_border_compact_cooldown
				duration = 1825
			}
			random_country = {
				limit = {
					OR = {
						tag = PLC
						tag = POL
						tag = LIT
						tag = MOS
						tag = RUS
					}
					is_at_war = no
					NOT = { war_with = ROOT }
					has_truce = no
					OR = {
						has_guaranteed = ROOT
						ROOT = { alliance_with = PREV }
					}
				}
				save_event_target_as = chr_border_compact_partner
				add_opinion = {
					who = ROOT
					modifier = chr_border_compact_support
				}
				if = {
					limit = {
						has_opinion = { who = ROOT value = 80 }
						trust = { who = ROOT value = 50 }
					}
					custom_tooltip = chr_border_compacts_confidence_tt
					add_country_modifier = {
						name = chr_border_compact_confidence_partner
						duration = 1825
					}
					ROOT = {
						add_country_modifier = {
							name = chr_border_compact_confidence
							duration = 1825
						}
					}
				}
				else = {
					custom_tooltip = chr_border_compacts_tension_tt
					add_country_modifier = {
						name = chr_border_compact_watchposts
						duration = 1095
					}
					ROOT = {
						add_country_modifier = {
							name = chr_border_compact_tension
							duration = 1095
						}
					}
				}
			}
			hidden_effect = {
				set_country_flag = chr_border_compacts_pending_review
				country_event = {
					id = chr_flavor.36
					days = 540
				}
			}
		}
		ai_will_do = {
			factor = 0
		}
	}
