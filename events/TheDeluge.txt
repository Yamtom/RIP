# ========================================================================
# The Swedish war of 1655-1660.
# Authenticity: A - documented. Follows recorded events, people and dates.
# Sources:
#   Ujscie, the Union of Kedainiai, the Radnot partition compact, Jasna Gora,
#   the Lviv Oath and the Peace of Oliva.
#   The Cossack choice in deluge.3 is the one moment three crowns bid for the
#   Host at once; Hadiach three years later follows from the Commonwealth's
#   weakness here.
# ========================================================================

namespace = deluge

# ===========================================================================
# THE DELUGE, 1655-1660
# ---------------------------------------------------------------------------
# The Swedish war is the hinge of the Cossack century. It broke the
# Commonwealth's field armies, put Lithuania under Swedish protection at
# Kedainiai, and gave Khmelnytsky the one moment when three crowns bid for him
# at once. Vyhovsky's Hadiach settlement three years later is unintelligible
# without it - Poland only offered a Ruthenian duchy because Poland was on its
# knees.
#
# Flags for other files to read:
#   deluge_underway          - the Commonwealth is collapsing
#   deluge_swedish_league    - this country joined the northern coalition
#   deluge_held_with_tsar    - this country kept the Pereiaslav bargain
#   deluge_commonwealth_held - the Commonwealth survived without partition
# ===========================================================================


# --- 1. The northern storm --------------------------------------------------
country_event = {
	id = deluge.1
	title = deluge.1.t
	desc = deluge.1.d
	picture = BATTLE_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = POL
			tag = PLC
			tag = LIT
		}
		is_year = 1655
		NOT = { is_year = 1665 }
		NOT = { has_country_flag = deluge_underway }
		OR = {
			war_with = SWE
			AND = {
				SWE = { exists = yes }
				is_at_war = yes
			}
		}
	}

	mean_time_to_happen = {
		months = 12
		modifier = {
			factor = 0.5
			war_with = SWE
		}
		modifier = {
			factor = 0.6
			war_with = RUS
		}
	}

	option = {		# Fight on every front
		name = deluge.1.a
		set_country_flag = deluge_underway
		add_country_modifier = {
			name = deluge_the_flood
			duration = 3650
		}
		add_war_exhaustion = 2
		add_estate_loyalty = { estate = estate_nobles loyalty = -10 }
	}

	option = {		# Buy off one enemy at any price
		name = deluge.1.b
		set_country_flag = deluge_underway
		add_country_modifier = {
			name = deluge_the_flood
			duration = 3650
		}
		add_treasury = -200
		add_prestige = -15
		add_stability_or_adm_power = yes
	}
}


# --- 2. The Union of Kedainiai ----------------------------------------------
country_event = {
	id = deluge.2
	title = deluge.2.t
	desc = deluge.2.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = LIT
			tag = PLC
		}
		has_country_flag = deluge_underway
		NOT = { is_year = 1662 }
		SWE = { exists = yes }
	}

	mean_time_to_happen = { months = 8 }

	option = {		# Let the magnates treat with Stockholm
		name = deluge.2.a
		add_country_modifier = {
			name = deluge_kedainiai
			duration = 5475
		}
		add_estate_loyalty = { estate = estate_nobles loyalty = 15 }
		add_legitimacy = -10
		add_prestige = -10
	}

	option = {		# Attaint the Radziwills
		name = deluge.2.b
		add_legitimacy = 10
		add_estate_loyalty = { estate = estate_nobles loyalty = -20 }
		add_country_modifier = {
			name = deluge_magnate_attainder
			duration = 3650
		}
	}
}


# --- 3. Three crowns bid for the Host ---------------------------------------
country_event = {
	id = deluge.3
	title = deluge.3.t
	desc = deluge.3.d
	picture = DIPLOMACY_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = ZAZ
			tag = HET
			tag = CHR
			tag = KIE
			tag = UKR
			tag = KRU
		}
		is_year = 1655
		NOT = { is_year = 1663 }
		NOT = { has_country_flag = deluge_choice_made }
		OR = {
			SWE = { exists = yes }
			any_neighbor_country = { has_country_flag = deluge_underway }
			POL = { has_country_flag = deluge_underway }
			PLC = { has_country_flag = deluge_underway }
		}
	}

	mean_time_to_happen = {
		months = 14
		modifier = {
			factor = 0.6
			OR = {
				POL = { is_at_war = yes }
				PLC = { is_at_war = yes }
			}
		}
	}

	option = {		# Take the Swedish league
		name = deluge.3.a
		set_country_flag = deluge_choice_made
		set_country_flag = deluge_swedish_league
		add_country_modifier = {
			name = deluge_northern_league
			duration = 5475
		}
		add_prestige = 10
		add_papal_influence = -20
		if = {
			limit = { SWE = { exists = yes } }
			add_opinion = { who = SWE modifier = deluge_league_partner }
		}
		if = {
			limit = { RUS = { exists = yes } }
			add_opinion = { who = RUS modifier = deluge_broke_faith }
		}
	}

	option = {		# Keep the bargain with the Tsar
		name = deluge.3.b
		set_country_flag = deluge_choice_made
		set_country_flag = deluge_held_with_tsar
		add_country_modifier = {
			name = deluge_tsars_shadow
			duration = 5475
		}
		add_legitimacy = 10
		if = {
			limit = { RUS = { exists = yes } }
			add_opinion = { who = RUS modifier = deluge_kept_faith }
		}
	}

	option = {		# Treat with all and bind to none
		name = deluge.3.c
		set_country_flag = deluge_choice_made
		add_country_modifier = {
			name = deluge_hetmans_double_game
			duration = 3650
		}
		add_dip_power = 50
		add_prestige = -5
	}
}

# --- 5. The tide turns ------------------------------------------------------
country_event = {
	id = deluge.5
	title = deluge.5.t
	desc = deluge.5.d
	picture = RELIGIOUS_TURMOIL_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = POL
			tag = PLC
		}
		has_country_flag = deluge_underway
		is_year = 1656
		NOT = { is_year = 1666 }
	}

	mean_time_to_happen = {
		months = 16
		modifier = {
			factor = 0.6
			religious_unity = 0.9
		}
	}

	option = {		# Swear the Lviv Oath
		name = deluge.5.a
		add_country_modifier = {
			name = deluge_lviv_oath
			duration = 7300
		}
		add_papal_influence = 30
		add_prestige = 15
		add_estate_loyalty = { estate = estate_church loyalty = 15 }
		# The oath promised relief to the peasantry and named the Catholic
		# crown as the realm's defender - both read badly in the east.
		hidden_effect = {
			every_owned_province = {
				limit = { religion = orthodox }
				add_province_modifier = {
					name = deluge_eastern_misgivings
					duration = 5475
				}
			}
		}
	}

	option = {		# Win it with soldiers, not vows
		name = deluge.5.b
		add_army_tradition = 10
		add_country_modifier = {
			name = deluge_soldiers_peace
			duration = 5475
		}
		add_treasury = -150
	}
}
