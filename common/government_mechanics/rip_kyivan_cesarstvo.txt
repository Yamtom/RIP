kyivan_peoples_gathering_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1
	available = { has_reform = kyivan_cesarstvo_reform }
	powers = {
		kyivan_peoples_gathering = {
			gui = rip_peoples_gathering_power
			max = 100
			min = 0
			reset_on_new_ruler = no
			base_monthly_growth = 0
			is_good = yes
			range_modifier = {
				trigger = { rip_cesar_integration_eligible = yes }
				modifier = { monthly_kyivan_peoples_gathering = 0.3 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { rip_cesar_integration_eligible = yes num_accepted_cultures = 3 }
				modifier = { monthly_kyivan_peoples_gathering = 0.15 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { rip_cesar_integration_eligible = yes check_variable = { which = rip_cesar_minority_share value = 0.3 } }
				modifier = { monthly_kyivan_peoples_gathering = 0.15 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { rip_cesar_integration_eligible = yes has_government_power = { mechanic_type = kyivan_throne_concord_mechanic power_type = kyivan_throne_concord value = 60 } }
				modifier = { monthly_kyivan_peoples_gathering = 0.1 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { check_variable = { which = rip_cesar_excluded_share value = 0.15 } }
				modifier = { monthly_kyivan_peoples_gathering = -0.25 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { check_variable = { which = rip_cesar_excluded_share value = 0.3 } }
				modifier = { monthly_kyivan_peoples_gathering = -0.35 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { NOT = { has_government_power = { mechanic_type = kyivan_throne_concord_mechanic power_type = kyivan_throne_concord value = 30 } } }
				modifier = { monthly_kyivan_peoples_gathering = -0.2 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { war_exhaustion = 5 }
				modifier = { monthly_kyivan_peoples_gathering = -0.15 }
				start_value = 0
				end_value = 100
			}
			range_modifier = {
				trigger = { NOT = { rip_cesar_integration_eligible = yes } }
				modifier = { monthly_kyivan_peoples_gathering = -0.5 }
				start_value = 0
				end_value = 100
			}
			on_max_reached = {
				# Replace legacy permanent reward and renew a ten-year compact.
				remove_country_modifier = kyivan_peoples_gathered
				add_country_modifier = { name = kyivan_peoples_gathered duration = 3650 }
				add_prestige = 10
				add_government_power = { mechanic_type = kyivan_throne_concord_mechanic power_type = kyivan_throne_concord value = 10 }
				set_government_power = { mechanic_type = kyivan_peoples_gathering_mechanic power_type = kyivan_peoples_gathering value = 40 }
			}
		}
	}
}
