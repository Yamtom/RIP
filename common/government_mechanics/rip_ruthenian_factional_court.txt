# =============================================================================
# rip_ruthenian_factional_court - the player-facing half of the KRU "factional
# empire". The three ruthenian_factional_empire_*_reform ids are ONE government
# that swaps itself as the dominant faction pair changes:
#
#   ruthenian_factional_empire_reform                 = Knyazi + Boyary
#   ruthenian_factional_empire_princes_hetmans_reform = Knyazi + Starshyna
#   ruthenian_factional_empire_boyars_hetmans_reform  = Boyary + Starshyna
#
# The pair is decided by three influence tracks (kru_knyazi_influence,
# kru_boyary_influence, kru_hetmany_influence - the last keeps its saved
# variable id although the faction is now localised "Starshyna"), recomputed on
# a hidden pulse from the country's own state
# (common/scripted_effects/kru_effects.txt : kru_recalculate_faction_influence_
# effect / kru_update_dominant_pair_effect).
#
# What was missing: the player had no lever on it. With Res Publica the shift
# was fully automatic; without Res Publica only the AI got the fallback court
# event. This mechanic gives every holder of the reform three buttons to court
# a faction directly - each raises that track for a decade and pays a small
# themed dividend. Buttons-only (devshirme pattern): no powers, no .gui, monarch
# power spent inside the effect, vanilla ability sprites for the icons.
# =============================================================================

ruthenian_factional_court_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 1

	available = {
		OR = {
			has_reform = ruthenian_factional_empire_reform
			has_reform = ruthenian_factional_empire_princes_hetmans_reform
			has_reform = ruthenian_factional_empire_boyars_hetmans_reform
		}
	}

	interactions = {
		# Court the princes of the blood. Legitimacy and standing follow; the
		# Knyazi track climbs for a decade.
		kru_court_the_knyazi = {
			icon = GFX_cossacks_ability_DIP
			trigger = {
				dip_power = 50
				NOT = { has_country_modifier = KRU_courting_knyazi }
			}
			effect = {
				add_dip_power = -50
				add_country_modifier = {
					name = KRU_courting_knyazi
					duration = 3650
				}
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					check_variable = { which = kru_knyazi_influence value = 55 }
					NOT = { check_variable = { which = kru_knyazi_influence value = 70 } }
				}
			}
		}
		# Court the boyar houses and the chartered towns. The treasury and the
		# ledgers steady; the Boyary track climbs.
		kru_court_the_boyary = {
			icon = GFX_cossacks_ability_ADM
			trigger = {
				adm_power = 50
				NOT = { has_country_modifier = KRU_courting_boyary }
			}
			effect = {
				add_adm_power = -50
				add_country_modifier = {
					name = KRU_courting_boyary
					duration = 3650
				}
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 6
					check_variable = { which = kru_boyary_influence value = 55 }
					NOT = { check_variable = { which = kru_boyary_influence value = 70 } }
				}
			}
		}
		# Court the Cossack starshyna. The host answers more readily; the
		# Starshyna track climbs.
		kru_court_the_starshyna = {
			icon = GFX_cossacks_ability_MIL
			trigger = {
				mil_power = 50
				NOT = { has_country_modifier = KRU_courting_starshyna }
			}
			effect = {
				add_mil_power = -50
				add_country_modifier = {
					name = KRU_courting_starshyna
					duration = 3650
				}
			}
			cooldown_years = 6
			ai_chance = {
				factor = 1
				modifier = {
					factor = 8
					is_at_war = yes
				}
				modifier = {
					factor = 6
					check_variable = { which = kru_hetmany_influence value = 55 }
					NOT = { check_variable = { which = kru_hetmany_influence value = 70 } }
				}
			}
		}
	}
}
