# ========================================================================
# The Hadiach compact, and the ratification that killed it.
# Authenticity: A - documented, with the outcome left open.
# Sources:
#   the articles agreed at Hadiach in 1658 between Vyhovsky and the Crown -
#   a Grand Duchy of Ruthenia standing beside Poland and Lithuania, with its
#   own chancery, courts, treasury, army and seat at the diet; the Sejm's
#   ratification of a thinner text in 1659, which struck the Duchy's army
#   down to a token and left the Uniate question open; and the war that
#   resumed anyway.
#
# WHY THIS IS SPLIT ACROSS THREE TOOLS, since the author asked:
#   - the OFFER is a diplomatic action (common/new_diplomatic_actions/
#     RIP_PLC_diplomatic_actions.txt). Only a diplomatic action can model a
#     proposal that the other side is free to refuse; a decision would let
#     the Crown impose it, and a mission would make it a milestone rather
#     than a bargain.
#   - the RATIFICATION is these events. It is where the thing actually died,
#     and it needs both courts to answer in sequence, which is what events
#     are for.
#   - nothing here is a mission, because a mission cannot be refused either.
# ========================================================================

namespace = hadiach

# --- 1. The articles go to the Sejm.
country_event = {
	id = hadiach.1
	title = hadiach.1.t
	desc = hadiach.1.d
	picture = SEJM_eventPicture

	is_triggered_only = yes

	# The Crown puts the text before the diet. It does not decide what comes back
	# out of it - that is hadiach.4, and it is the whole reason this chain exists.
	option = {
		name = hadiach.1.a
		ai_chance = { factor = 40 }
		add_prestige = -5
		rip_estate_mood_effect = { ESTATE = estate_magnates LOYALTY = -20 }
		country_event = { id = hadiach.4 days = 30 }
	}

	option = {
		name = hadiach.1.b
		ai_chance = { factor = 45 }
		add_absolutism = -5
		add_country_modifier = {
			name = rip_hadiach_thinned
			duration = 9125
		}
		rip_estate_mood_effect = { ESTATE = estate_magnates LOYALTY = 5 }
		if = {
			limit = { has_saved_event_target = hadiach_ruthenia }
			event_target:hadiach_ruthenia = {
				country_event = { id = hadiach.2 days = 10 }
			}
		}
	}

	option = {
		name = hadiach.1.c
		ai_chance = { factor = 15 }
		add_prestige = -10
		if = {
			limit = { has_saved_event_target = hadiach_ruthenia }
			event_target:hadiach_ruthenia = {
				# The princedom was created by the diplomatic action as the price of
				# the articles. Withdrawing the articles and keeping the vassal was
				# the one outcome the old chain allowed and history did not.
				if = {
					limit = { is_subject_of = ROOT }
					grant_independence = yes
				}
				add_opinion = {
					who = ROOT
					modifier = lit_union_grievance
				}
				add_casus_belli = {
					target = ROOT
					type = cb_disloyal_vassal
					months = 300
				}
				country_event = { id = hadiach.2 days = 10 }
			}
		}
	}
}

# --- 2. What came back from the Sejm.
country_event = {
	id = hadiach.2
	title = hadiach.2.t
	desc = hadiach.2.d
	picture = ANGRY_MOB_eventPicture

	is_triggered_only = yes

	option = {
		name = hadiach.2.a
		ai_chance = { factor = 45 }
		add_country_modifier = {
			name = rip_hadiach_thinned
			duration = 9125
		}
		add_liberty_desire = 20
		every_owned_province = {
			limit = { is_state_core = ROOT }
			add_unrest = 3
		}
	}

	option = {
		name = hadiach.2.b
		ai_chance = { factor = 55 }
		add_prestige = 10
		rip_estate_mood_effect = { ESTATE = estate_cossacks LOYALTY = 15 }
		# Repudiating the compact and staying a subject of the crown that thinned
		# it is not a thing the Host did. Vyhovsky broke with Warsaw and the war
		# resumed; so does this.
		if = {
			limit = { is_subject = yes }
			overlord = {
				add_opinion = {
					who = ROOT
					modifier = lit_union_grievance
				}
				save_event_target_as = hadiach_the_crown
			}
			grant_independence = yes
			add_casus_belli = {
				target = event_target:hadiach_the_crown
				type = cb_disloyal_vassal
				months = 300
			}
		}
	}
}

# --- 3. The compact stands.
country_event = {
	id = hadiach.3
	title = hadiach.3.t
	desc = hadiach.3.d
	picture = COURT_eventPicture

	is_triggered_only = yes

	option = {
		name = hadiach.3.a
		ai_chance = { factor = 100 }
		add_prestige = 15
		add_legitimacy = 10
		add_liberty_desire = -25
		set_country_flag = rip_hadiach_ratified
	}
}

# --- 4. The Sejm answers.
#
# Historically it did not ratify what was signed. The articles agreed in September
# 1658 gave the Duchy a chancery, courts, a treasury, an army of thirty thousand
# and a seat at the diet; what the Sejm passed in May 1659 struck the army down to
# a token, left the Union standing and cut the nobilitations to a hundred once.
# The Host read that as the bargain being broken, and it was.
#
# So the full text passes only for a Crown that has actually built the ground for
# it - magnates it has kept loyal and has not let grow too strong, and a realm
# steady enough to carry a constitutional change. Otherwise the diet does what the
# diet did.
country_event = {
	id = hadiach.4
	title = hadiach.4.t
	desc = hadiach.4.d
	picture = SEJM_eventPicture

	is_triggered_only = yes
	hidden = yes

	immediate = {
		random_list = {
			# ratified as signed
			15 = {
				modifier = {
					factor = 3
					estate_loyalty = { estate = estate_magnates loyalty = 60 }
				}
				modifier = {
					factor = 2
					NOT = { estate_influence = { estate = estate_magnates influence = 60 } }
				}
				modifier = {
					factor = 2
					stability = 2
				}
				modifier = {
					factor = 0.25
					estate_influence = { estate = estate_magnates influence = 80 }
				}
				country_event = { id = hadiach.5 days = 1 }
			}
			# ratified thin, which is what happened
			85 = {
				modifier = {
					factor = 2
					estate_influence = { estate = estate_magnates influence = 60 }
				}
				modifier = {
					factor = 0.5
					estate_loyalty = { estate = estate_magnates loyalty = 70 }
				}
				country_event = { id = hadiach.6 days = 1 }
			}
		}
	}

	option = {
		name = hadiach.4.a
	}
}

# --- 5. The diet carried it.
country_event = {
	id = hadiach.5
	title = hadiach.5.t
	desc = hadiach.5.d
	picture = COURT_eventPicture

	is_triggered_only = yes

	option = {
		name = hadiach.5.a
		add_country_modifier = {
			name = rip_hadiach_compact
			duration = -1
		}
		set_country_flag = rip_hadiach_ratified
		if = {
			limit = { has_saved_event_target = hadiach_ruthenia }
			event_target:hadiach_ruthenia = {
				add_country_modifier = {
					name = rip_hadiach_compact
					duration = -1
				}
				country_event = { id = hadiach.3 days = 10 }
			}
		}
	}
}
