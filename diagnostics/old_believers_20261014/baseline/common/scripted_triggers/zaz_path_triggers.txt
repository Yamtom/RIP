# Visibility gates for the four roads out of the Sich.
#
# Each reform is shown to a country that either already holds it - otherwise a
# country loses sight of its own government - or is still at the crossroads
# with that road's flag set. zaz_path_chosen closes the other three.

# A later tag change does not erase the political lineage of the Sich. This is
# deliberately narrower than is_cossack_polity: the Last Sich is a Zaporozhian
# survival path, not a generic reward for every Cossack host.
zaz_lineage_country = {
	OR = {
		tag = ZAZ
		was_tag = ZAZ
		has_country_flag = rip_hetmanate_origin_zaz
	}
}

zaz_at_the_crossroads = {
	OR = {
		tag = ZAZ
		AND = {
			tag = HET
			has_reform = rip_cossacks_reform
		}
	}
	NOT = { has_country_flag = zaz_path_chosen }
}

zaz_path_cantons_visible = {
	OR = {
		has_reform = zaz_cossack_cantons_reform
		AND = {
			zaz_at_the_crossroads = yes
			has_country_flag = zaz_path_cantons_open
		}
	}
}

zaz_path_host_state_visible = {
	OR = {
		has_reform = zaz_host_and_state_reform
		AND = {
			zaz_at_the_crossroads = yes
			has_country_flag = zaz_path_host_state_open
		}
	}
}

zaz_path_sacred_order_visible = {
	OR = {
		has_reform = zaz_sacred_host_order_reform
		AND = {
			zaz_at_the_crossroads = yes
			has_country_flag = zaz_path_sacred_order_open
		}
	}
}

zaz_path_sacred_horde_visible = {
	OR = {
		has_reform = zaz_sacred_horde_reform
		AND = {
			zaz_at_the_crossroads = yes
			has_country_flag = zaz_path_sacred_horde_open
		}
	}
}

# --- what each road asks of the Host before it will open
#
# Replaced 2026-09: these used to be an instant numeric snapshot
# (num_of_cities, republican_tradition and the like), which meant all four
# roads sat visible from the first day at the crossroads, waiting on stats a
# player could just watch tick over. The argument is not settled by a
# census. events/ZaporizhianCrossroadsHistory.txt now runs a decades-long
# chain of events (namespace zaz_crossroads, no rarer than every 45 months
# per candidate road) that nudges one hidden lean variable per road
# (zaz_lean_cantons and its three siblings, in
# common/scripted_effects/zaz_crossroads_effects.txt) and sets the matching
# zaz_path_*_ascendant flag the moment a lean crosses its threshold. That
# flag is the only thing these four triggers check now; nothing else in the
# mod reads zaz_path_*_ascendant, so this is a safe, self-contained swap.

zaz_can_take_cantons_road = {
	zaz_at_the_crossroads = yes
	has_country_flag = zaz_path_cantons_ascendant
}

zaz_can_take_host_state_road = {
	zaz_at_the_crossroads = yes
	has_country_flag = zaz_path_host_state_ascendant
}

zaz_can_take_sacred_order_road = {
	zaz_at_the_crossroads = yes
	has_country_flag = zaz_path_sacred_order_ascendant
}

zaz_can_take_sacred_horde_road = {
	zaz_at_the_crossroads = yes
	has_country_flag = zaz_path_sacred_horde_ascendant
}

# --- a Cossack polity that can pick its seat up and carry it

is_cossack_polity = {
	OR = {
		has_reform = rip_cossacks_reform
		has_reform = zaz_sich_brotherhood_reform
		has_reform = zaz_kosh_elections_reform
		has_reform = zaz_free_host_reform
		has_reform = zaz_chaiky_trade_reform
		has_reform = zaz_general_rada_reform
		has_reform = zaz_last_sich_reform
		has_reform = het_starshyna_oligarchy_reform
		has_reform = zaz_cossack_cantons_reform
		has_reform = zaz_host_and_state_reform
		has_reform = zaz_sacred_host_order_reform
		has_reform = zaz_sacred_horde_reform
		has_reform = cossacks_reform
		has_reform = cossacks_legacy_reform
	}
}

can_carry_the_host = {
	is_cossack_polity = yes
	NOT = { has_country_flag = host_recently_moved }
	is_at_war = no
	stability = 0
}
