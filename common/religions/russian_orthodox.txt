christian = {
	russian_orthodox = {
		color = { 204 51 51 }
		icon = 30

		# Scope kept at vanilla parity: every vanilla religion has 2-3 country
		# modifiers, and none grants administrative_efficiency, core_creation,
		# culture_conversion_cost, missionaries or legitimacy. The zeal against
		# heretics is what sets the state church apart, so that is what it keeps.
		country = {
			tolerance_own = 2					# vanilla ceiling (inti)
			tolerance_heretic = -2
			global_heretic_missionary_strength = 0.02	# matches hussite
		}

		country_as_secondary = {
			tolerance_own = 1
			global_colonial_growth = 10
		}

		province = {
			local_missionary_strength = -0.01	# as vanilla orthodox
		}

		# ONE PANEL, AND WHERE THE REST OF IT LIVES.
		#
		# Every faith mechanic in vanilla draws into one rectangle of
		# interface/countryreligionview.gui, roughly y 231..320. I tried moving
		# them apart and it was wrong: `province_listbox` sits at y=402 and is
		# 530x450, so the band below the panel is not free at all - the holy-site
		# rows landed on top of the province list in game. That override is
		# reverted and this mod ships no countryreligionview.gui.
		#
		# EU4 has no scripted GUI either - there is no common/scripted_guis, the
		# eleven *_specific_window names are hardcoded in the executable, and no
		# vanilla .gui carries a trigger or visible block. So the panel is what
		# vanilla gives and no more.
		#
		# What that leaves, and it is enough:
		#
		#   orthodox_specific_window - patriarch authority as the scale, the
		#     standing icon, and the SELECT button. That button opens
		#     icon_selection_screen, which is a SEPARATE window with its own
		#     listbox - the picker, exactly like the blessings picker.
		#   coptic_specific_window   - the five holy sites, and the popup they
		#     open (interface/blessings.gui, untouched vanilla).
		#   fervor_specific_window   - the Sobor.
		#
		# These three do overlap in the tab. That is the cost of running three
		# mechanics on one faith and it cannot be fixed from script; cutting the
		# blessings below removes the worst of the clutter.

		# Patriarchs
		has_patriarchs = yes

		# The icons of the Muscovite church.
		#
		# The frame is chosen by POSITION in this block, not by name - vanilla
		# carries no sprite field and the save stores a bare index - so these
		# five keep vanilla's five paintings under our own keys. Do not reorder
		# them or the pictures shuffle. Payloads are carried over unchanged from
		# the decisions they replace, so nothing is lost in the move.
		orthodox_icons = {
			rip_ro_icon_michael = {
				discipline = 0.05
				manpower_recovery_speed = 0.1
				core_creation = -0.15	# from blessing_of_orthodox_conquest

				# HOLY SITES UNLOCK THIS ICON.
				# The Archangel needs no see behind him. This is the icon a church has
				# before it has anything else.
				allow = {
					always = yes
				}
				visible = { always = yes }

				ai_will_do = {
					factor = 0.05
					modifier = {
						factor = 0
						is_at_war = no
					}
					modifier = {
						factor = 3
						is_in_important_war = yes
					}
				}
			}

			rip_ro_icon_eleusa = {
				global_unrest = -3
				harsh_treatment_cost = -0.25
				tolerance_own = 1	# from blessing_of_eastern_liturgy, halved

				# HOLY SITES UNLOCK THIS ICON.
				# Held from the first, but only by a church with parishes of its own
				# to carry her through. Three is a metropolitanate, not a chapel.
				allow = {
					num_of_owned_provinces_with = {
						value = 3
						religion = russian_orthodox
					}
				}
				visible = { always = yes }

				ai_will_do = {
					factor = 0.02
					modifier = {
						factor = 4
						NOT = { stability = 0 }
					}
					modifier = {
						factor = 3
						num_of_rebel_armies = 1
					}
				}
			}

			rip_ro_icon_pancreator = {
				development_cost = -0.1
				build_cost = -0.1
				church_loyalty_modifier = 0.1	# from blessing_of_patriarch_of_moscow

				# HOLY SITES UNLOCK THIS ICON.
				# Moscow. The Almighty over the builders is the Third Rome's own icon and
				# it belongs to whoever holds the city.
				allow = {
					owns_or_non_sovereign_subject_of = 295
				}
				visible = { always = yes }

				ai_will_do = {
					factor = 0.05
					modifier = {
						factor = 2
						is_at_war = no
					}
				}
			}

			rip_ro_icon_nicholas = {
				improve_relation_modifier = 0.25
				ae_impact = -0.1
				prestige_decay = -0.01	# from blessing_of_third_rome

				# HOLY SITES UNLOCK THIS ICON.
				# Novgorod or Kyiv - the two sees that traded and negotiated for the Rus
				# long before Moscow did.
				allow = {
					OR = { owns_or_non_sovereign_subject_of = 310 owns_or_non_sovereign_subject_of = 280 }
				}
				visible = { always = yes }

				ai_will_do = {
					factor = 0.03
					modifier = {
						factor = 3
						is_at_war = no
					}
				}
			}

			rip_ro_icon_climacus = {
				global_institution_spread = 0.25
				embracement_cost = -0.2
				culture_conversion_cost = -0.15	# from blessing_of_saints_cyril_and_methodius

				# HOLY SITES UNLOCK THIS ICON.
				# Constantinople or Jerusalem. The Ladder is a Sinai book and the learning
				# it stands for comes up the pilgrim roads, not out of the forest.
				allow = {
					OR = { owns_or_non_sovereign_subject_of = 151 owns_or_non_sovereign_subject_of = 379 }
				}
				visible = { always = yes }

				ai_will_do = {
					factor = 0.03
					modifier = {
						factor = 4
						NOT = { has_institution = printing_press }
					}
				}
			}
		}

		# The Sobor.
		#
		# Vanilla fervour is exactly the shape the Sobor already had in script:
		# a pool that fills by itself, buys one of three exclusive focuses, and
		# drains while that focus runs. It is scriptable - common/fervor - so the
		# hand-rolled version is retired: decisions/RIP_SoborZeal.txt is deleted,
		# and with it the rip_ro_zeal variable, its two-yearly tick and the five
		# rip_ro_zeal_band_* modifiers that faked a progress bar out of tooltips.
		fervor = yes

		# Five sites, as vanilla does it.
		#
		# Both vanilla holy-site religions declare exactly five - coptic and
		# zoroastrian - and interface/countryreligionview.gui has widgets for
		# exactly five: holy_site_icon_1..5 and holy_site_name_1..5. A sixth
		# still works in the engine, but its row has nowhere to draw in the
		# small panel, which is what the game's own error.log was reporting:
		# "Missing Icon 'holy_site_icon_6' in window 'coptic_specific_window'".
		# Showing it would mean overriding the whole 3486-line vanilla file and
		# relocating the blessings box, for one row.
		#
		# Rome is the site that left, and it is the right one to drop: it is the
		# only see here the Muscovite church never held or claimed to hold. The
		# doctrine this faith runs on says Rome fell, which is the whole point of
		# there being a third one. Holding Rome is a trophy instead - see
		# rip_ro_the_first_rome in decisions/RIP_OrthodoxIcons.txt.
		holy_sites = {
			310 # Novgorod - the Sophia of the north, and the older see
			295 # Moscow - the Third Rome
			280 # Kyiv - where the baptism happened
			151 # Constantinople - the Second Rome, which fell
			379 # Jerusalem - the beginning
		}

		# THE BLESSINGS ARE GONE, AND THE ICONS TOOK THEM.
		#
		# Seven blessings, one running for every holy site held, stacked to five
		# at the ceiling. That is five permanent country modifiers on top of a
		# standing icon, which is more than any vanilla faith carries and more
		# than this one should. They are cut.
		#
		# Nothing good is lost: the five best payloads are folded into the five
		# icons below, one apiece. Because exactly ONE icon stands at a time,
		# the faith now carries one modifier where it used to carry six, and the
		# choice between them is a real choice rather than a collection.
		#
		# Cut outright as the strongest and least interesting: the missionary
		# strength of blessing_of_religious_unity and the -25% culture
		# conversion of blessing_of_cultural_dominance.
		#
		# common/church_aspects/RIP_blessings.txt still defines them - nothing
		# references it now, and it is left in place so the payloads stay
		# readable next to the icons that inherited them.

		heretic = { BOGOMILIST }

		# THE THREE DATES OF THIS CHURCH
		#
		# 1448.12.15 - the metropolitan is elected at Moscow without
		#              Constantinople's blessing, and the metropolitanate splits
		#              along the border it will keep. This is the birth of the
		#              faith, so it is the religion's own date: `date` is what
		#              is_religion_enabled reads, and setting it to 1589 left the
		#              mod offering a religion the engine had not switched on for
		#              another 136 years - convert_to_russian_orthodox_decision
		#              opens at 1450 and russian_orthodox.1 fired from 1453.
		# 1589.1.26  - the metropolitan becomes a patriarch: russian_orthodox.2.
		# 1653       - Nikon's books, and the Raskol: russian_orthodox.6.
		#
		# The three are a ladder, not three separate stories. Each stage sets its
		# flag and the next one asks for it:
		#   rip_ro_stage_autocephaly -> rip_ro_stage_patriarchate -> rip_ro_stage_raskol
		date = 1448.12.15

		on_convert = {
			add_prestige = 15
			add_legitimacy = 10

			# The trade node policy Propagate Religion, unlocked once and for
			# good - see common/scripted_effects/rip_faith_spread_effects.txt.
			rip_faith_unlock_trade_propagation_effect = yes
			add_country_modifier = {
				name = third_rome_ideology
				duration = 5475
			}
		}
	}
}
