# Greek Catholic Curia petitions have their own bounded slot.
rip_church_gc_curia_contact = {
	religion = greek_catholic
	exists = PAP
	PAP = { religion = catholic }
	NOT = { war_with = PAP }
}
rip_church_gc_has_curia_petition = {
	OR = {
		has_country_modifier = rip_church_gc_petition_church_tax
		has_country_modifier = rip_church_gc_petition_blessing
		has_country_modifier = rip_church_gc_petition_indulgence
		has_country_modifier = rip_church_gc_petition_saint
		has_country_modifier = rip_church_gc_petition_usury
		has_country_modifier = rip_church_gc_petition_holy_war
		has_country_modifier = rip_church_gc_petition_legate
		has_country_modifier = rip_church_gc_petition_monopoly
	}
}
rip_church_gc_can_petition_base = {
	rip_church_gc_curia_contact = yes
	NOT = { rip_church_gc_has_curia_petition = yes }
	NOT = { check_variable = { which = rip_church_communion value = -40.001 } }
	treasury = 100
	has_opinion = { who = PAP value = 50 }
}
rip_church_gc_can_petition_church_tax = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } }
rip_church_gc_can_petition_blessing = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } }
rip_church_gc_can_petition_indulgence = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } }
rip_church_gc_can_petition_saint = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 60 } NOT = { stability = 3 } }
rip_church_gc_can_petition_usury = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } }
rip_church_gc_can_petition_holy_war = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } is_at_war = yes }
rip_church_gc_can_petition_legate = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } }
rip_church_gc_can_petition_monopoly = { rip_church_gc_can_petition_base = yes check_variable = { which = rip_church_papal_standing value = 30 } NOT = { mercantilism = 100 } }

# This is a diplomatic audience, not a cardinal appointment or Curia vote.
rip_church_gc_can_depute_to_curia = {
	rip_church_gc_curia_contact = yes
	is_at_war = no
	treasury = 50
	patriarch_authority = 0.1
	NOT = { had_country_flag = { flag = rip_church_gc_deputation_sent days = 1825 } }
}
rip_church_gc_has_active_icon = {
	OR = {
		has_country_modifier = rip_church_gc_icon_liturgy
		has_country_modifier = rip_church_gc_icon_learning
		has_country_modifier = rip_church_gc_icon_charity
	}
}
rip_church_gc_can_activate_icons = {
	religion = greek_catholic
	patriarch_authority = 0.2
	NOT = { rip_church_gc_has_active_icon = yes }
}
