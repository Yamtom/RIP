# Country scope. Church-estate privileges that react to the Greek Catholic
# (Uniate) faith. See common/estate_privileges/RIP_greek_catholic_privileges.txt.

# A Latin or Orthodox crown that holds Greek Catholic parishes.
rip_gc_owns_greek_catholic_province = {
	any_owned_province = {
		religion = greek_catholic
	}
}

	any_owned_province = {
		OR = {

			religion = russian_orthodox
		}
	}
}
