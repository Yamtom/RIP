# RIP - Church-estate privileges that react to the Greek Catholic faith.
#
# Three groups, one question each:
#   Catholic crowns holding Greek Catholic parishes: protect the Eastern rite
#   or latinise it (the Synod of Zamosc, 1720, took the second road).
#   Orthodox crowns holding Greek Catholic parishes: brotherhood charters
#   against the Union, or a concord with its parishes.
#   Greek Catholic crowns: how the Union stands toward Rome and toward the
#   Orthodox, and the Basilian schools that served both.
# The numbers are gameplay values, not a documented historical formula.

rip_privilege_eastern_rite_eparchies = {
	icon = privilege_grant_autonomy
	max_absolutism = -5
	loyalty = 0.1
	influence = 0.05

	is_valid = {
		religion = catholic
		rip_gc_owns_greek_catholic_province = yes
		NOT = { has_estate_privilege = rip_privilege_latin_rite_primacy }
	}

	can_select = {
		religion = catholic
		rip_gc_owns_greek_catholic_province = yes
		NOT = { has_estate_privilege = rip_privilege_latin_rite_primacy }
	}

	penalties = {
		stability_cost_modifier = 0.05
	}

	benefits = {
		tolerance_heretic = 1
		improve_relation_modifier = 0.1
	}

	ai_will_do = {
		factor = 10
		modifier = {
			factor = 3
			is_at_war = no
		}
	}
}
