rip_church_gc_donate_effect = {
 if = { limit = { rip_church_gc_can_donate = yes }
  add_treasury = -100
  if = { limit = { has_dlc = "Emperor" }
   add_curia_treasury = 50
   PAP = { add_treasury = 50 }
  }
  else = { PAP = { add_treasury = 100 } }
  change_variable = { which = rip_church_papal_standing value = 10 }
  PAP = { add_opinion = { who = ROOT modifier = rip_church_gc_donation_opinion } }
  set_country_flag = rip_church_gc_donated
  rip_church_gc_refresh_rome_effect = yes
 }
}
# Shared targets identify world offices, not the currently selected player.
# The displayed opinion is country-local and is never shared between GC states.
rip_church_gc_refresh_rome_effect = {
 clear_global_event_target = rip_church_gc_rome
 clear_global_event_target = rip_church_gc_controller
 clr_global_flag = rip_church_gc_rome_known
 clr_global_flag = rip_church_gc_controller_known
 if = { limit = { rip_church_gc_rome_present = yes }
  PAP = { save_global_event_target_as = rip_church_gc_rome }
  set_global_flag = rip_church_gc_rome_known
 }
 every_country = {
  limit = { is_papal_controller = yes }
  save_global_event_target_as = rip_church_gc_controller
  set_global_flag = rip_church_gc_controller_known
 }
 if = { limit = { religion = greek_catholic }
  set_variable = { which = rip_church_papal_opinion value = 0 }
  if = { limit = { rip_church_gc_rome_present = yes }
   export_to_variable = { which = rip_church_papal_opinion value = opinion who = PAP with = ROOT }
  }
 }
}
