# --- 1. Second Serfdom & Folwark Corvee

country_event = {
	id = rip_plc_reckoning.1
	title = rip_plc_reckoning.1.t
	desc = rip_plc_reckoning.1.d
	picture = FARMING_eventPicture
	is_triggered_only = yes

	option = {
		name = rip_plc_reckoning.1.a
		custom_tooltip = rip_plc_reckoning.1.result_tt
		ai_chance = { factor = 2 }
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = { estate = estate_nobles loyalty = 15 }
		}
		add_treasury = 80
		set_country_flag = rip_plc_magnate_pacification_enforced
		country_event = { id = rip_plc_reckoning.2 days = 180 }
	}
	option = {
		name = rip_plc_reckoning.1.b
		ai_chance = { factor = 1 }
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = { estate = estate_nobles loyalty = -15 }
		}
		add_adm_power = -25
		add_country_modifier = {
			name = rip_plc_capped_corvee
			duration = 5475
		}
	}
}
