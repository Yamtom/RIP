country_event = {
	id = rip_vol_alt_history.3
	title = rip_vol_alt_history.3.t
	desc = rip_vol_alt_history.3.d
	picture = RELIGION_eventPicture
	is_triggered_only = yes
	# Alternative history: the synod is a game model, not a recorded council.
	# Every option leaves rip_vol_synod_held behind so later Volhynia content
	# can tell that the confessional question was settled. The reward is the
	# recipient's own religious currency (rip_faith_mission_resource_effect):
	# a Greek Catholic state has no papal influence to spend, and gets
	# patriarch_authority instead.
	option = {
		name = rip_vol_alt_history.3.a
		# Orthodox Foundation
		set_country_flag = rip_vol_synod_held
		rip_faith_mission_resource_effect = {
			PA = 0.2
			PI = 20
			ADM = 25
		}
	}
	option = {
		name = rip_vol_alt_history.3.b
		# Uniate Synthesis (Greek-Catholic)
		set_country_flag = rip_vol_synod_held
		rip_faith_mission_resource_effect = {
			PA = 0.2
			PI = 50
			ADM = 50
		}
	}
	option = {
		name = rip_vol_alt_history.3.c
		# Pluralist Balance
		set_country_flag = rip_vol_synod_held
		add_dip_power = 50
	}
}

country_event = {
	id = rip_vol_alt_history.9
	title = rip_vol_alt_history.9.t
	desc = rip_vol_alt_history.9.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	option = {
		name = rip_vol_alt_history.9.a
		trigger = { is_subject = no FROM = { exists = yes alliance_with = ROOT is_subject = no } }
		ai_chance = { factor = 70 }
		add_dip_power = -50
		remove_country_modifier = rip_vol_carpathian_compact
		add_country_modifier = { name = rip_vol_federal_compact duration = 7300 }
		add_trust = { who = FROM value = 10 }
		add_favors = { who = FROM amount = 15 }
		FROM = {
			set_country_flag = rip_vol_federal_compact_accepted
			add_dip_power = -50
			remove_country_modifier = rip_vol_carpathian_compact
			add_country_modifier = { name = rip_vol_federal_compact duration = 7300 }
			add_trust = { who = ROOT value = 10 }
			add_favors = { who = ROOT amount = 15 }
		}
	}
	option = {
		name = rip_vol_alt_history.9.b
		ai_chance = { factor = 30 }
		add_prestige = 5
		FROM = { add_dip_power = 25 }
	}
}
