# Chronology is historical; alternative options and numerical costs are design.
namespace = rip_ro_history

# 1653: correcting service books is distinct from the council's later rupture.
country_event = {
 id = rip_ro_history.1
 title = rip_ro_history.1.t desc = rip_ro_history.1.d
 picture = REFORM_eventPicture
 fire_only_once = yes
 trigger = {
  rip_church_ro_jurisdiction = yes has_country_flag = rip_ro_stage_patriarchate
  OR = { tag = MOS tag = RUS }
  is_year = 1653 NOT = { is_year = 1700 }
  NOT = { has_country_flag = rip_ro_books_reformed }
  NOT = { has_country_flag = rip_ro_ritual_compromise }
 }
 mean_time_to_happen = { months = 18 }
 option = {
  name = rip_ro_history.1.a ai_chance = { factor = 80 }
  set_country_flag = rip_ro_books_reformed
  add_patriarch_authority = 0.05
 }
 option = {
  name = rip_ro_history.1.b ai_chance = { factor = 20 }
  rip_church_ro_ritual_compromise_effect = yes
 }
}
