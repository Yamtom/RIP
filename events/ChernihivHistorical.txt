# ========================================================================
# Historical processes of the Siverian land, in the order they happened.
# Authenticity: A - documented, with the choices left open.
# Sources:
#   the service princes Mozhaisky and Shemyachich, who held Chernihiv and
#   Novhorod-Siverskyi between Vilnius and Moscow and changed sides twice;
#   the Muscovite watch line along the Desna from the 1490s;
#   the Starodub hemp and potash trade to Riga and Danzig;
#   Magdeburg charters for Chernihiv (1623) and Nizhyn (1625);
#   the Chernihiv voivodeship of 1635 and the szlachta grants inside it;
#   the regimental order of 1648 that replaced it;
#   Baturyn as the hetman's seat and its building in brick;
#   Doroshenko's protection from the Porte after 1669;
#   the Chernihiv Collegium of 1700 and the chancery men it trained;
#   the codification finished in 1743 as the Rights by which the land is
#   judged.
# ========================================================================

namespace = chr_history

# --- 1. The service princes. They came from Moscow with retinues and a claim
#        to udils, and whoever took them in also took in their quarrel.
country_event = {
	id = chr_history.1
	title = chr_history.1.t
	desc = chr_history.1.d
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = CHR
			chr_on_a_path = yes
		}
		NOT = { is_year = 1490 }
		owns_core_province = 289	# Chernihiv
	}

	mean_time_to_happen = {
		months = 60
	}

	option = {
		name = chr_history.1.a
		ai_chance = { factor = 40 }
		add_country_modifier = {
			name = chr_service_princes
			duration = -1
		}
		add_mil_power = 25
	}

	option = {
		name = chr_history.1.b
		ai_chance = { factor = 40 }
		add_republican_tradition = 5
		add_adm_power = 25
		if = {
			limit = { has_estate = estate_nobles }
			add_estate_loyalty = {
				estate = estate_nobles
				loyalty = -10
			}
		}
	}

	option = {
		name = chr_history.1.c
		ai_chance = { factor = 20 }
		add_dip_power = 50
		add_prestige = -5
	}
}

# --- 2. The Desna becomes a border. Muscovite watchtowers went up along it in
#        the 1490s and the towns behind them had to decide what they were.
country_event = {
	id = chr_history.2
	title = chr_history.2.t
	desc = chr_history.2.d
	picture = FORT_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = CHR
			chr_on_a_path = yes
		}
		is_year = 1490
		NOT = { is_year = 1545 }
		owns_core_province = 289	# Chernihiv
	}

	mean_time_to_happen = {
		months = 48
		modifier = {
			factor = 0.6
			any_neighbor_country = {
				OR = {
					tag = MOS
					tag = RUS
				}
			}
		}
	}

	option = {
		name = chr_history.2.a
		ai_chance = { factor = 50 }
		add_mil_power = -50
		add_country_modifier = {
			name = chr_desna_line
			duration = 9125
		}
	}

	option = {
		name = chr_history.2.b
		ai_chance = { factor = 30 }
		add_treasury = -100
		add_dip_power = 25
		random_country = {
			limit = {
				OR = {
					tag = MOS
					tag = RUS
				}
				NOT = { war_with = ROOT }
			}
			add_opinion = {
				who = ROOT
				modifier = chr_border_compact_support
			}
		}
	}

	option = {
		name = chr_history.2.c
		ai_chance = { factor = 20 }
		add_adm_power = 50
		add_prestige = -10
		every_owned_province = {
			limit = { is_state_core = ROOT }
			add_unrest = 2
		}
	}
}

# --- 3. Starodub hemp. Rope for other people's fleets, carted north to Riga
#        and Danzig, was the one export the Siverian land could not be talked
#        out of.
country_event = {
	id = chr_history.3
	title = chr_history.3.t
	desc = chr_history.3.d
	picture = TRADEGOODS_eventPicture

	fire_only_once = yes

	trigger = {
		OR = {
			tag = CHR
			chr_on_a_path = yes
		}
		is_year = 1550
		NOT = { is_year = 1660 }
		owns = 4244	# Starodub
	}

	mean_time_to_happen = {
		months = 72
	}

	option = {
		name = chr_history.3.a
		ai_chance = { factor = 50 }
		add_country_modifier = {
			name = chr_hemp_contract
			duration = -1
		}
	}

	option = {
		name = chr_history.3.b
		ai_chance = { factor = 30 }
		add_treasury = -150
		4244 = {			# Starodub
			add_base_production = 1
		}
	}

	option = {
		name = chr_history.3.c
		ai_chance = { factor = 20 }
		add_years_of_income = 0.5
		every_owned_province = {
			limit = { trade_goods = naval_supplies }
			add_unrest = 3
		}
	}
}
