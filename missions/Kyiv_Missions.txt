	KIE_byzantine_relations = {
		icon = carolingian_empire
		required_missions = { KIE_true_ruler }
		position = 2
		provinces_to_highlight = {
			OR = {
				owned_by = BYZ
				AND = {
					province_id = 280
					ROOT = { NOT = { exists = BYZ } }
				}
			}
		}
		trigger = {
			if = {
				limit = {
					exists = BYZ
				}
				OR = {
					alliance_with = BYZ
					AND = {
						BYZ = {
							custom_trigger_tooltip = {
								tooltip = rivalled_or_rival_of
								OR = {
									is_rival = root
									root = {
										is_rival = prev
									}
								}
							}
						}
						# Rivalry alone is not the quarrel. The Kyiv court has to say
						# aloud what it claims, and pay for it, through the decision
						# rip_kie_declare_primacy (kyiv_choices.4).
						custom_trigger_tooltip = {
							tooltip = KIE_byzantine_primacy_declared_tt
							has_country_flag = rip_kie_byz_primacy_declared
						}
					}
				}
			}
			else = {
				advisor = theologian
				280 = {
					owned_by = ROOT
					OR = {
						has_building = temple
						has_building = cathedral
					}
				}
			}
		}
		effect = {
			add_dip_power = 50
			add_adm_power = 50
			if = {
				limit = {
					alliance_with = BYZ
				}
				add_historical_friend = BYZ
			}
		}
	}
