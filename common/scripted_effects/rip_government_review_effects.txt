# Country-scope effects of the institutional resolutions. The six resolutions
# themselves are the effects of the buttons in
# common/government_mechanics/rip_institutional_resolutions.txt: each pays its
# monarch power, calls the shared payment below and adds its ten-year modifier.

# The ducats and the shared lock. The flag is dated by the engine; the status
# panel counts the years left from it, and the buttons' trigger refuses a new
# resolution until it is ten years old (rip_gov_resolution_lock_over).
rip_gov_review_cost_effect = {
	add_treasury = -50
	set_country_flag = rip_gov_review_recent
}

rip_gov_clear_municipal_effect = {
	remove_country_modifier = rip_gov_municipal_charters
	remove_country_modifier = rip_gov_municipal_accounts
	# Never clear rip_gov_review_recent: replacing a reform cannot reset time.
}

rip_gov_clear_chancery_effect = {
	remove_country_modifier = rip_gov_chancery_service
	remove_country_modifier = rip_gov_chancery_rights
	# Never clear rip_gov_review_recent: replacing a reform cannot reset time.
}

rip_gov_clear_register_effect = {
	remove_country_modifier = rip_gov_register_muster
	remove_country_modifier = rip_gov_register_settlement
	# Never clear rip_gov_review_recent: replacing a reform cannot reset time.
}

rip_gov_review_cleanup_effect = {
	if = {
		limit = { NOT = { rip_gov_has_municipal_review = yes } }
		rip_gov_clear_municipal_effect = yes
	}
	if = {
		limit = { NOT = { rip_gov_has_chancery_review = yes } }
		rip_gov_clear_chancery_effect = yes
	}
	if = {
		limit = { NOT = { rip_gov_has_register_review = yes } }
		rip_gov_clear_register_effect = yes
	}
}
