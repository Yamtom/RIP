namespace = zaz_monument

# --- 1. The Sich is walled. Fired from the monument's on_built: unlike
#        Palanok this one is constructed during play - a 1.37.5 save has
#        the AI finishing it on 1556.12.30, four years after its date.
#        UNVERIFIED IN GAME: no vanilla monument fires an event from any hook,
#        so on_built here and on_upgraded below both still need one run.
country_event = {
	id = zaz_monument.1
	title = zaz_monument.1.t
	desc = zaz_monument.1.d
	picture = MILITARY_CAMP_eventPicture

	is_triggered_only = yes

	option = {
		name = zaz_monument.1.a
		ai_chance = { factor = 60 }
		add_country_modifier = {
			name = zaz_cossack_democracy
			duration = -1
		}
		rip_estate_mood_effect = { ESTATE = estate_cossacks LOYALTY = 10 }
	}

	option = {
		name = zaz_monument.1.b
		ai_chance = { factor = 40 }
		add_army_tradition = 10
		add_mil_power = 50
		rip_estate_mood_effect = { ESTATE = estate_cossacks LOYALTY = -5 }
	}
}
