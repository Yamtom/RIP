kyivan_rus_reform = {
	icon = "chieftain"
	legacy_equivalent = tsardom_legacy
	potential = {
		OR = {
			has_reform = kyivan_rus_reform
			tag = KIE
			tag = KRU
		}
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = yes
	lock_level_when_selected = yes
	queen = yes
	royal_marriage = yes
	heir = yes
	rulers_can_be_generals = yes
	heirs_can_be_generals = yes
	allow_convert = no
	allow_force_tributary = yes
	allow_vassal_war = yes
	allow_vassal_alliance = yes
	trigger = {
		OR = {
			has_reform = kyivan_rus_reform
			tag = KRU
			tag = KIE
		}
	}
	modifiers = {
		global_autonomy = -0.025
		global_manpower_modifier = 0.15
		core_creation = -0.10
		vassal_forcelimit_bonus = 0.15
	}
	custom_attributes = {
		locked_government_type = yes
	}
	conditional = {
		allow = { has_dlc = "Third Rome" }
		claim_states = yes
	}
	# Seniority of the Rus, not the Tsar's toolkit. russian_mechanic (Sudebnik /
	# Oprichnina / Streltsy) is a 16th-17th century anachronism for the Kyivan
	# order; kyivan_seniority_mechanic models the rota, the snem and the
	# druzhyna levy. common/government_mechanics/rip_kyivan_seniority.txt.
	conditional = {
		allow = {
			OR = {
				has_dlc = "Third Rome"
				has_dlc = "Domination"
			}
		}
		government_abilities = {
			kyivan_seniority_mechanic
		}
	}
}

kyivan_shogunate_reform = {
	icon = "shogunate"
	legacy_equivalent = shogunate_legacy
	potential = {
		OR = {
			has_reform = kyivan_shogunate_reform
			tag = KIE
		}
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = yes
	lock_level_when_selected = yes
	queen = yes
	heir = yes
	royal_marriage = yes
	rulers_can_be_generals = yes
	heirs_can_be_generals = yes
	allow_convert = no
	allow_vassal_war = yes
	allow_vassal_alliance = yes
	allow_force_tributary = yes
	conditional = {
		allow = {
			has_dlc = "Res Publica"
			NOT = { has_dlc = "Mare Nostrum" }
		}
		can_use_trade_post = yes
	}
	conditional = {
		allow = {
			has_dlc = "Mare Nostrum"
			NOT = { has_dlc = "Res Publica" }
		}
		can_form_trade_league = yes
	}
	conditional = {
		allow = {
			has_dlc = "Res Publica"
			has_dlc = "Mare Nostrum"
		}
		can_use_trade_post = yes
		can_form_trade_league = yes
	}
	# The Seniorate, not the Bakufu. shogunate_mechanic is the Japanese daimyo
	# UI; kyivan_seniorate_mechanic works the senior_udil subjects - the rota,
	# the levy, and the reconfirmation of grants on succession.
	# common/government_mechanics/rip_kyivan_seniorate.txt.
	conditional = {
		allow = {
			OR = {
				has_dlc = "Third Rome"
				has_dlc = "Domination"
			}
		}
		government_abilities = {
			kyivan_seniorate_mechanic
		}
	}
	trigger = {
		OR = {
			has_reform = kyivan_shogunate_reform
			tag = KIE
		}
	}
	modifiers = {
		vassal_forcelimit_bonus = 0.15
		diplomatic_reputation = 1
		max_absolutism = -10
	}
	custom_attributes = {
		locked_government_type = yes
	}
}
