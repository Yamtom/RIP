# The three roads out of the Siversk Veche.
#
# Chernihiv starts as a merchant veche of chartered towns and does not have to
# stay one. Each road is a tier-1 government type in three stages, taken
# through decisions in decisions/ChernihivPaths.txt. The flag chr_path_chosen
# shuts the other two roads the moment one is opened; stage 3 of every road
# waits on Great Chernihiv.
#
# Every reform carries a potential block, or EU4 lists it in the reform slot of
# every country in the game and merely greys it out. Every reform is registered
# in common/governments/00_governments.txt, or it is never offered at all.
#
# Stage 1 locks its level, so the tier cannot be swapped from the government
# view; the later stages arrive only through their decision.

# ============================ ROAD A - THE TOWNS ============================
# The road Chernihiv is already on: chartered towns on a river system, a staple
# on the Desna, and an assembly that bargains rather than commands. It reaches
# for what the Dutch reached and stops one step short of it, on purpose - there
# is no sea at the end of the Desna.

chr_desna_staple_reform = {
	icon = "trading_city"
	legacy_equivalent = merchant_republic
	potential = {
		chr_desna_staple_visible = yes
	}
	trigger = {
		chr_desna_staple_visible = yes
	}
	allow_normal_conversion = no
	valid_for_nation_designer = no
	valid_for_new_country = no
	lock_level_when_selected = yes
	republic = yes
	republican_name = yes
	has_term_election = yes
	duration = 4
	queen = no
	heir = no
	royal_marriage = no
	rulers_can_be_generals = yes
	boost_income = yes
	modifiers = {
		merchants = 1
		global_trade_power = 0.1
		trade_efficiency = 0.05
		burghers_loyalty_modifier = 0.1

}
