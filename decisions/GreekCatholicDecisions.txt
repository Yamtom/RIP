
country_decisions = {

	convert_to_greek_catholic_decision = {
		major = yes
		# The Union of Brest was an ORTHODOX hierarchy accepting Rome while keeping
		# the Byzantine rite. A Latin-rite country has no rite to keep and no reason
		# to take an eastern one, and none ever did - Rome's interest in the union
		# was precisely that it did not have to latinise anybody. Catholic sponsors
		# press the union onto their Orthodox subjects instead; that is
		# rip_uc_press_the_union_effect, and it is a different action.
		#
		# Alternative history: the effect below changes the faith of the WHOLE
		# state, which no crown did. The player-visible marker for a state with no
		# Catholic overlord is STATE_UNION_ALT in tools/build_church_localisation.py.
		potential = {
			religion = orthodox
			# The union as an ACT OF STATE belongs to a state whose weight is
			# in these lands. An empire that merely holds Ruthenia receives it
			# through the locally available historical conversion routes.
			rip_uc_is_regional_state = yes
			# Florence is a separate early route; this decision concludes the
			# later Union of Brest and must not fire before its 1596 date.
			is_year = 1596
		}
		
		provinces_to_highlight = {
			owned_by = ROOT
			OR = {
				religion = orthodox
				religion = catholic
			}
		}
		
		allow = {
			is_religion_enabled = greek_catholic
			is_at_war = no
			stability = 1
			adm_power = 100
			dip_power = 100
			# Brest 1596, Uzhhorod 1646, Peremyshl 1691, Lviv 1700, Lutsk 1702: every
			# one of them was signed by bishops living under a Catholic crown, and
			# none by a church that answered to nobody.
			OR = {
				has_country_flag = pursuing_uniate_union
				has_country_flag = uniate_pressure_mounting
				overlord = { religion = catholic }
				AND = {
					any_neighbor_country = {
						religion = catholic
						NOT = { war_with = ROOT }
					}
					any_owned_province = { religion = greek_catholic }
				}
			}
			# and by sees that had Ruthenian parishes to bring with them
			num_of_owned_provinces_with = {
				value = 3
				religion = orthodox
				OR = {
					culture = ruthenian
					culture = byelorussian
					culture = rusyn
					culture = rusyn_new
					culture = rusyn_new_new
				}
			}
		}

		effect = {
			add_adm_power = -100
			add_dip_power = -100
			set_country_flag = pursuing_uniate_union
			rip_faith_adopt_union_effect = yes
			set_country_flag = union_of_brest_happened
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 2
				has_country_flag = pursuing_uniate_union
			}
			modifier = {
				factor = 0.5
				stability = 2
			}
		}
	}
